# Qt容器类QList与QVector对比

## 题目

Qt中QList和QVector有什么区别？如何选择使用哪个容器类？

## 答案

### 基本区别

**QList特点：**
- 基于索引标签存储元素
- 插入、追加操作速度快
- 适合频繁更新数据的场景
- 内存不连续

**QVector特点：**
- 连续内存空间存储
- 访问数据速度快
- 适合频繁读取数据的场景
- 内存连续，类似数组

### 性能对比

class ContainerBenchmark : public QObject {
    Q_OBJECT

public:
    ContainerBenchmark(QObject *parent = nullptr) : QObject(parent) {
        count = 100000;

        QTime timer;

            vec2.insert(0, i);

        timer.restart();

        for (int i = 0; i < count; i++) {

        }

    }

### QList使用场景

```cpp
class ListUsage : public QObject {
    Q_OBJECT

public:
    ListUsage(QObject *parent = nullptr) : QObject(parent) {}

    void demonstrateList() {
        QStringList list;
        
        //往后追加
        list.append("aaa");
        list.append("bbb");
        
        //往前追加
        list.prepend("xxx");
        
        //使用push_back（等同于append）
        list.push_back("ccc");
        
        //使用push_front（等同于prepend）
        list.push_front("yyy");
        
        //指定位置插入
        list.insert(0, "zzz");
        
        //删除元素
        list.removeOne("bbb");
        list.removeAll("xxx");
        
        //清空
        list.clear();
        
        //遍历
        list << "item1" << "item2" << "item3";
        
        //方式1：使用迭代器
        QList<QString>::iterator it;
        for (it = list.begin(); it != list.end(); ++it) {
            qDebug() << *it;
        }
        
        //方式2：使用at()
        for (int i = 0; i < list.count(); i++) {
            qDebug() << list.at(i);
        }
        
        //方式3：使用foreach
        foreach (const QString &str, list) {
            qDebug() << str;
        }
        
        //方式4：使用for循环
        for (const QString &str : list) {
            qDebug() << str;
        }
    }
};
```

### QVector使用场景

```cpp
class VectorUsage : public QObject {
    Q_OBJECT

public:
    VectorUsage(QObject *parent = nullptr) : QObject(parent) {}

    void demonstrateVector() {
        QVector<int> vec;
        
        //预留空间
        vec.reserve(1000);
        
        //填充数据
        for (int i = 0; i < 1000; i++) {
            vec.append(i);
        }
        
        //访问数据（QVector访问速度快）
        for (int i = 0; i < vec.size(); i++) {
            int value = vec.at(i);
            //处理数据
        }
        
        //使用[]
        vec[0] = 100;
        vec[1] = 200;
        
        //批量操作
        std::fill(vec.begin(), vec.end(), 0);
        
        //排序
        std::sort(vec.begin(), vec.end());
        
        //查找
        QVector<int>::iterator it = std::find(vec.begin(), vec.end(), 500);
        if (it != vec.end()) {
            qDebug() << "Found at position:" << (it - vec.begin());
        }
    }
};
```

### 数据列表管理

```cpp
class DataListManager : public QObject {
    Q_OBJECT

public:
    DataListManager(QObject *parent = nullptr) : QObject(parent) {}

    //使用QList管理数据列表（频繁增删）
    void addData(const QString &name, int value) {
        Data data;
        data.name = name;
        data.value = value;
        
        dataList.append(data);
    }

    void insertData(int index, const QString &name, int value) {
        Data data;
        data.name = name;
        data.value = value;
        
        dataList.insert(index, data);
    }

    void removeData(int index) {
        if (index >= 0 && index < dataList.size()) {
            dataList.removeAt(index);
        }
    }

    void clearData() {
        dataList.clear();
    }

    QList<Data> getDataList() {
        return dataList;
    }

private:
    struct Data {
        QString name;
        int value;
    };
    
    QList<Data> dataList;
};
```

### 内存管理

```cpp
class ContainerMemory : public QObject {
    Q_OBJECT

public:
    ContainerMemory(QObject *parent = nullptr) : QObject(parent) {}

    void compareMemory() {
        QList<int> list;
        QVector<int> vec;
        
        int count = 1000;
        
        for (int i = 0; i < count; i++) {
            list.append(i);
            vec.append(i);
        }
        
        qDebug() << "List size:" << list.size();
        qDebug() << "List capacity:" << (list.capacity() * sizeof(int));
        qDebug() << "Vector size:" << vec.size();
        qDebug() << "Vector capacity:" << (vec.capacity() * sizeof(int));
        
        //QVector连续内存
        const int *vecData = vec.constData();
        qDebug() << "Vector is contiguous:" 
                 << (vecData + 1 == &vec[1]);
    }
};
```

### 总结

**QList vs QVector对比：**

| 特性 | QList | QVector |
|------|-------|---------|
| 插入速度 | 快 | 慢（需要移动元素）|
| 访问速度 | 较快 | 快（连续内存）|
| 内存使用 | 不连续 | 连续 |
| 适用场景 | 频繁增删 | 频繁读取 |
| 迭代器 | 基于索引 | 基于指针 |

**选择建议：**
- 需要频繁插入、删除数据：使用QList
- 需要频繁访问数据：使用QVector
- 图表数据存储：使用QVector
- 列表管理：使用QList
- 数据量小时：两者差别不大

**Qt6变化：**
- QVector = QList（别名）
- 底层统一实现
- 兼具两者优点

**最佳实践：**
- 根据使用场景选择容器
- 图表类使用QVector
- 列表管理使用QList
- 注意内存占用和性能
