# QT信号槽的连接方式与应用场景

## 题目

QT的信号与槽，有哪几种连接方式，对应的应用场景是什么？（第五个参数）

## 答案

QT提供了5种信号槽连接方式，通过`Qt::ConnectionType`枚举指定：

### 1. AutoConnection（自动连接）

**默认连接方式，根据发送者和接收者是否在同一线程自动选择：**

```cpp
#include <QApplication>
#include <QObject>
#include <QThread>
#include <QDebug>

class Sender : public QObject {
    Q_OBJECT
public:
    void sendSignal() {
        emit mySignal("Hello from sender");
    }
    
signals:
    void mySignal(const QString& message);
};

class Receiver : public QObject {
    Q_OBJECT
public slots:
    void receiveSlot(const QString& message) {
        qDebug() << "Received in thread:" << QThread::currentThreadId()
                 << "Message:" << message;
    }
};

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);
    
    Sender sender;
    Receiver receiver;
    
    QObject::connect(&sender, &Sender::mySignal,
                     &receiver, &Receiver::receiveSlot,
                     Qt::AutoConnection);
    
    sender.sendSignal();
    
    return app.exec();
}
```

**行为：**
- 同线程：直接调用（DirectConnection）
- 异线程：队列调用（QueuedConnection）

### 2. DirectConnection（直接连接）

**信号发出时立即执行槽函数，在同一线程同步调用：**

```cpp
#include <QApplication>
#include <QPushButton>
#include <QLabel>
#include <QVBoxLayout>

class MyWidget : public QWidget {
    Q_OBJECT
private:
    QLabel* label;
    
public:
    MyWidget(QWidget* parent = nullptr) : QWidget(parent) {
        QVBoxLayout* layout = new QVBoxLayout(this);
        
        QPushButton* button = new QPushButton("Click Me");
        label = new QLabel("Count: 0");
        
        connect(button, &QPushButton::clicked,
                this, &MyWidget::onButtonClicked,
                Qt::DirectConnection);
        
        layout->addWidget(button);
        layout->addWidget(label);
    }
    
private slots:
    void onButtonClicked() {
        static int count = 0;
        count++;
        label->setText(QString("Count: %1").arg(count));
        qDebug() << "Direct slot called";
    }
};
```

**应用场景：**
- 同线程内的高性能调用
- 需要立即执行的场景
- 信号槽在同一线程

**注意事项：**
- 跨线程使用会导致线程安全问题
- 槽函数在信号发送者线程执行

### 3. QueuedConnection（队列连接）

**信号发出后放入事件队列，异步执行：**

```cpp
#include <QApplication>
#include <QObject>
#include <QThread>
#include <QDebug>

class Worker : public QObject {
    Q_OBJECT
public:
    Worker() {
        qDebug() << "Worker created in thread:" << QThread::currentThreadId();
    }
    
    void doWork() {
        qDebug() << "Working in thread:" << QThread::currentThreadId();
        emit workFinished();
    }
    
signals:
    void workFinished();
};

class Controller : public QObject {
    Q_OBJECT
public:
    Controller() {
        qDebug() << "Controller created in thread:" << QThread::currentThreadId();
    }
    
public slots:
    void onWorkFinished() {
        qDebug() << "Slot called in thread:" << QThread::currentThreadId();
    }
};

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);
    
    QThread workerThread;
    Worker worker;
    Controller controller;
    
    worker.moveToThread(&workerThread);
    
    QObject::connect(&workerThread, &QThread::started,
                     &worker, &Worker::doWork);
    
    QObject::connect(&worker, &Worker::workFinished,
                     &controller, &Controller::onWorkFinished,
                     Qt::QueuedConnection);
    
    QObject::connect(&worker, &Worker::workFinished,
                     &workerThread, &QThread::quit);
    
    workerThread.start();
    workerThread.wait();
    
    return 0;
}
```

**应用场景：**
- 跨线程通信
- 异步处理
- 避免界面线程阻塞

### 4. BlockingQueuedConnection（阻塞队列连接）

**队列连接的阻塞版本，等待槽函数执行完成：**

```cpp
#include <QApplication>
#include <QThread>
#include <QDebug>

class Sender : public QObject {
    Q_OBJECT
public:
    void sendBlockingSignal() {
        qDebug() << "Signal sent from:" << QThread::currentThreadId();
        emit blockingSignal();
        qDebug() << "Signal processing completed";
    }
    
signals:
    void blockingSignal();
};

class Receiver : public QObject {
    Q_OBJECT
public slots:
    void onBlockingSignal() {
        qDebug() << "Slot called in:" << QThread::currentThreadId();
        QThread::sleep(2);
        qDebug() << "Slot completed";
    }
};

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);
    
    QThread receiverThread;
    Sender sender;
    Receiver receiver;
    
    receiver.moveToThread(&receiverThread);
    
    QObject::connect(&sender, &Sender::blockingSignal,
                     &receiver, &Receiver::onBlockingSignal,
                     Qt::BlockingQueuedConnection);
    
    receiverThread.start();
    sender.sendBlockingSignal();
    
    receiverThread.quit();
    receiverThread.wait();
    
    return 0;
}
```

**应用场景：**
- 需要等待结果
- 同步跨线程调用
- 避免竞态条件

**注意事项：**
- 不能在同一线程中使用（会死锁）
- 可能导致性能问题

### 5. UniqueConnection（唯一连接）

**确保信号槽只连接一次：**

```cpp
#include <QApplication>
#include <QPushButton>
#include <QLabel>
#include <QVBoxLayout>

class MyWidget : public QWidget {
    Q_OBJECT
private:
    QLabel* label;
    int clickCount;
    
public:
    MyWidget(QWidget* parent = nullptr) : QWidget(parent), clickCount(0) {
        QVBoxLayout* layout = new QVBoxLayout(this);
        
        QPushButton* button = new QPushButton("Setup Connection");
        label = new QLabel("Clicks: 0");
        
        connect(button, &QPushButton::clicked,
                this, &MyWidget::setupConnection);
        
        layout->addWidget(button);
        layout->addWidget(label);
    }
    
private slots:
    void setupConnection() {
        QPushButton* button = qobject_cast<QPushButton*>(sender());
        
        bool connected = connect(button, &QPushButton::clicked,
                                 this, &MyWidget::onButtonClick,
                                 Qt::UniqueConnection);
        
        if (connected) {
            qDebug() << "Connection established";
        } else {
            qDebug() << "Connection already exists";
        }
    }
    
    void onButtonClick() {
        clickCount++;
        label->setText(QString("Clicks: %1").arg(clickCount));
    }
};
```

**应用场景：**
- 避免重复连接
- 动态连接管理
- 插件系统

### 连接方式对比表

| 连接方式 | 执行方式 | 线程安全 | 适用场景 | 性能 |
|---------|---------|---------|---------|------|
| AutoConnection | 自动选择 | 是 | 通用场景 | 中等 |
| DirectConnection | 同步 | 否 | 同线程高性能 | 最高 |
| QueuedConnection | 异步 | 是 | 跨线程 | 中等 |
| BlockingQueuedConnection | 阻塞同步 | 是 | 等待结果 | 最低 |
| UniqueConnection | 确保唯一 | 取决于其他 | 避免重复 | 取决于其他 |

### 信号槽连接断开

    QPushButton* button;

**延迟信号发送：**

```cpp
class OptimizedSender : public QObject {
    Q_OBJECT
private:
    QTimer* timer;
    bool pendingUpdate;
    
public:
    OptimizedSender(QObject* parent = nullptr) : QObject(parent), pendingUpdate(false) {
        timer = new QTimer(this);
        timer->setSingleShot(true);
        timer->setInterval(16);  // ~60FPS
        
        connect(timer, &QTimer::timeout, this, [this]() {
            if (pendingUpdate) {
                emit updateNeeded();
                pendingUpdate = false;
            }
        });
    }
    
    void requestUpdate() {
        pendingUpdate = true;
        timer->start();
    }
    
signals:
    void updateNeeded();
};
```

### 信号槽与多线程

**线程安全的信号槽：**

```cpp
class Worker : public QObject {
    Q_OBJECT
public:
    Worker(QObject* parent = nullptr) : QObject(parent) {}
    
public slots:
    void processTask(const QString& task) {
        qDebug() << "Processing in thread:" << QThread::currentThreadId();
        
        QThread::msleep(100);
        
        emit taskCompleted(task + " done");
    }
    
signals:
    void taskCompleted(const QString& result);
};

class TaskManager : public QObject {
    Q_OBJECT
private:
    Worker* worker;
    QThread* workerThread;
    
public:
    TaskManager(QObject* parent = nullptr) : QObject(parent) {
        worker = new Worker();
        workerThread = new QThread();
        
        worker->moveToThread(workerThread);
        
        connect(workerThread, &QThread::started, [this]() {
            qDebug() << "Worker thread started";
        });
        
        connect(this, &TaskManager::newTask,
                 worker, &Worker::processTask,
                 Qt::QueuedConnection);
        
        connect(worker, &Worker::taskCompleted,
                 this, &TaskManager::onTaskCompleted,
                 Qt::QueuedConnection);
        
        workerThread->start();
    }
    
    ~TaskManager() {
        workerThread->quit();
        workerThread->wait();
        delete worker;
        delete workerThread;
    }
    
    void addTask(const QString& task) {
        emit newTask(task);
    }
    
signals:
    void newTask(const QString& task);
    
private slots:
    void onTaskCompleted(const QString& result) {
        qDebug() << "Task completed in main thread:" << QThread::currentThreadId();
        qDebug() << "Result:" << result;
    }
};
```

### 总结

**连接方式选择原则：**

1. **AutoConnection**：默认选择，自动适应场景
2. **DirectConnection**：同线程高性能调用
3. **QueuedConnection**：跨线程异步通信
4. **BlockingQueuedConnection**：需要等待结果的跨线程调用
5. **UniqueConnection**：确保不重复连接

**最佳实践：**
- 使用函数指针语法提高性能
- Lambda表达式简化代码
- 合理选择连接方式
- 注意线程安全
- 及时断开不再需要的连接
