# Qt事件处理与事件过滤器

## 题目

Qt中事件处理机制是什么？如何使用事件过滤器拦截和处理事件？

## 答案

### 事件处理基本概念

**Qt事件处理流程：**
1. 事件产生（键盘、鼠标、定时器等）
2. 事件发送（postEvent/sendEvent）
3. 事件过滤（installEventFilter）
4. 事件处理（event函数）
5. 事件分发（特定的事件处理函数）

### 基本事件重写

```cpp
class CustomWidget : public QWidget {
    Q_OBJECT

public:
    CustomWidget(QWidget *parent = nullptr) : QWidget(parent) {}

protected:
    //重写事件处理函数
    bool event(QEvent *e) override {
        switch (e->type()) {
            case QEvent::MouseButtonPress:
                qDebug() << "Mouse button pressed";
                break;
            case QEvent::MouseButtonRelease:
                qDebug() << "Mouse button released";
                break;
            case QEvent::KeyPress:
                qDebug() << "Key pressed";
                break;
            default:
                break;
        }
        return QWidget::event(e);
    }

    void mousePressEvent(QMouseEvent *event) override {
        qDebug() << "Mouse press at:" << event->pos();
        QWidget::mousePressEvent(event);
    }

    void mouseReleaseEvent(QMouseEvent *event) override {
        qDebug() << "Mouse release at:" << event->pos();
        QWidget::mouseReleaseEvent(event);
    }

    void mouseMoveEvent(QMouseEvent *event) override {
        if (event->buttons() & Qt::LeftButton) {
            qDebug() << "Mouse move with left button:" << event->pos();
        }
        QWidget::mouseMoveEvent(event);
    }

    void keyPressEvent(QKeyEvent *event) override {
        qDebug() << "Key pressed:" << event->key();
        QWidget::keyPressEvent(event);
    }
};
```

### 事件过滤器

**安装事件过滤器：**

```cpp
class EventFilter : public QObject {
    Q_OBJECT

public:
    EventFilter(QObject *parent = nullptr) : QObject(parent) {}

protected:
    bool eventFilter(QObject *watched, QEvent *event) override {
        if (event->type() == QEvent::MouseButtonPress) {
            QMouseEvent *mouseEvent = static_cast<QMouseEvent *>(event);
            
            if (mouseEvent->button() == Qt::RightButton) {
                qDebug() << "Right button clicked on:" << watched->objectName();
                
                //处理右键菜单
                QMenu *menu = new QMenu;
                menu->addAction("Action 1");
                menu->addAction("Action 2");
                
                menu->exec(QCursor::pos());
                return true;
            }
        }
        
        return QObject::eventFilter(watched, event);
    }
};

//使用事件过滤器
MainWindow::MainWindow(QWidget *parent) : QMainWindow(parent) {
    EventFilter *filter = new EventFilter(this);
    
    //为多个控件安装同一个过滤器
    ui->button1->installEventFilter(filter);
    ui->button2->installEventFilter(filter);
    ui->label->installEventFilter(filter);
}
```

### 全局事件过滤器

```cpp
class GlobalEventFilter : public QObject {
    Q_OBJECT

public:
    GlobalEventFilter(QObject *parent = nullptr) : QObject(parent) {}

protected:
    bool eventFilter(QObject *watched, QEvent *event) override {
        if (event->type() == QEvent::KeyPress) {
            QKeyEvent *keyEvent = static_cast<QKeyEvent *>(event);
            
            //拦截F1键
            if (keyEvent->key() == Qt::Key_F1) {
                qDebug() << "F1 key pressed globally";
                return true;
            }
        }
        
        return QObject::eventFilter(watched, event);
    }
};

//在main函数中安装全局过滤器
int main(int argc, char *argv[])
{
    QApplication a(argc, argv);
    
    GlobalEventFilter *filter = new GlobalEventFilter;
    a.installEventFilter(filter);
    
    MainWindow w;
    w.show();
    return a.exec();
}
```

### 拖拽事件

```cpp
class DragDropWidget : public QWidget {
    Q_OBJECT

public:
    DragDropWidget(QWidget *parent = nullptr) : QWidget(parent) {
        setAcceptDrops(true);
    }

protected:
    void dragEnterEvent(QDragEnterEvent *event) override {
        if (event->mimeData()->hasFormat("text/plain")) {
            event->acceptProposedAction();
        }
    }

    void dragMoveEvent(QDragMoveEvent *event) override {
        if (event->mimeData()->hasFormat("text/plain")) {
            event->acceptProposedAction();
        }
    }

    void dropEvent(QDropEvent *event) override {
        if (event->mimeData()->hasFormat("text/plain")) {
            QString text = event->mimeData()->text();
            qDebug() << "Dropped text:" << text;
            event->acceptProposedAction();
        }
    }

    void mousePressEvent(QMouseEvent *event) override {
        if (event->button() == Qt::LeftButton) {
            QDrag *drag = new QDrag(this);
            QMimeData *mimeData = new QMimeData;
            
            mimeData->setText("Dragged text");
            drag->setMimeData(mimeData);
            
            drag->exec(Qt::CopyAction | Qt::MoveAction);
        }
        QWidget::mousePressEvent(event);
    }
};
```

### 系统原生事件

```cpp
class NativeEventWidget : public QMainWindow {
    Q_OBJECT

public:
    NativeEventWidget(QWidget *parent = nullptr) : QMainWindow(parent) {}

protected:
#ifdef Q_OS_WIN
    bool nativeEvent(const QByteArray &eventType, 
                     void *message, 
                     long *result) override {
        if (eventType == "windows_generic_MSG") {
            MSG *msg = static_cast<MSG *>(message);
            
            if (msg->message == WM_POWERBROADCAST) {
                if (msg->wParam == PBT_APMSUSPEND) {
                    qDebug() << "System suspending";
                    this->showMinimized();
                } else if (msg->wParam == PBT_APMRESUMEAUTOMATIC) {
                    qDebug() << "System resuming";
                    this->showNormal();
                }
            }
        }
        return QMainWindow::nativeEvent(eventType, message, result);
    }

    bool winEvent(MSG *message, long *result) override {
        return nativeEvent("windows_generic_MSG", message, result);
    }
#endif
};
```

### 定时器事件

class TimerWidget : public QWidget {
    Q_OBJECT

public:
    TimerWidget(QWidget *parent = nullptr) : QWidget(parent) {
        //启动定时器
        timerId1 = startTimer(1000, Qt::PreciseTimer);
        timerId2 = startTimer(5000, Qt::CoarseTimer);

protected:
    void timerEvent(QTimerEvent *event) override {
        if (event->timerId() == timerId1) {
            qDebug() << "Timer 1 triggered (1 second)";
        } else if (event->timerId() == timerId2) {
            qDebug() << "Timer 2 triggered (5 seconds)";
        }
    }

private:
    int timerId1;
    int timerId2;
};

class PaintWidget : public QWidget {
    Q_OBJECT

public:
    PaintWidget(QWidget *parent = nullptr) : QWidget(parent) {
        //设置背景自动绘制属性
        setAttribute(Qt::WA_StyledBackground, true);
    }

protected:
    void paintEvent(QPaintEvent *event) override {
        QPainter painter(this);
        
        //设置抗锯齿
        painter.setRenderHint(QPainter::Antialiasing);
        
        //绘制圆形
        painter.setBrush(QBrush(QColor(255, 100, 100)));
        painter.drawEllipse(50, 50, 100, 100);
        
        //绘制矩形
        painter.setBrush(QBrush(QColor(100, 100, 255)));
        painter.drawRect(200, 50, 100, 100);
        
        //绘制文本
        painter.setPen(QColor(0, 0, 0));
        painter.drawText(50, 200, "Hello Qt!");
    }
};

### 总结

**事件处理要点：**
- event函数是事件的总入口
- 事件过滤器可以拦截多个控件的事件
- 鼠标跟踪需要setMouseTracking(true)
- 系统原生事件使用nativeEvent

**事件过滤器使用：**
- installEventFilter安装过滤器
- eventFilter中实现过滤逻辑
- 返回true表示事件已处理
- 返回false继续传递事件

**最佳实践：**
- 使用事件过滤器统一处理相似事件
- nativeEvent处理系统级消息
- 合理使用鼠标跟踪和Hover事件
- 注意事件的传递顺序
