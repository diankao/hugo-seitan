# Qt性能优化技巧

## 题目

在Qt开发中有哪些常见的性能优化技巧？如何避免不必要的深拷贝？如何优化循环中的操作？

## 答案

### 字符串转换优化

```cpp
class StringConversionOptimization : public QObject {
    Q_OBJECT

public:
    StringConversionOptimization(QObject *parent = nullptr) : QObject(parent) {}

    //推荐做法：使用constData避免深拷贝
    const char *toConstCharOptimized(const QString &text) {
        QByteArray data = text.toUtf8();
        return data.constData();
    }

    //不推荐：使用data()可能深拷贝
    const char *toConstCharNotOptimized(const QString &text) {
        QByteArray data = text.toUtf8();
        return data.data();
    }

    //直接作为参数传入，避免中间变量
    void processString(const QString &text) {
        processText(text.toUtf8().constData());
    }

    //作为const引用传递避免拷贝
    void processStringRef(const QString &text) {
        processText(text.toUtf8().constData());
    }
};
```

### 循环变量优化

```cpp
class LoopOptimization : public QObject {
    Q_OBJECT

public:
    LoopOptimization(QObject *parent = nullptr) : QObject(parent) {}

    void optimizedLoop() {
        QElapsedTimer timer;
        timer.start();

        QString s;
        QString text = "abc";
        
        for (int i = 0; i < 10000; ++i) {
            s = text.at(0);
        }

        qDebug() << "优化后时间:" << timer.nsecsElapsed() << "ns";
    }

    void notOptimizedLoop() {
        QElapsedTimer timer;
        timer.start();

        QString text = "abc";
        
        for (int i = 0; i < 10000; ++i) {
            QString s = text.at(0);
        }

        qDebug() << "未优化时间:" << timer.nsecsElapsed() << "ns";
    }
};
```

### 容器循环优化

```cpp
class ContainerLoopOptimization : public QObject {
    Q_OBJECT

public:
    ContainerLoopOptimization(QObject *parent = nullptr) : QObject(parent) {}

    void optimizedContainerLoop() {
        QElapsedTimer timer;
        timer.start();

        QVector<QString> list;
        for (int i = 0; i < 10000; i++) {
            list.append(QString("Item%1").arg(i));
        }

        for (const QString &item : qAsConst(list)) {
            Q_UNUSED(item);
        }

        qDebug() << "容器循环优化后:" << timer.nsecsElapsed() << "ns";
    }

    void notOptimizedContainerLoop() {
        QElapsedTimer timer;
        timer.start();

        QVector<QString> list;
        for (int i = 0; i < 10000; i++) {
            list.append(QString("Item%1").arg(i));
        }

        for (const QString &item : list) {
            Q_UNUSED(item);
        }

        qDebug() << "容器循环未优化:" << timer.nsecsElapsed() << "ns";
    }
};
```

### 数据库查询优化

```cpp
class DatabaseQueryOptimization : public QObject {
    Q_OBJECT

public:
    DatabaseQueryOptimization(QObject *parent = nullptr) : QObject(parent) {}

    QSqlQuery optimizedQuery(const QString &sql) {
        QSqlQuery query;
        
        query.setForwardOnly(true);
        
        if (query.exec(sql)) {
            return query;
        }
        
        return QSqlQuery();
    }

    QSqlQuery batchInsertOptimized(const QString &table,
                                    const QList<QStringList> &data) {
        QSqlDatabase db = QSqlDatabase::database();
        db.transaction();

        QSqlQuery query;
        QString sql = QString("INSERT INTO %1 VALUES (?, ?, ?)").arg(table);
        query.prepare(sql);

        foreach (const QStringList &row, data) {
            query.addBindValue(row[0]);
            query.addBindValue(row[1]);
            query.addBindValue(row[2]);
            
            if (!query.exec()) {
                db.rollback();
                return QSqlQuery();
            }
        }

        db.commit();
        return query;
    }
};
```

### 界面绘制优化

```cpp
class PaintOptimization : public QObject {
    Q_OBJECT

public:
    PaintOptimization(QObject *parent = nullptr) : QObject(parent) {}

    void optimizedPaint(QWidget *widget) {
        QPainter painter(widget);
        
        painter.setRenderHint(QPainter::Antialiasing, false);
        painter.setRenderHint(QPainter::SmoothPixmapTransform, false);
        
        painter.fillRect(widget->rect(), Qt::white);
    }

    void batchDraw() {
        QImage image(1000, 1000, QImage::Format_ARGB32);
        QPainter painter(&image);
        
        painter.setRenderHint(QPainter::Antialiasing);
        
        painter.setBrush(Qt::NoBrush);
        painter.setPen(QPen(Qt::black, 2));
        
        for (int i = 0; i < 100; i++) {
            painter.drawEllipse(i * 10, i * 10, 50, 50);
        }
    }
};
```

### QSettings优化

```cpp
class SettingsOptimization : public QObject {
    Q_OBJECT

public:
    SettingsOptimization(QObject *parent = nullptr) : QObject(parent) {}

    void saveFloatOptimized(QSettings &settings, const QString &key, float value) {
        settings.setValue(key, QString::number(value));
    }

    float loadFloatOptimized(QSettings &settings, const QString &key, float defaultValue) {
        QString str = settings.value(key, QString::number(defaultValue)).toString();
        return str.toFloat();
    }

    void saveDoubleOptimized(QSettings &settings, const QString &key, double value) {
        settings.setValue(key, value);
    }

    double loadDoubleOptimized(QSettings &settings, const QString &key, double defaultValue) {
        return settings.value(key, defaultValue).toDouble();
    }
};
```

### 信号槽优化

```cpp
class SignalSlotOptimization : public QObject {
    Q_OBJECT

public:
    SignalSlotOptimization(QObject *parent = nullptr) : QObject(parent) {}

    void setupOptimizedConnections(QObject *sender, const char *signal,
                                   QObject *receiver, const char *slot) {
        disconnect(sender, signal, receiver, slot);
        connect(sender, signal, receiver, slot, Qt::UniqueConnection);
    }

    void batchUpdateOptimized() {
        blockSignals(true);
        
        for (int i = 0; i < 100; i++) {
            updateItem(i);
        }
        
        blockSignals(false);
        emit updateCompleted();
    }

signals:
    void updateCompleted();

private:
    void updateItem(int index) {
        Q_UNUSED(index);
    }
};
```

### 总结

**性能优化要点：**
- 使用constData避免深拷贝
- 循环变量定义在循环外
- 使用qAsConst避免容器detach
- 数据库查询使用setForwardOnly
- 批量操作使用事务
- 界面绘制关闭不必要的渲染提示

**QString转换优化：**
- 优先使用constData()
- 避免中间临时变量
- 直接作为参数传递

**循环优化：**
- 复杂类型定义在循环外
- 使用引用避免拷贝
- 使用qAsConst避免detach

**数据库优化：**
- 使用只前进查询
- 批量操作使用事务
- 合理使用索引

**最佳实践：**
- 避免不必要的深拷贝
- 合理使用const引用
- 批量操作减少重复调用
- 选择合适的数据结构
