# Qt安卓开发与JNI调用

## 题目

Qt中如何进行安卓开发？如何使用QAndroidJniObject调用Java方法？JNI调用有哪些注意事项？

## 答案

### pro文件配置

```qmake
# 引入安卓拓展模块
QT += androidextras

# 指定安卓打包目录
ANDROID_PACKAGE_SOURCE_DIR = $$PWD/android

# 引入安卓相关文件
OTHER_FILES += android/AndroidManifest.xml
OTHER_FILES += android/src/QtAndroidTest.java
```

### 安卓目录结构

```
android/
├── AndroidManifest.xml    # 全局配置文件
├── res/
│   ├── drawable-hdpi/     # 应用图标
│   ├── layout/           # 布局文件
│   └── values/           # 变量值
├── libs/                 # jar库文件
└── src/                  # java代码文件
```

### Java类定义

```java
package org.qt;

public class QtAndroidTest {
    // 无返回值方法
    public static void printMsg() {
        System.out.println("printMsg");
    }

    // 带参数方法
    public static void printValue(int value) {
        System.out.println("printValue:" + value);
    }

    // 多参数方法
    public static void setValue(float value1, double value2, char value3) {
        System.out.println("value1:" + value1 + " value2:" + value2 + " value3:" + value3);
    }

    // 返回int
    public static int getValue() {
        return 65536;
    }

    // 带参数返回int
    public static int getValue(int value) {
        return value + 1;
    }

    // 字符串参数
    public static void setMsg(String message) {
        System.out.println("setMsg:" + message);
    }

    // 返回字符串
    public static String getMsg() {
        return "hello from java";
    }

    // 混合参数
    public static String getText(int value1, float value2, boolean value3, String message) {
        return "value1:" + value1 + " value2:" + value2 + " value3:" + value3 + " message:" + message;
    }
}
```

### JNI调用示例

```cpp
#include <QAndroidJniObject>
#include <QtAndroid>

class AndroidJniCaller : public QObject {
    Q_OBJECT

public:
    AndroidJniCaller(QObject *parent = nullptr) : QObject(parent) {}

    void testJavaBuiltIn() {
        jint a = 12;
        jint b = 4;
        
        jint max = QAndroidJniObject::callStaticMethod<jint>(
            "java/lang/Math", "max", "(II)I", a, b
        );
        
        jdouble value = QAndroidJniObject::callStaticMethod<jdouble>(
            "java/lang/Math", "random"
        );
        
        qDebug() << "Max:" << max << "Random:" << value;
    }

    void callVoidMethod() {
        QAndroidJniObject::callStaticMethod<void>(
            "org/qt/QtAndroidTest", 
            "printMsg", 
            "()V"
        );
    }

    void callIntMethod() {
        jint result = QAndroidJniObject::callStaticMethod<jint>(
            "org/qt/QtAndroidTest", 
            "getValue", 
            "()I"
        );
        
        qDebug() << "Result:" << result;
    }

    void callIntMethodWithParam() {
        jint result = QAndroidJniObject::callStaticMethod<jint>(
            "org/qt/QtAndroidTest", 
            "getValue", 
            "(I)I", 
            100
        );
        
        qDebug() << "Result:" << result;
    }

    void callStringMethod() {
        QAndroidJniObject result = QAndroidJniObject::callStaticObjectMethod(
            "org/qt/QtAndroidTest", 
            "getMsg", 
            "()Ljava/lang/String;"
        );
        
        QString msg = result.toString();
        qDebug() << "Message:" << msg;
    }

    void callStringMethodWithParam() {
        QAndroidJniObject result = QAndroidJniObject::callStaticObjectMethod(
            "org/qt/QtAndroidTest", 
            "getText", 
            "(IFZLjava/lang/String;)Ljava/lang/String;",
            100, 
            3.14f, 
            true, 
            QAndroidJniObject::fromString("hello")
        );
        
        QString text = result.toString();
        qDebug() << "Text:" << text;
    }
};
```

### Java回调Qt

```java
package org.qt;

public class QtAndroidCallback {
    public static void sendToQt(String message, int value) {
        QtAndroidData.receiveData(message, value);
    }
}
```

```cpp
class QtAndroidData : public QObject {
    Q_OBJECT

public:
    QtAndroidData(QObject *parent = nullptr) : QObject(parent) {}

    static void receiveData(const QString &message, int value) {
        qDebug() << "Received from Java:" << message << value;
        
        emit dataReceived(message, value);
    }

signals:
    void dataReceived(const QString &message, int value);
};
```

### 安卓权限请求

```cpp
#include <QtAndroid>
#include <QAndroidJniObject>

class PermissionManager : public QObject {
    Q_OBJECT

public:
    PermissionManager(QObject *parent = nullptr) : QObject(parent) {}

    void requestPermission(const QString &permission) {
        auto result = QtAndroid::checkPermission(permission);
        
        if (result == QtAndroid::PermissionResult::Denied) {
            QtAndroid::requestPermissionsSync(QStringList() << permission);
        }
    }

    bool hasPermission(const QString &permission) {
        auto result = QtAndroid::checkPermission(permission);
        return result == QtAndroid::PermissionResult::Granted;
    }
};
```

### 安卓原生通知

```cpp
#include <QAndroidJniObject>
#include <QCoreApplication>

class AndroidNotification : public QObject {
    Q_OBJECT

public:
    AndroidNotification(QObject *parent = nullptr) : QObject(parent) {}

    void showNotification(const QString &title, const QString &message) {
        QAndroidJniObject activity = QAndroidJniObject::callStaticObjectMethod(
            "org/qtproject/qt5/android/QtNative", 
            "activity", 
            "()Landroid/app/Activity;"
        );
        
        if (!activity.isValid()) {
            return;
        }

        QAndroidJniObject notification = QAndroidJniObject::callStaticObjectMethod(
            "org/qtproject/qt5/android/QtNotificationManager",
            "showNotification",
            "(Landroid/content/Context;Ljava/lang/String;Ljava/lang/String;)I",
            activity.object(),
            QAndroidJniObject::fromString(title).object(),
            QAndroidJniObject::fromString(message).object()
        );
    }
};
```

### 总结

**pro配置要点：**
- 引入androidextras模块
- 指定安卓打包目录
- 引入AndroidManifest.xml和Java文件

**JNI调用要点：**
- 使用QAndroidJniObject进行调用
- 包名必须与Java文件package一致
- callStaticMethod调用静态方法
- callStaticObjectMethod调用返回对象的方法

**方法签名规则：**
- ()V：无参数无返回值
- (I)I：int参数，int返回值
- (Ljava/lang/String;)V：String参数，void返回值

**类型映射：**
- void → V
- int → I
- float → F
- double → D
- boolean → Z
- String → Ljava/lang/String;

**注意事项：**
- Java类名必须与文件名一致
- Java文件必须在android/src目录
- 包名必须严格匹配
- 注意大小写敏感

**最佳实践：**
- 在Java中提供静态方法供Qt调用
- 使用QtAndroidData接收Java回调
- 合理处理权限请求
- 注意线程安全
