# Qt信号槽连接与线程管理

## 题目

Qt中信号槽的连接方式有哪些？如何避免重复连接？如何正确地停止线程？

## 答案

### 信号槽连接类型

Qt5提供了五种连接方式：

**Qt::AutoConnection（自动连接）：**
- 默认连接方式
- 发送者和接收者在同一线程：使用Qt::DirectConnection
- 发送者和接收者在不同线程：使用Qt::QueuedConnection

**Qt::DirectConnection（直接连接）：**
- 信号发出时，槽函数立即执行
- 在发送者的线程中执行槽函数
- 相当于直接函数调用

**Qt::QueuedConnection（队列连接）：**
- 信号发出后，槽函数放入事件队列
- 在接收者的线程中执行槽函数
- 线程安全

**Qt::BlockingQueuedConnection（阻塞队列连接）：**
- 类似Qt::QueuedConnection
- 发送者线程会阻塞，直到槽函数执行完毕
- 只能用于跨线程连接

**Qt::UniqueConnection（唯一连接）：**
- 避免重复连接同一个信号和槽
- 可以和其他连接类型组合使用

### 避免重复连接

```cpp
class SignalSlotManager : public QObject {
    Q_OBJECT

public:
    SignalSlotManager(QObject *parent = nullptr) : QObject(parent) {
        button = new QPushButton("Click Me", this);
        timer = new QTimer(this);
    }

    void setupConnections() {
        //方法1：先断开再连接
        disconnect(button, &QPushButton::clicked, this, &SignalSlotManager::onClicked);
        connect(button, &QPushButton::clicked, this, &SignalSlotManager::onClicked);

        //方法2：使用Qt::UniqueConnection
        connect(button, &QPushButton::clicked, this, &SignalSlotManager::onClicked, 
                Qt::UniqueConnection);

        //避免Qt4的SIGNAL和SLOT宏的重复连接问题
        disconnect(button, SIGNAL(clicked()), this, SLOT(onClicked()));
        connect(button, SIGNAL(clicked()), this, SLOT(onClicked()));

        //方法3：检查是否已连接
        if (!isConnected(button, &QPushButton::clicked, this, &SignalSlotManager::onClicked)) {
            connect(button, &QPushButton::clicked, this, &SignalSlotManager::onClicked);
        }
    }

    void setupTimerConnection() {
        //定时器连接
        connect(timer, &QTimer::timeout, this, &SignalSlotManager::onTimeout, 
                Qt::UniqueConnection);
    }

private slots:
    void onClicked() {
        qDebug() << "Button clicked";
    }

    void onTimeout() {
        qDebug() << "Timer timeout";
    }

private:
    QPushButton *button;
    QTimer *timer;
};
```

### 线程停止策略

```cpp
class WorkerThread : public QThread {
    Q_OBJECT

public:
    explicit WorkerThread(QObject *parent = nullptr) : QThread(parent) {
        stopped = false;
    }

    void stop() {
        stopped = true;
    }

protected:
    void run() override {
        while (!stopped) {
            doTask();
            
            //方法1：分割长延时
            int count = 0;
            while (!stopped && count < 30) {
                msleep(100);
                count++;
            }
        }
        
        stopped = false;
    }

private:
    void doTask() {
        while (!stopped) {
            processItem();
        }
    }

    void processItem() {
        //处理单个项目
        if (stopped) {
            return;
        }
        
        msleep(10);
    }

    volatile bool stopped;
};
```

### 线程使用示例

```cpp
class ThreadManager : public QObject {
    Q_OBJECT

public:
    ThreadManager(QObject *parent = nullptr) : QObject(parent) {
        workerThread = new WorkerThread(this);
        
        connect(workerThread, &QThread::finished, workerThread, &QThread::deleteLater);
        connect(workerThread, &WorkerThread::finished, this, &ThreadManager::onThreadFinished);
    }

    void startWork() {
        if (workerThread->isRunning()) {
            qWarning() << "Thread is already running";
            return;
        }
        
        workerThread->start();
    }

    void stopWork() {
        if (!workerThread->isRunning()) {
            return;
        }
        
        workerThread->stop();
        workerThread->quit();
        workerThread->wait();
    }

    bool isWorking() {
        return workerThread->isRunning();
    }

private slots:
    void onThreadFinished() {
        qDebug() << "Thread finished";
    }

private:
    WorkerThread *workerThread;
};
```

### 线程间通信

```cpp
class ProducerThread : public QThread {
    Q_OBJECT

public:
    explicit ProducerThread(QObject *parent = nullptr) : QThread(parent) {}

signals:
    void dataProduced(const QString &data);

protected:
    void run() override {
        int count = 0;
        
        while (!isInterruptionRequested()) {
            QString data = QString("Data %1").arg(count++);
            emit dataProduced(data);
            
            msleep(100);
        }
    }
};

class ConsumerThread : public QObject {
    Q_OBJECT

public:
    ConsumerThread(QObject *parent = nullptr) : QObject(parent) {}

public slots:
    void consumeData(const QString &data) {
        qDebug() << "Consuming:" << data << "in thread:" << QThread::currentThreadId();
        process(data);
    }

private:
    void process(const QString &data) {
        msleep(50);
    }
};

class ThreadCommunication : public QObject {
    Q_OBJECT

public:
    ThreadCommunication(QObject *parent = nullptr) : QObject(parent) {
        producer = new ProducerThread;
        consumer = new ConsumerThread;
        consumerThread = new QThread;
        
        consumer->moveToThread(consumerThread);
        
        connect(producer, &ProducerThread::dataProduced, 
                consumer, &ConsumerThread::consumeData,
                Qt::QueuedConnection);
        
        connect(producer, &QThread::finished, consumerThread, &QThread::quit);
        connect(consumerThread, &QThread::finished, consumer, &QObject::deleteLater);
    }

    void start() {
        consumerThread->start();
        producer->start();
    }

    void stop() {
        producer->requestInterruption();
        producer->quit();
        producer->wait();
        
        consumerThread->quit();
        consumerThread->wait();
    }

private:
    ProducerThread *producer;
    ConsumerThread *consumer;
    QThread *consumerThread;
};
```

### 智能指针使用

```cpp
class SmartPointerExample : public QObject {
    Q_OBJECT

public:
    SmartPointerExample(QObject *parent = nullptr) : QObject(parent) {}

    void traditionalWay() {
        QWidget *widget = new QWidget;
        
        widget->show();
        
        //需要手动释放
        widget->deleteLater();
    }

    void smartPointerWay() {
        QScopedPointer<QWidget> widget;
        widget.reset(new QWidget);
        
        widget->show();
        
        //自动释放，无需手动deleteLater
    }

    void multipleWidgets() {
        QList<QWidget *> widgets;
        
        for (int i = 0; i < 10; i++) {
            QWidget *w = new QWidget;
            widgets.append(w);
            w->show();
        }
        
        //批量释放
        foreach (QWidget *w, widgets) {
            w->deleteLater();
        }
    }
};
```

### 总结

**信号槽连接类型：**
- AutoConnection：自动选择连接方式
- DirectConnection：直接调用
- QueuedConnection：队列调用
- BlockingQueuedConnection：阻塞队列调用
- UniqueConnection：避免重复连接

**避免重复连接：**
- 使用Qt::UniqueConnection
- 先disconnect再connect
- 检查是否已连接

**线程停止策略：**
- 使用volatile标志位
- 分割长延时为短延时
- 在耗时函数中检查标志位
- 避免使用terminate

**线程间通信：**
- 使用Qt::QueuedConnection
- 信号槽自动处理线程安全
- 注意对象所属线程

**智能指针：**
- QScopedPointer自动释放
- 避免内存泄漏
- 简化代码
