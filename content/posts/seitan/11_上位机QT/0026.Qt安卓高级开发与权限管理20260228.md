### Qt5中调用

```cpp
#include <QAndroidJniObject>
#include <QtAndroid>

class AndroidActivityManager : public QObject {
    Q_OBJECT

public:
    AndroidActivityManager(QObject *parent = nullptr) : QObject(parent) {}

    void showNotification(const QString &title, const QString &message) {
        QAndroidJniObject javaTitle = QAndroidJniObject::fromString(title);
        QAndroidJniObject javaMessage = QAndroidJniObject::fromString(message);

        QAndroidJniObject activity = QtAndroid::androidActivity();
        if (!activity.isValid()) {
            return;
        }

        QAndroidJniObject::callStaticMethod<void>(
            "org/qt/QtAndroidActivity",
            "showNotification",
            "(Landroid/content/Context;Ljava/lang/String;Ljava/lang/String;)V",
            activity.object(),
            javaTitle.object<jstring>(),
            javaMessage.object<jstring>()
        );
    }
};
```

### 权限请求

```cpp
#include <QtAndroid>
#include <QAndroidJniObject>
#include <QAndroidPermissionResult>

class PermissionHandler : public QObject {
    Q_OBJECT

public:
    PermissionHandler(QObject *parent = nullptr) : QObject(parent) {}

    void requestCameraPermission() {
#ifdef Q_OS_ANDROID
        QString cameraPermission = "android.permission.CAMERA";
        auto result = QtAndroid::checkPermission(cameraPermission);

        if (result == QtAndroid::PermissionResult::Denied) {
            QtAndroid::requestPermissionsSync(QStringList() << cameraPermission);
        }

        if (QtAndroid::checkPermission(cameraPermission) == QtAndroid::PermissionResult::Granted) {
            emit permissionGranted();
        }
#endif
    }

    void requestStoragePermission() {
#ifdef Q_OS_ANDROID
        QString storagePermission = "android.permission.WRITE_EXTERNAL_STORAGE";
        auto result = QtAndroid::checkPermission(storagePermission);

        if (result == QtAndroid::PermissionResult::Denied) {
            QtAndroid::requestPermissionsSync(QStringList() << storagePermission);
        }

        if (QtAndroid::checkPermission(storagePermission) == QtAndroid::PermissionResult::Granted) {
            emit permissionGranted();
        }
#endif
    }

    bool hasPermission(const QString &permission) {
#ifdef Q_OS_ANDROID
        auto result = QtAndroid::checkPermission(permission);
        return result == QtAndroid::PermissionResult::Granted;
#else
        Q_UNUSED(permission);
        return true;
#endif
    }

signals:
    void permissionGranted();
    void permissionDenied();
};
```

### 横竖屏识别

```cpp
class ScreenOrientationDetector : public QObject {
    Q_OBJECT

public:
    ScreenOrientationDetector(QObject *parent = nullptr) : QObject(parent) {}

    bool isLandscape() {
#ifdef Q_OS_ANDROID
        QAndroidJniObject activity = QtAndroid::androidActivity();
        if (!activity.isValid()) {
            return false;
        }

        QAndroidJniObject windowManager = activity.callObjectMethod(
            "getWindowManager", "()Landroid/view/WindowManager;"
        );

        QAndroidJniObject display = windowManager.callObjectMethod(
            "getDefaultDisplay", "()Landroid/view/Display;"
        );

        jint rotation = display.callMethod<jint>("getRotation");
        
        return rotation == 0 || rotation == 2;
#else
        return false;
#endif
    }

    bool isPortrait() {
        return !isLandscape();
    }
};
```

### 震动控制

```java
package org.qt;

import android.os.Vibrator;
import android.content.Context;

public class VibratorHelper {
    private static Vibrator vibrator;

    public static void init(Context context) {
        vibrator = (Vibrator)context.getSystemService(Context.VIBRATOR_SERVICE);
    }

    public static void vibrate(long milliseconds) {
        if (vibrator != null) {
            vibrator.vibrate(milliseconds);
        }
    }

    public static void vibrate(long[] pattern, int repeat) {
        if (vibrator != null) {
            vibrator.vibrate(pattern, repeat);
        }
    }
}
```

```cpp
class VibratorManager : public QObject {
    Q_OBJECT

public:
    VibratorManager(QObject *parent = nullptr) : QObject(parent) {}

    void vibrate(int milliseconds) {
#ifdef Q_OS_ANDROID
        QAndroidJniObject::callStaticMethod<void>(
            "org/qt/VibratorHelper",
            "vibrate",
            "(J)V",
            (jlong)milliseconds
        );
#endif
    }

    void vibratePattern() {
#ifdef Q_OS_ANDROID
        jlongArray pattern = QAndroidJniObject::callStaticMethod<jlongArray>(
            "org/qt/VibratorHelper",
            "getDefaultPattern"
        );

        QAndroidJniObject::callStaticMethod<void>(
            "org/qt/VibratorHelper",
            "vibrate",
            "([JI)V",
            pattern,
            -1
        );
#endif
    }
};
```

### 总结

**自定义Activity要点：**
- 继承QtActivity
- 提供静态方法获取Context
- 实现onCreate初始化
- 在AndroidManifest.xml中配置

**权限管理：**
- AndroidManifest.xml中声明权限
- 使用checkPermission检查权限
- 使用requestPermissionsSync请求权限
- 注意运行时权限和安装时权限

**横竖屏识别：**
- 通过getRotation获取屏幕方向
- 0和2为横屏，1和3为竖屏
- 在resizeEvent中处理布局变化

**JNI调用要点：**
- Qt5使用QAndroidJniObject
- Qt6使用QJniObject
- 包名必须严格匹配
- 注意方法签名规则

**Qt5与Qt6差异：**
- 类名变化：QAndroidJniObject → QJniObject
- Activity获取方式变化
- 包名去掉qt5标识
- 使用QNativeInterface

**最佳实践：**
- 使用自定义Activity简化交互
- 提供静态方法避免重复传递Context
- 合理处理权限请求
- 注意横竖屏切换的布局适配
