# QT多线程技术

## 题目

QT的多线程，你用了哪些技术？哪些是只有QThread能做的，QtConcurrent办不到的？

## 答案

### QT多线程技术概览

QT提供了多种多线程技术，每种都有其适用场景：

### 1. QThread（线程类）

**基本使用：**

```cpp
#include <QCoreApplication>
#include <QThread>
#include <QDebug>

class MyThread : public QThread {
    Q_OBJECT
protected:
    void run() override {
        qDebug() << "Thread started:" << QThread::currentThreadId();
        
        for (int i = 0; i < 5; i++) {
            qDebug() << "Working:" << i;
            QThread::msleep(1000);
        }
        
        qDebug() << "Thread finished";
    }
};

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    qDebug() << "Main thread:" << QThread::currentThreadId();
    
    MyThread thread;
    thread.start();
    thread.wait();
    
    return 0;
}
```

**信号槽与线程：**

```cpp
#include <QCoreApplication>
#include <QThread>
#include <QDebug>

class Worker : public QObject {
    Q_OBJECT
public:
    Worker(QObject* parent = nullptr) : QObject(parent) {}
    
public slots:
    void doWork(int count) {
        qDebug() << "Worker thread:" << QThread::currentThreadId();
        
        for (int i = 0; i < count; i++) {
            qDebug() << "Processing:" << i;
            QThread::msleep(500);
            emit progress(i, count);
        }
        
        emit finished();
    }
    
signals:
    void progress(int current, int total);
    void finished();
};

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    qDebug() << "Main thread:" << QThread::currentThreadId();
    
    QThread workerThread;
    Worker worker;
    
    worker.moveToThread(&workerThread);
    
    QObject::connect(&workerThread, &QThread::started, [&]() {
        worker.doWork(10);
    });
    
    QObject::connect(&worker, &Worker::progress, [](int current, int total) {
        qDebug() << "Progress:" << current << "/" << total;
    });
    
    QObject::connect(&worker, &Worker::finished, &app, &QCoreApplication::quit);
    
    QObject::connect(&worker, &Worker::finished, &workerThread, &QThread::quit);
    
    workerThread.start();
    
    return app.exec();
}
```

### 2. QtConcurrent（并发框架）

**基本使用：**

```cpp
#include <QCoreApplication>
#include <QtConcurrent>
#include <QVector>
#include <QDebug>

int processValue(int value) {
    qDebug() << "Processing:" << value;
    QThread::msleep(100);
    return value * 2;
}

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    QVector<int> numbers = {1, 2, 3, 4, 5};
    
    QFuture<int> future = QtConcurrent::mapped(numbers, processValue);
    
    future.waitForFinished();
    
    QVector<int> results = future.results();
    qDebug() << "Results:" << results;
    
    return 0;
}
```

**map-reduce操作：**

```cpp
#include <QCoreApplication>
#include <QtConcurrent>
#include <QVector>
#include <QDebug>

void square(int& value) {
    value = value * value;
}

int sum(int start, const QVector<int>& values) {
    int total = start;
    for (int value : values) {
        total += value;
    }
    return total;
}

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    QVector<int> numbers = {1, 2, 3, 4, 5};
    
    QtConcurrent::map(numbers, square);
    qDebug() << "Squared:" << numbers;
    
    QFuture<int> result = QtConcurrent::filteredReduced(
        numbers,
        [](int value) { return value > 10; },
        sum
    );
    
    result.waitForFinished();
    qDebug() << "Filtered sum:" << result.result();
    
    return 0;
}
```

### 3. QThreadPool（线程池）

**使用线程池：**

```cpp
#include <QCoreApplication>
#include <QThreadPool>
#include <QRunnable>
#include <QDebug>

class MyTask : public QRunnable {
public:
    MyTask(int id) : m_id(id) {}
    
    void run() override {
        qDebug() << "Task" << m_id << "running in thread:" 
                 << QThread::currentThreadId();
        QThread::msleep(1000);
        qDebug() << "Task" << m_id << "completed";
    }
    
private:
    int m_id;
};

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    QThreadPool::globalInstance()->setMaxThreadCount(4);
    
    for (int i = 0; i < 10; i++) {
        MyTask* task = new MyTask(i);
        QThreadPool::globalInstance()->start(task);
    }
    
    QThreadPool::globalInstance()->waitForDone();
    
    return 0;
}
```

### QThread vs QtConcurrent 对比

| 特性 | QThread | QtConcurrent |
|------|---------|--------------|
| 复杂度 | 较高 | 简单 |
| 控制 | 完全控制 | 自动管理 |
| 适用场景 | 长时间运行任务 | 短时间并发任务 |
| 线程池 | 需手动管理 | 自动使用 |
| 信号槽 | 支持 | 支持 |
| 细粒度控制 | 支持 | 有限 |

### 只有QThread能做的事情

**1. 长时间运行的独立线程**

```cpp
class NetworkThread : public QThread {
    Q_OBJECT
public:
    NetworkThread(QObject* parent = nullptr) : QThread(parent), m_running(false) {}
    
    void stop() {
        m_running = false;
        wait();
    }
    
protected:
    void run() override {
        m_running = true;
        
        while (m_running) {
            QTcpSocket socket;
            socket.connectToHost("localhost", 8080);
            
            if (socket.waitForConnected(5000)) {
                qDebug() << "Connected";
                
                while (m_running && socket.state() == QAbstractSocket::ConnectedState) {
                    QByteArray data = socket.readAll();
                    if (!data.isEmpty()) {
                        emit dataReceived(data);
                    }
                    socket.waitForReadyRead(100);
                }
            }
            
            QThread::sleep(1);
        }
    }
    
signals:
    void dataReceived(const QByteArray& data);
    
private:
    bool m_running;
};
```

**2. 需要精确控制线程生命周期**

```cpp
class ControlledThread : public QThread {
    Q_OBJECT
public:
    ControlledThread(QObject* parent = nullptr) : QThread(parent) {}
    
    void pause() {
        m_mutex.lock();
        m_paused = true;
        m_mutex.unlock();
    }
    
    void resume() {
        m_mutex.lock();
        m_paused = false;
        m_condition.wakeAll();
        m_mutex.unlock();
    }
    
    void stop() {
        m_mutex.lock();
        m_running = false;
        m_condition.wakeAll();
        m_mutex.unlock();
        wait();
    }
    
protected:
    void run() override {
        m_running = true;
        m_paused = false;
        
        while (m_running) {
            m_mutex.lock();
            if (m_paused) {
                m_condition.wait(&m_mutex);
            }
            m_mutex.unlock();
            
            if (!m_running) break;
            
            qDebug() << "Working...";
            QThread::msleep(100);
        }
    }
    
private:
    QMutex m_mutex;
    QWaitCondition m_condition;
    bool m_running;
    bool m_paused;
};
```

**3. 需要频繁线程间通信**

```cpp
class ProducerThread : public QThread {
    Q_OBJECT
public:
    ProducerThread(QObject* parent = nullptr) : QThread(parent) {}
    
signals:
    void dataProduced(const QString& data);
    
protected:
    void run() override {
        for (int i = 0; i < 100; i++) {
            QString data = QString("Data %1").arg(i);
            emit dataProduced(data);
            QThread::msleep(50);
        }
    }
};

class ConsumerThread : public QThread {
    Q_OBJECT
public:
    ConsumerThread(QObject* parent = nullptr) : QThread(parent) {}
    
public slots:
    void consumeData(const QString& data) {
        qDebug() << "Consumed:" << data;
    }
};

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    ProducerThread producer;
    ConsumerThread consumer;
    
    QObject::connect(&producer, &ProducerThread::dataProduced,
                     &consumer, &ConsumerThread::consumeData,
                     Qt::QueuedConnection);
    
    producer.start();
    consumer.start();
    
    producer.wait();
    consumer.quit();
    consumer.wait();
    
    return 0;
}
```

**4. 需要特定线程优先级**

```cpp
class PriorityThread : public QThread {
    Q_OBJECT
public:
    PriorityThread(QThread::Priority priority, QObject* parent = nullptr)
        : QThread(parent) {
        setPriority(priority);
    }
    
protected:
    void run() override {
        qDebug() << "Thread priority:" << priority();
        qDebug() << "Thread ID:" << QThread::currentThreadId();
        
        for (int i = 0; i < 10; i++) {
            qDebug() << "Working:" << i;
            QThread::msleep(100);
        }
    }
};

int main(int argc, char* argv[]) {
    QCoreApplication app(argc, argv);
    
    PriorityThread high(QThread::HighPriority);
    PriorityThread normal(QThread::NormalPriority);
    PriorityThread low(QThread::LowPriority);
    
    high.start();
    normal.start();
    low.start();
    
    high.wait();
    normal.wait();
    low.wait();
    
    return 0;
}
```

### 线程同步

**互斥锁：**

```cpp
class Counter {
public:
    void increment() {
        QMutexLocker locker(&m_mutex);
        m_count++;
    }
    
    int value() const {
        QMutexLocker locker(&m_mutex);
        return m_count;
    }
    
private:
    mutable QMutex m_mutex;
    int m_count = 0;
};
```

**读写锁：**

```cpp
class ThreadSafeMap {
public:
    void insert(const QString& key, int value) {
        QWriteLocker locker(&m_lock);
        m_map[key] = value;
    }
    
    bool getValue(const QString& key, int& value) const {
        QReadLocker locker(&m_lock);
        auto it = m_map.find(key);
        if (it != m_map.end()) {
            value = it.value();
            return true;
        }
        return false;
    }
    
private:
    mutable QReadWriteLock m_lock;
    QMap<QString, int> m_map;
};
```

**信号量：**

```cpp
class BoundedBuffer {
public:
    BoundedBuffer(int size) : m_size(size), m_available(size) {}
    
    void put(int item) {
        m_available.acquire();
        m_mutex.lock();
        m_buffer.append(item);
        qDebug() << "Put:" << item;
        m_mutex.unlock();
        m_used.release();
    }
    
    int take() {
        m_used.acquire();
        m_mutex.lock();
        int item = m_buffer.takeFirst();
        qDebug() << "Take:" << item;
        m_mutex.unlock();
        m_available.release();
        return item;
    }
    
private:
    QMutex m_mutex;
    QSemaphore m_available;
    QSemaphore m_used;
    QList<int> m_buffer;
    int m_size;
};
```

### 总结

**使用QThread的场景：**
- 需要长时间运行的独立线程
- 需要精确控制线程生命周期
- 需要频繁的线程间通信
- 需要特定线程优先级
- 需要复杂的线程同步

**使用QtConcurrent的场景：**
- 简单的并行任务
- 短时间计算任务
- 不需要复杂控制的并发
- map-reduce操作
- 自动使用线程池

**最佳实践：**
- 简单任务使用QtConcurrent
- 复杂任务使用QThread
- 合理使用线程池
- 注意线程安全
- 使用信号槽进行线程通信
