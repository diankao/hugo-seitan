# Qt窗体属性与全局对象

## 题目

Qt中有哪些常用的窗体属性？如何使用Qt的全局对象？

## 答案

### 窗体属性设置

**动态添加/移除窗体属性：**

```cpp
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        setupUi();
        
        //切换无边框模式
        connect(ui->btnToggleFrame, &QPushButton::clicked, 
                this, &MainWindow::toggleFrameless);
    }

private:
    void toggleFrameless() {
        Qt::WindowFlags flags = windowFlags();
        
        if (flags & Qt::FramelessWindowHint) {
            //移除无边框属性
            setWindowFlags(flags & ~Qt::FramelessWindowHint);
        } else {
            //添加无边框属性
            setWindowFlags(flags | Qt::FramelessWindowHint);
        }
        
        show();
    }
};
```

**Qt5.9新增方法：**

```cpp
//添加无边框属性
setWindowFlag(Qt::FramelessWindowHint, true);

//移除无边框属性
setWindowFlag(Qt::FramelessWindowHint, false);

//可以设置多个属性
setWindowFlags(Qt::Window | Qt::FramelessWindowHint | Qt::WindowStaysOnTopHint);
```

### 常用窗体属性

```cpp
//无边框窗体
setWindowFlags(Qt::FramelessWindowHint);

//不在任务栏显示
setWindowFlags(Qt::Tool);

//窗口置顶
setWindowFlags(Qt::WindowStaysOnTopHint);

//模态对话框
setWindowModality(Qt::ApplicationModal);

//自定义窗口
setWindowFlags(Qt::CustomizeWindowHint);

//保留窗口边框特性（可拉伸）
setWindowFlags(Qt::CustomizeWindowHint);

//最小化到托盘
setWindowFlags(Qt::Tool | Qt::FramelessWindowHint);
```

### Qt全局对象使用

**QApplication全局属性：**

```cpp
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        setupUi();
        
        //使用全局对象
        setupGlobalObjects();
    }

private:
    void setupGlobalObjects() {
        //全局剪切板
        QClipboard *clipboard = qApp->clipboard();
        clipboard->setText("Hello Qt");
        QString text = clipboard->text();
        
        //顶层控件集合
        QList<QWidget *> topWidgets = qApp->topLevelWidgets();
        qDebug() << "Top level widgets:" << topWidgets.count();
        
        //当前焦点控件
        QWidget *focusWidget = qApp->focusWidget();
        if (focusWidget) {
            qDebug() << "Focused widget:" << focusWidget->objectName();
        }
        
        //平台名称
        qDebug() << "Platform:" << qApp->platformName();
        
        //调用系统蜂鸣器
        qApp->beep();
        
        //关于Qt对话框
        qApp->aboutQt();
        
        //设置全局鼠标样式
        qApp->setOverrideCursor(Qt::WaitCursor);
        
        //恢复鼠标样式
        qApp->restoreOverrideCursor();
    }
};
```

### 全局鼠标操作

```cpp
//右键菜单弹出
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        setupUi();
        
        //安装事件过滤器
        qApp->installEventFilter(this);
    }

protected:
    bool eventFilter(QObject *watched, QEvent *event) override {
        if (event->type() == QEvent::MouseButtonPress) {
            QMouseEvent *mouseEvent = static_cast<QMouseEvent *>(event);
            
            //判断鼠标右键
            if (qApp->mouseButtons() == Qt::RightButton) {
                QMenu *menu = new QMenu;
                menu->addAction("Action 1");
                menu->addAction("Action 2");
                
                menu->exec(QCursor::pos());
                return true;
            }
        }
        
        return QMainWindow::eventFilter(watched, event);
    }
};

//获取全局鼠标坐标
QPoint globalPos = QCursor::pos();
qDebug() << "Global mouse position:" << globalPos;
```

### 阻塞信号

```cpp
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        setupUi();
        
        //填充下拉框时阻塞信号
        fillComboBox();
    }

private:
    void fillComboBox() {
        //方法1：使用blockSignals
        ui->comboBox->blockSignals(true);
        
        for (int i = 0; i <= 100; i++) {
            ui->comboBox->addItem(QString::number(i));
        }
        
        ui->comboBox->blockSignals(false);
        
        //方法2：临时断开连接
        disconnect(ui->comboBox, QOverload<int>::of(&QComboBox::currentIndexChanged),
                   this, &MainWindow::on_comboBox_currentIndexChanged);
        
        for (int i = 0; i <= 100; i++) {
            ui->comboBox->addItem(QString::number(i));
        }
        
        connect(ui->comboBox, QOverload<int>::of(&QComboBox::currentIndexChanged),
                this, &MainWindow::on_comboBox_currentIndexChanged);
    }

private slots:
    void on_comboBox_currentIndexChanged(int index) {
        qDebug() << "Current index changed:" << index;
    }
};
```

### 窗口大小调整后的悬停问题

```cpp
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        setupUi();
        
        connect(ui->btnMax, &QPushButton::clicked,
                this, &MainWindow::onMaximize);
    }

private slots:
    void onMaximize() {
        showMaximized();
        
        //最大化后悬停样式没有取消，需要模拟鼠标移动
        QEvent event(QEvent::Leave);
        QApplication::sendEvent(ui->btnMax, &event);
    }
};
```

### 窗体最小化后恢复处理

```cpp
class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {}

protected:
    void showEvent(QShowEvent *e) override {
        //解决最小化后恢复界面可能出现的假死冻结现象
        setAttribute(Qt::WA_Mapped);
        QMainWindow::showEvent(e);
    }
};
```

### 总结

**窗体属性要点：**
- 使用|添加属性，使用&~移除属性
- Qt5.9新增setWindowFlag方法
- 恢复可拉伸需要设置最小最大尺寸
- 使用setWindowModality设置模态

**全局对象使用：**
- qApp提供剪切板、焦点、鼠标等全局访问
- 使用QCursor::pos()获取全局鼠标坐标
- setDesktopSettingsAware控制系统样式
- 使用installEventFilter拦截全局事件

**最佳实践：**
- 填充控件时使用blockSignals避免不必要的信号
- 使用pri文件组织大型项目
- 编译器判断使用宏定义
- 注意窗口调整后的悬停样式问题
