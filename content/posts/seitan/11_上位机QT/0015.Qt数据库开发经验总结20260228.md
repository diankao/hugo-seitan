### 数据库表名大小写处理

```cpp
class DatabaseTableHelper : public QObject {
    Q_OBJECT

public:
    DatabaseTableHelper(QObject *parent = nullptr) : QObject(parent) {}

    void bindTable(const QString &dbType, 
                   QSqlTableModel *model, 
                   const QString &table) {
        QString flag = dbType.toUpper();
        
        if (flag == "POSTGRESQL") {
            model->setTable(table.toLower());
        } else if (flag == "ORACLE") {
            model->setTable(table.toUpper());
        } else {
            model->setTable(table);
        }
        
        model->setEditStrategy(QSqlTableModel::OnManualSubmit);
        model->select();
    }

    QString getTableName(const QString &dbType, 
                        const QString &table) {
        QString flag = dbType.toUpper();
        
        if (flag == "POSTGRESQL") {
            return table.toLower();
        } else if (flag == "ORACLE") {
            return table.toUpper();
        } else {
            return table;
        }
    }
};
```

### 数据库查询优化

```cpp
class QueryOptimizer : public QObject {
    Q_OBJECT

public:
    QueryOptimizer(QObject *parent = nullptr) : QObject(parent) {}

    QSqlQuery executeFastQuery(const QString &sql) {
        QSqlQuery query;
        
        //只向前查询，极大提高远程数据库查询效率
        query.setForwardOnly(true);
        
        if (query.exec(sql)) {
            return query;
        } else {
            qWarning() << "Query failed:" << query.lastError().text();
            return query;
        }
    }

    QSqlQuery executeQueryWithTransaction(const QString &sql) {
        QSqlDatabase db = QSqlDatabase::database();
        
        if (!db.transaction()) {
            qWarning() << "Transaction failed to start";
            return QSqlQuery();
        }

        QSqlQuery query;
        if (query.exec(sql)) {
            if (db.commit()) {
                return query;
            } else {
                db.rollback();
                qWarning() << "Commit failed";
            }
        } else {
            db.rollback();
            qWarning() << "Query failed";
        }

        return QSqlQuery();
    }

    QList<QVariantList> batchInsert(const QString &table,
                                    const QStringList &columns,
                                    const QList<QVariantList> &data) {
        QSqlDatabase db = QSqlDatabase::database();
        db.transaction();

        QSqlQuery query;
        QString placeholders;
        for (int i = 0; i < columns.size(); i++) {
            placeholders += (i == 0 ? "?" : ",?");
        }

        QString sql = QString("INSERT INTO %1 (%2) VALUES (%3)")
                      .arg(table)
                      .arg(columns.join(","))
                      .arg(placeholders);
        
        query.prepare(sql);

        foreach (const QVariantList &row, data) {
            for (int i = 0; i < row.size(); i++) {
                query.addBindValue(row[i]);
            }
            
            if (!query.exec()) {
                db.rollback();
                qWarning() << "Batch insert failed:" << query.lastError().text();
                return QList<QVariantList>();
            }
        }

        if (db.commit()) {
            return data;
        } else {
            return QList<QVariantList>();
        }
    }
};
```

### 数据库创建与删除

```cpp
class DatabaseManager : public QObject {
    Q_OBJECT

public:
    DatabaseManager(QObject *parent = nullptr) : QObject(parent) {}

    bool createDatabase(const QString &dbType,
                       const QString &host, int port,
                       const QString &dbName,
                       const QString &user,
                       const QString &password) {
        QSqlDatabase db = QSqlDatabase::addDatabase("QMYSQL");
        db.setHostName(host);
        db.setPort(port);
        db.setUserName(user);
        db.setPassword(password);

        if (db.open()) {
            QSqlQuery query(db);
            
            if (query.exec(QString("DROP DATABASE IF EXISTS %1").arg(dbName))) {
                qDebug() << "Database dropped";
            }

            if (query.exec(QString("CREATE DATABASE %1").arg(dbName))) {
                qDebug() << "Database created";
                return true;
            } else {
                qWarning() << "Create database failed:" << query.lastError().text();
            }
        } else {
            qWarning() << "Connection failed:" << db.lastError().text();
        }

        return false;
    }

    bool createTable(const QString &tableName) {
        QSqlQuery query;
        
        QString sql = QString(
            "CREATE TABLE IF NOT EXISTS %1 ("
            "id INTEGER PRIMARY KEY AUTOINCREMENT,"
            "name TEXT,"
            "age INTEGER,"
            "score REAL"
            ")").arg(tableName);
        
        if (query.exec(sql)) {
            qDebug() << "Table created successfully";
            return true;
        } else {
            qWarning() << "Create table failed:" << query.lastError().text();
            return false;
        }
    }
};
```

### 自定义查询模型

```cpp
class CustomSqlQueryModel : public QSqlQueryModel {
    Q_OBJECT

public:
    CustomSqlQueryModel(QObject *parent = nullptr) : QSqlQueryModel(parent) {}

    QVariant data(const QModelIndex &index, int role) const override {
        QVariant value = QSqlQueryModel::data(index, role);
        
        if (role == Qt::DisplayRole) {
            int intValue = value.toInt();
            
            if (intValue >= 1000000) {
                return QString::number(intValue);
            }
            
            double doubleValue = value.toDouble();
            if (doubleValue > 1000000.0) {
                return QString::number(doubleValue, 'f', 0);
            }
        }
        
        if (role == Qt::TextAlignmentRole) {
            return Qt::AlignCenter;
        }
        
        if (role == Qt::FontRole) {
            QFont font;
            font.setFamily("Arial");
            font.setPointSize(9);
            return font;
        }
        
        return value;
    }

    void setQuery(const QSqlQuery &query) {
        QSqlQueryModel::setQuery(query);
        
        if (lastError().isValid()) {
            qWarning() << "Query error:" << lastError().text();
        }
    }
};
```

### 数据库事务处理

```cpp
class TransactionHelper : public QObject {
    Q_OBJECT

public:
    TransactionHelper(QObject *parent = nullptr) : QObject(parent) {}

    bool executeInTransaction(const QStringList &sqlStatements) {
        QSqlDatabase db = QSqlDatabase::database();
        
        if (!db.driver()->hasFeature(QSqlDriver::Transactions)) {
            qWarning() << "Database does not support transactions";
            return false;
        }

        if (!db.transaction()) {
            qWarning() << "Failed to start transaction";
            return false;
        }

        QSqlQuery query;
        foreach (const QString &sql, sqlStatements) {
            if (!query.exec(sql)) {
                db.rollback();
                qWarning() << "Transaction failed:" << query.lastError().text();
                return false;
            }
        }

        if (db.commit()) {
            qDebug() << "Transaction committed successfully";
            return true;
        } else {
            if (!db.lastError().isValid()) {
                qDebug() << "Transaction committed (MyISAM engine)";
                return true;
            }
            
            db.rollback();
            qWarning() << "Commit failed";
            return false;
        }
    }
};
```

### 总结

**数据库连接要点：**
- 注意32/64位对应关系
- MySQL需要libmysql.dll
- PostgreSQL需要多个库文件
- Oracle需要oci.dll等大文件
- ODBC只需设置DSN名称

**表名大小写处理：**
- PostgreSQL：全部小写
- Oracle：全部大写
- MySQL：不区分大小写
- SQLite：不区分大小写

**查询优化技巧：**
- 使用setForwardOnly(true)提高远程查询效率
- 使用事务批量操作
- 选择合适的数据库引擎
- 考虑使用ODBC或直连方式

**性能对比：**
- 千万级：sqlite > postgresql > oracle > mysql > odbc
- 千万级以上：postgresql > oracle > mysql > sqlite > odbc
- 亿级以上：oracle > postgresql > 其他

**注意事项：**
- 避免代理设置干扰
- 正确处理事务提交
- 注意数据库引擎特性
- 合理选择数据库类型
