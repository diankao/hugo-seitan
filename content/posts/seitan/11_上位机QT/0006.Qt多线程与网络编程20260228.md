# Qt多线程与网络编程

## 题目

Qt中多线程和网络编程有哪些注意事项？如何正确处理网络通信和线程问题？

## 答案

### 多线程常见误区

**不要滥用多线程：**

```cpp
//错误做法：简单任务也用线程
void simpleTask() {
    //这种简单操作不需要用线程
    int result = a + b;
}

//正确做法：真正耗时的操作才用线程
void heavyTask() {
    //耗时操作：编码解码、复杂计算、大文件处理
    for (int i = 0; i < 1000000; i++) {
        process(data[i]);
    }
}
```

### QTcpServer的incomingConnection

**Qt5版本变化：**

```cpp
class MyTcpServer : public QTcpServer {
    Q_OBJECT

public:
    MyTcpServer(QObject *parent = nullptr) : QTcpServer(parent) {}

protected:
#if (QT_VERSION > QT_VERSION_CHECK(5,0,0))
    void incomingConnection(qintptr handle) override {
#else
    void incomingConnection(int handle) override {
#endif
        QTcpSocket *socket = new QTcpSocket(this);
        socket->setSocketDescriptor(handle);
        
        connect(socket, &QTcpSocket::readyRead, this, [socket]() {
            QByteArray data = socket->readAll();
            processData(data);
        });
        
        connect(socket, &QTcpSocket::disconnected, socket, &QTcpSocket::deleteLater);
    }

private:
    void processData(const QByteArray &data) {
        qDebug() << "Received:" << data;
    }
};
```

### 网络通信异步特性

**串口和网络收发数据默认异步：**

```cpp
class NetworkClient : public QObject {
    Q_OBJECT

public:
    NetworkClient(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
        
        connect(socket, &QTcpSocket::connected, this, &NetworkClient::onConnected);
        connect(socket, &QTcpSocket::readyRead, this, &NetworkClient::onReadyRead);
        connect(socket, &QTcpSocket::disconnected, this, &NetworkClient::onDisconnected);
        
        socket->connectToHost("127.0.0.1", 8888);
    }

private slots:
    void onConnected() {
        qDebug() << "Connected to server";
    }

    void onReadyRead() {
        //异步接收数据，不会阻塞界面
        QByteArray data = socket->readAll();
        processData(data);
    }

    void onDisconnected() {
        qDebug() << "Disconnected from server";
    }

    void processData(const QByteArray &data) {
        //数据处理在这里进行
        //如果数据处理很复杂，可以放到工作线程
    }

private:
    QTcpSocket *socket;
};
```

### 数据处理移到工作线程

```cpp
class DataProcessor : public QObject {
    Q_OBJECT

public:
    DataProcessor(QObject *parent = nullptr) : QObject(parent) {}

public slots:
    void processData(const QByteArray &data) {
        //复杂的数据处理
        QByteArray result = heavyProcessing(data);
        emit dataProcessed(result);
    }

signals:
    void dataProcessed(const QByteArray &result);

private:
    QByteArray heavyProcessing(const QByteArray &data) {
        //耗时操作：解码、压缩、加密等
        QThread::msleep(100);
        return data.toUpper();
    }
};

class NetworkClient : public QObject {
    Q_OBJECT

public:
    NetworkClient(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
        
        QThread *workerThread = new QThread;
        DataProcessor *processor = new DataProcessor;
        processor->moveToThread(workerThread);
        
        connect(workerThread, &QThread::started, workerThread, &QThread::quit);
        connect(workerThread, &QThread::finished, processor, &QObject::deleteLater);
        connect(workerThread, &QThread::finished, workerThread, &QThread::deleteLater);
        
        connect(socket, &QTcpSocket::readyRead, this, [this, processor]() {
            QByteArray data = socket->readAll();
            QMetaObject::invokeMethod(processor, "processData",
                                     Qt::QueuedConnection,
                                     Q_ARG(QByteArray, data));
        });
        
        connect(processor, &DataProcessor::dataProcessed,
                this, &NetworkClient::onDataProcessed);
        
        workerThread->start();
    }

private slots:
    void onDataProcessed(const QByteArray &result) {
        qDebug() << "Processed data:" << result;
    }

private:
    QTcpSocket *socket;
};
```

### TCP保活机制

```cpp
class TcpKeepAlive : public QObject {
    Q_OBJECT

public:
    TcpKeepAlive(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
        
        connect(socket, &QTcpSocket::connected, this, &TcpKeepAlive::enableKeepAlive);
        connect(socket, &QTcpSocket::readyRead, this, &TcpKeepAlive::onReadyRead);
    }

    void connectToHost(const QString &host, quint16 port) {
        socket->connectToHost(host, port);
    }

private slots:
    void enableKeepAlive() {
        int fd = socket->socketDescriptor();
        
        int keepAlive = 1;
        int keepIdle = 5;
        int keepInterval = 2;
        int keepCount = 2;
        
#ifdef Q_OS_WIN
        setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE,
                   (const char *)&keepAlive, sizeof(keepAlive));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE,
                   (const char *)&keepIdle, sizeof(keepIdle));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL,
                   (const char *)&keepInterval, sizeof(keepInterval));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT,
                   (const char *)&keepCount, sizeof(keepCount));
#else
        setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE,
                   &keepAlive, sizeof(keepAlive));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE,
                   &keepIdle, sizeof(keepIdle));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL,
                   &keepInterval, sizeof(keepInterval));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT,
                   &keepCount, sizeof(keepCount));
#endif
        
        qDebug() << "KeepAlive enabled";
    }

    void onReadyRead() {
        QByteArray data = socket->readAll();
        qDebug() << "Received:" << data;
    }

private:
    QTcpSocket *socket;
};
```

### 高并发注意事项

**Qt网络连接数限制：**

```cpp
class ConnectionManager : public QObject {
    Q_OBJECT

public:
    ConnectionManager(QObject *parent = nullptr) 
        : QObject(parent), maxConnections(500) {
        
        connectionPool.reserve(maxConnections);
    }

    void addConnection(QTcpSocket *socket) {
        if (connectionPool.size() < maxConnections) {
            connectionPool.append(socket);
            qDebug() << "Connection added, total:" << connectionPool.size();
        } else {
            socket->close();
            socket->deleteLater();
            qWarning() << "Connection limit reached";
        }
    }

    void removeConnection(QTcpSocket *socket) {
        connectionPool.removeAll(socket);
        qDebug() << "Connection removed, total:" << connectionPool.size();
    }

private:
    QList<QTcpSocket *> connectionPool;
    int maxConnections;
};
```

### UI绘制注意事项

**UI只能在主线程绘制：**

```cpp
class WorkerThread : public QThread {
    Q_OBJECT

public:
    WorkerThread(QObject *parent = nullptr) : QThread(parent) {}

protected:
    void run() override {
        //错误：在工作线程中更新UI
        // ui->label->setText("Hello");
        
        //正确：通过信号槽更新UI
        emit updateUI("Hello from worker thread");
        
        exec();
    }

signals:
    void updateUI(const QString &text);
};

class MainWindow : public QMainWindow {
    Q_OBJECT

public:
    MainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        ui->setupUi(this);
        
        WorkerThread *worker = new WorkerThread(this);
        
        connect(worker, &WorkerThread::updateUI,
                this, &MainWindow::onUpdateUI);
        
        worker->start();
    }

private slots:
    void onUpdateUI(const QString &text) {
        //在主线程中更新UI
        ui->label->setText(text);
    }

private:
    Ui::MainWindow *ui;
};
```

### 线程数量控制

```cpp
class ThreadPool : public QObject {
    Q_OBJECT

public:
    ThreadPool(QObject *parent = nullptr) 
        : QObject(parent), maxThreads(QThread::idealThreadCount()) {
        
        qDebug() << "CPU cores:" << maxThreads;
        activeThreads = 0;
    }

    void addTask(std::function<void()> task) {
        if (activeThreads < maxThreads) {
            QThread *thread = new QThread;
            Worker *worker = new Worker(task);
            
            worker->moveToThread(thread);
            
            connect(thread, &QThread::started, worker, &Worker::run);
            connect(worker, &Worker::finished, thread, &QThread::quit);
            connect(worker, &Worker::finished, worker, &QObject::deleteLater);
            connect(thread, &QThread::finished, thread, &QObject::deleteLater);
            
            connect(thread, &QThread::finished, this, [this]() {
                activeThreads--;
                qDebug() << "Thread finished, active:" << activeThreads;
            });
            
            activeThreads++;
            thread->start();
        } else {
            qWarning() << "Too many threads, waiting...";
        }
    }

private:
    int maxThreads;
    int activeThreads;
};

class Worker : public QObject {
    Q_OBJECT

public:
    Worker(std::function<void()> task) : m_task(task) {}

public slots:
    void run() {
        m_task();
        emit finished();
    }

signals:
    void finished();

private:
    std::function<void()> m_task;
};
```

### 总结

**多线程使用要点：**
- 不要滥用多线程，线程有调度开销
- 串口和网络收发数据默认异步，不需要额外用线程
- 数据处理复杂时才移到工作线程
- UI只能在主线程绘制
- 线程数量不要超过CPU核心数

**网络编程注意事项：**
- Qt网络连接数建议在500以内
- 使用TCP保活机制检测连接状态
- 注意incomingConnection的参数类型变化
- 避免阻塞UI的操作

**最佳实践：**
- 简单任务不需要用线程
- 耗时操作（编码解码、复杂计算）才用线程
- 使用信号槽进行跨线程通信
- 合理控制线程数量
- 监控连接数和资源使用
