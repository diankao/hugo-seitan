# Qt事件过滤器高级应用

## 题目

Qt事件过滤器如何使用？如何实现全局事件拦截？有哪些实际应用场景？

## 答案

### 事件过滤器基础

事件过滤器是Qt中的一种强大机制，允许一个对象监视另一个对象的事件。通过重写eventFilter函数，可以在事件到达目标对象之前拦截和处理事件。

```cpp
class EventFilterExample : public QObject {
    Q_OBJECT

public:
    EventFilterExample(QObject *parent = nullptr) : QObject(parent) {}

    bool eventFilter(QObject *watched, QEvent *event) override {
        QWidget *widget = qobject_cast<QWidget *>(watched);
        
        if (!widget) {
            return QObject::eventFilter(watched, event);
        }

        switch (event->type()) {
            case QEvent::MouseButtonPress:
                return handleMousePress(widget, static_cast<QMouseEvent *>(event));
            case QEvent::MouseMove:
                return handleMouseMove(widget, static_cast<QMouseEvent *>(event));
            case QEvent::MouseButtonRelease:
                return handleMouseRelease(widget, static_cast<QMouseEvent *>(event));
            case QEvent::KeyPress:
                return handleKeyPress(widget, static_cast<QKeyEvent *>(event));
            case QEvent::Resize:
                return handleResize(widget, static_cast<QResizeEvent *>(event));
            default:
                break;
        }

        return QObject::eventFilter(watched, event);
    }

private:
    bool handleMousePress(QWidget *widget, QMouseEvent *event) {
        if (event->button() == Qt::LeftButton) {
            qDebug() << "Mouse press on" << widget->objectName();
        }
        return false;
    }

    bool handleMouseMove(QWidget *widget, QMouseEvent *event) {
        if (widget->property("canMove").toBool()) {
            static QPoint mousePoint;
            static bool mousePressed = false;

            if (mousePressed) {
                widget->move(event->globalPos() - mousePoint);
                return true;
            }
        }
        return false;
    }

    bool handleMouseRelease(QWidget *widget, QMouseEvent *event) {
        Q_UNUSED(widget);
        Q_UNUSED(event);
        return false;
    }

    bool handleKeyPress(QWidget *widget, QKeyEvent *event) {
        if (event->key() == Qt::Key_Escape) {
            widget->close();
            return true;
        }
        return false;
    }

    bool handleResize(QWidget *widget, QResizeEvent *event) {
        qDebug() << "Widget resized to" << event->size();
        return false;
    }
};
```

### 全局事件过滤器

```cpp
class GlobalEventFilter : public QObject {
    Q_OBJECT

public:
    GlobalEventFilter(QObject *parent = nullptr) : QObject(parent) {}

    void install() {
        qApp->installEventFilter(this);
    }

    void uninstall() {
        qApp->removeEventFilter(this);
    }

    bool eventFilter(QObject *watched, QEvent *event) override {
        if (event->type() == QEvent::NonClientAreaMouseButtonPress) {
            qDebug() << "系统标题栏按下";
            handleTitleBarPress(watched, static_cast<QMouseEvent *>(event));
        } else if (event->type() == QEvent::NonClientAreaMouseButtonRelease) {
            qDebug() << "系统标题栏松开";
            handleTitleBarRelease(watched, static_cast<QMouseEvent *>(event));
        }

        return QObject::eventFilter(watched, event);
    }

private:
    void handleTitleBarPress(QObject *watched, QMouseEvent *event) {
        Q_UNUSED(watched);
        Q_UNUSED(event);
    }

    void handleTitleBarRelease(QObject *watched, QMouseEvent *event) {
        Q_UNUSED(watched);
        Q_UNUSED(event);
    }
};

void installGlobalFilter() {
    static GlobalEventFilter filter;
    filter.install();
}
```

### 无边框窗体移动

```cpp
class FramelessWindowMover : public QObject {
    Q_OBJECT

public:
    FramelessWindowMover(QObject *parent = nullptr) : QObject(parent) {}

    bool eventFilter(QObject *watched, QEvent *event) override {
        QWidget *window = qobject_cast<QWidget *>(watched);
        
        if (!window || !window->property("canMove").toBool()) {
            return QObject::eventFilter(watched, event);
        }

        QMouseEvent *mouseEvent = static_cast<QMouseEvent *>(event);

        if (mouseEvent->type() == QEvent::MouseButtonPress) {
            if (mouseEvent->button() == Qt::LeftButton) {
                mousePressed = true;
                mousePoint = mouseEvent->globalPos() - window->pos();
            }
        } else if (mouseEvent->type() == QEvent::MouseButtonRelease) {
            mousePressed = false;
        } else if (mouseEvent->type() == QEvent::MouseMove) {
            if (mousePressed) {
                window->move(mouseEvent->globalPos() - mousePoint);
                return true;
            }
        }

        return QObject::eventFilter(watched, event);
    }

private:
    QPoint mousePoint;
    bool mousePressed = false;
};

class FramelessWindow : public QWidget {
    Q_OBJECT

public:
    FramelessWindow(QWidget *parent = nullptr) : QWidget(parent) {
        setWindowFlags(Qt::FramelessWindowHint);
        setProperty("canMove", true);
        
        mover = new FramelessWindowMover(this);
        installEventFilter(mover);
    }

private:
    FramelessWindowMover *mover;
};
```

### 应用程序初始化

```cpp
class AppInitializer : public QObject {
    Q_OBJECT

public:
    AppInitializer(QObject *parent = nullptr) : QObject(parent) {}

    void initialize() {
        installEventFilter();
        setupFont();
        setStyle();
    }

private:
    void installEventFilter() {
        qApp->installEventFilter(this);
    }

    void setupFont() {
        QFont font("Microsoft YaHei", 9);
        qApp->setFont(font);
    }

    void setStyle() {
        qApp->setStyle("Fusion");
    }

    bool eventFilter(QObject *watched, QEvent *event) override {
        if (event->type() == QEvent::ApplicationActivate) {
            qDebug() << "Application activated";
        } else if (event->type() == QEvent::ApplicationDeactivate) {
            qDebug() << "Application deactivated";
        }

        return QObject::eventFilter(watched, event);
    }
};
```

### 性能优化建议

```cpp
class OptimizedEventFilter : public QObject {
    Q_OBJECT

public:
    OptimizedEventFilter(QObject *parent = nullptr) : QObject(parent) {}

    bool eventFilter(QObject *watched, QEvent *event) override {
        //快速过滤不关心的事件
        if (event->type() != QEvent::KeyPress &&
            event->type() != QEvent::MouseButtonPress) {
            return QObject::eventFilter(watched, event);
        }

        //避免在事件过滤器中执行耗时操作
        if (event->type() == QEvent::KeyPress) {
            QKeyEvent *keyEvent = static_cast<QKeyEvent *>(event);
            
            if (keyEvent->key() == Qt::Key_F1) {
                QTimer::singleShot(0, this, []() {
                    performAsyncOperation();
                });
            }
        }

        return QObject::eventFilter(watched, event);
    }

    static void performAsyncOperation() {
        qDebug() << "Performing async operation";
    }
};
```

### 总结

**事件过滤器特点：**
- 在事件到达目标对象前拦截
- 可以修改或丢弃事件
- 支持全局事件过滤
- 有一定性能损耗

**应用场景：**
- 全局快捷键处理
- 无边框窗体移动
- 事件统计和记录
- 统一事件处理逻辑
- 调试和监控

**注意事项：**
- 避免在过滤器中执行耗时操作
- 注意返回值影响事件传递
- 及时安装和卸载过滤器
- 全局过滤器会影响所有对象

**最佳实践：**
- 只过滤必要的事件
- 使用异步操作处理耗时任务
- 合理设置过滤器对象生命周期
- 注意事件传播规则
