# Qt字符串QString进制转换详解

## 题目

Qt中QString支持哪些进制转换？如何实现不同进制之间的相互转换？

## 答案

### QString支持的进制

QString支持2到36之间的任意进制转换，包括常用的二进制、八进制、十进制、十六进制等。

```cpp
class StringConverter : public QObject {
    Q_OBJECT

public:
    StringConverter(QObject *parent = nullptr) : QObject(parent) {}

    void demonstrateConversions() {
        char data[2];
        data[0] = 0x10;
        data[1] = 25;

        //二进制显示
        qDebug() << "2进制显示:" << QString::number(data[0], 2) 
                 << QString::number(data[1], 2);
        
        //五进制显示
        qDebug() << "5进制显示:" << QString::number(data[0], 5) 
                 << QString::number(data[1], 5);
        
        //十进制显示
        qDebug() << "10进制显示:" << QString::number(data[0]) 
                 << QString::number(data[1]);
        
        //十六进制显示
        qDebug() << "16进制显示:" << QString::number(data[0], 16) 
                 << QString::number(data[1], 16);
    }
};
```

### 二进制数据处理

```cpp
class BinaryDataHandler : public QObject {
    Q_OBJECT

public:
    BinaryDataHandler(QObject *parent = nullptr) : QObject(parent) {}

    QString bytesToHex(const QByteArray &data) {
        QString hex;
        for (int i = 0; i < data.size(); i++) {
            hex += QString("%1").arg((unsigned char)data[i], 2, 16, QChar('0'));
            if (i < data.size() - 1) {
                hex += " ";
            }
        }
        return hex;
    }

    QString bytesToBinary(const QByteArray &data) {
        QString binary;
        for (int i = 0; i < data.size(); i++) {
            binary += QString::number((unsigned char)data[i], 2);
            if (i < data.size() - 1) {
                binary += " ";
            }
        }
        return binary;
    }

    QByteArray hexToBytes(const QString &hex) {
        QByteArray data;
        QString cleanHex = hex;
        cleanHex.remove(' ');
        cleanHex.remove('0x');
        
        for (int i = 0; i < cleanHex.length(); i += 2) {
            QString byteStr = cleanHex.mid(i, 2);
            bool ok;
            char byte = (char)byteStr.toInt(&ok, 16);
            if (ok) {
                data.append(byte);
            }
        }
        
        return data;
    }
};
```

### 数据格式化

```cpp
class DataFormatter : public QObject {
    Q_OBJECT

public:
    DataFormatter(QObject *parent = nullptr) : QObject(parent) {}

    QString formatHexData(const QByteArray &data, int columns = 16) {
        QString result;
        
        for (int i = 0; i < data.size(); i++) {
            if (i % columns == 0) {
                if (i > 0) {
                    result += "\n";
                }
                result += QString("%1  ").arg(i, 4, 16, QChar('0'));
            }
            
            result += QString("%1").arg((unsigned char)data[i], 2, 16, QChar('0'));
            result += " ";
            
            if ((i + 1) % 8 == 0 && (i + 1) % 16 != 0) {
                result += " ";
            }
            
            if ((i + 1) % columns == 0) {
                result += " |";
                for (int j = i - columns + 1; j <= i; j++) {
                    char c = data[j];
                    if (c >= 32 && c <= 126) {
                        result += c;
                    } else {
                        result += '.';
                    }
                }
                result += "|";
            }
        }
        
        return result;
    }

    QString formatNumber(int value, int base = 10, int width = 0) {
        if (width > 0) {
            return QString("%1").arg(value, width, base, QChar('0'));
        }
        return QString::number(value, base);
    }

    QString formatAddress(quint32 address) {
        return QString("0x%1").arg(address, 8, 16, QChar('0'));
    }
};
```

### 实际应用示例

```cpp
class ProtocolParser : public QObject {
    Q_OBJECT

public:
    ProtocolParser(QObject *parent = nullptr) : QObject(parent) {}

    struct ProtocolHeader {
        quint8 startByte;
        quint8 length;
        quint16 command;
        quint8 checksum;
    };

    QString parseHeader(const QByteArray &data) {
        QString result;
        
        if (data.size() >= 5) {
            ProtocolHeader header;
            header.startByte = data[0];
            header.length = data[1];
            header.command = (quint8)data[2] << 8 | (quint8)data[3];
            header.checksum = data[4];
            
            result += "协议头解析:\n";
            result += QString("起始字节: 0x%1\n")
                     .arg(header.startByte, 2, 16, QChar('0'));
            result += QString("数据长度: %1 (0x%2)\n")
                     .arg(header.length)
                     .arg(header.length, 2, 16, QChar('0'));
            result += QString("命令字: 0x%1\n")
                     .arg(header.command, 4, 16, QChar('0'));
            result += QString("校验和: 0x%1")
                     .arg(header.checksum, 2, 16, QChar('0'));
        }
        
        return result;
    }

    QByteArray buildHeader(quint16 command, quint8 length) {
        QByteArray data;
        data.append(0xAA);
        data.append(length);
        data.append((command >> 8) & 0xFF);
        data.append(command & 0xFF);
        data.append(0x00);
        return data;
    }
};
```

### 总结

**QString进制转换：**
- 支持2-36任意进制
- QString::number(value, base)转换
- toInt(nullptr, base)反向转换

**常用进制：**
- 二进制(2)：二进制数据表示
- 八进制(8)：权限、文件模式
- 十进制(10)：常规数值
- 十六进制(16)：内存地址、颜色、协议

**转换技巧：**
- 使用arg进行格式化
- toUpper转换大写
- 前缀0x标识十六进制
- 补零保持固定宽度

**实际应用：**
- 协议解析
- 颜色转换
- 数据格式化
- 调试输出
