# Qt环境变量与系统设置

## 题目

Qt中如何设置和获取环境变量？有哪些常用的Qt环境变量？如何通过环境变量控制程序行为？

## 答案

### 环境变量设置与获取

```cpp
class EnvironmentManager : public QObject {
    Q_OBJECT

public:
    EnvironmentManager(QObject *parent = nullptr) : QObject(parent) {}

    void setEnvironmentVariable(const QString &name, const QString &value) {
        qputenv(name.toUtf8(), value.toUtf8());
        qDebug() << "Set environment variable:" << name << "=" << value;
    }

    QString getEnvironmentVariable(const QString &name) {
        QByteArray value = qgetenv(name.toUtf8());
        return QString::fromUtf8(value);
    }

    bool hasEnvironmentVariable(const QString &name) {
        return !qgetenv(name.toUtf8()).isEmpty();
    }

    void printAllEnvironmentVariables() {
        QStringList env = QProcess::systemEnvironment();
        foreach (const QString &var, env) {
            qDebug() << var;
        }
    }
};
```

### 程序初始化配置

```cpp
class AppInitializer : public QObject {
    Q_OBJECT

public:
    AppInitializer(QObject *parent = nullptr) : QObject(parent) {}

    void initialize(int argc, char *argv[]) {
        QApplication app(argc, argv);

        if (isLinuxSystem()) {
            qputenv("QT_QPA_PLATFORM", "xcb");
        }

        if (isWebEngineEnabled()) {
            qputenv("QTWEBENGINE_DISABLE_SANDBOX", "1");
        }

        if (isOpenGLEnabled()) {
            qputenv("QT_MEDIA_BACKEND", "ffmpeg");
        }

        app.exec();
    }

private:
    bool isLinuxSystem() {
#ifdef Q_OS_LINUX
        return true;
#else
        return false;
#endif
    }

    bool isWebEngineEnabled() {
#ifdef QT_WEBENGINEWIDGETS_LIB
        return true;
#else
        return false;
#endif
    }

    bool isOpenGLEnabled() {
#ifdef QT_OPENGL_LIB
        return true;
#else
        return false;
#endif
    }
};
```

### 配置文件读取

```cpp
class ConfigReader : public QObject {
    Q_OBJECT

public:
    ConfigReader(QObject *parent = nullptr) : QObject(parent) {}

    QString readIniValue(char *argv[], const QString &key, const QString &dir = QString()) {
        QString path, name;
        getCurrentInfo(argv, path, name);

        QString fileName = QString("%1/%2%3.ini").arg(path).arg(dir).arg(name);
        return readIniValue(fileName, key);
    }

    QString readIniValue(const QString &fileName, const QString &key) {
        QString value;
        QFile file(fileName);

        if (file.open(QFile::ReadOnly | QFile::Text)) {
            while (!file.atEnd()) {
                QString line = file.readLine();
                if (line.startsWith(key)) {
                    line = line.replace("\n", "");
                    line = line.trimmed();
                    value = line.split("=").last();
                    break;
                }
            }
        }

        return value;
    }

    void applyEnvironmentSettings(char *argv[]) {
        int openGLType = readIniValue(argv, "OpenGLType").toInt();
        int mediaBackend = readIniValue(argv, "MediaBackend").toInt();

        initOpenGL(openGLType);
        initMediaBackend(mediaBackend);
    }

private:
    void getCurrentInfo(char *argv[], QString &path, QString &name) {
        QString argv0 = QString::fromLocal8Bit(argv[0]);
        QFileInfo file(argv0);
        path = file.path();
        name = file.baseName();
    }

    void initOpenGL(int type) {
        switch (type) {
            case 0:
                qputenv("QT_OPENGL", "angle");
                break;
            case 1:
                qputenv("QT_OPENGL", "desktop");
                break;
            case 2:
                qputenv("QT_OPENGL", "software");
                break;
            default:
                break;
        }
    }

    void initMediaBackend(int backend) {
        switch (backend) {
            case 0:
                qputenv("QT_MEDIA_BACKEND", "windows");
                break;
            case 1:
                qputenv("QT_MEDIA_BACKEND", "ffmpeg");
                break;
            default:
                break;
        }
    }
};
```

### 平台特定设置

```cpp
class PlatformSettings : public QObject {
    Q_OBJECT

public:
    PlatformSettings(QObject *parent = nullptr) : QObject(parent) {}

    void applyPlatformSettings() {
#ifdef Q_OS_WIN
        applyWindowsSettings();
#elif defined(Q_OS_LINUX)
        applyLinuxSettings();
#elif defined(Q_OS_MAC)
        applyMacSettings();
#endif
    }

private:
    void applyWindowsSettings() {
        qputenv("QT_QPA_PLATFORM", "windows");
        qputenv("QT_AUTO_SCREEN_SCALE_FACTOR", "1");
    }

    void applyLinuxSettings() {
        qputenv("QT_QPA_PLATFORM", "xcb");
    }

    void applyMacSettings() {
        qputenv("QT_QPA_PLATFORM", "cocoa");
    }
};
```

### 总结

**常用环境变量：**
- QT_QPA_PLATFORM：指定平台插件
- QT_USE_NATIVE_WINDOWS：启用原生窗口
- QT_AUTO_SCREEN_SCALE_FACTOR：自动缩放
- QT_SCALE_FACTOR：缩放因子
- QT_MEDIA_BACKEND：多媒体后端
- QTWEBENGINE_DISABLE_SANDBOX：禁用沙箱

**设置方法：**
- qputenv：设置环境变量
- qgetenv：获取环境变量
- QProcess::systemEnvironment()：获取所有环境变量

**应用场景：**
- 控制平台选择
- 设置多媒体后端
- 调试输出控制
- DPI缩放设置
- 浏览器沙箱控制

**注意事项：**
- 环境变量设置后需要重启才能生效
- 修改环境变量会影响所有Qt程序
- 某些设置需要在QApplication创建前设置
- Linux系统可能需要重启才能生效
