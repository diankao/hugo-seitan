# Qt跨平台开发与条件编译

## 题目

如何在Qt中实现跨平台开发？pro文件中的条件编译语法有哪些？如何处理不同平台的差异？

## 答案

### pro文件平台判断

```qmake
# Windows平台
win32 {
    message("Building for Windows")
    DEFINES += WINDOWS_OS
    SOURCES += windows_specific.cpp
}

# Linux平台
unix {
    message("Building for Linux")
    DEFINES += LINUX_OS
    SOURCES += linux_specific.cpp
}

# macOS平台
macx {
    message("Building for macOS")
    DEFINES += MAC_OS
    SOURCES += mac_specific.cpp
}

# Android平台
android {
    message("Building for Android")
    DEFINES += ANDROID_OS
}

# iOS平台
ios {
    message("Building for iOS")
    DEFINES += IOS_OS
}
```

### 硬件平台判断

```qmake
# 32位平台
contains(QT_ARCH, i386) {
    message("Building for 32-bit")
    DEFINES += ARCH_32BIT
}

# 64位平台
contains(QT_ARCH, x86_64) {
    message("Building for 64-bit")
    DEFINES += ARCH_64BIT
}

# ARM平台
contains(QT_ARCH, arm) {
    message("Building for ARM")
    DEFINES += ARCH_ARM
}

# ARM64平台
contains(QT_ARCH, arm64) {
    message("Building for ARM64")
    DEFINES += ARCH_ARM64
}
```

### 编译器判断

```qmake
# MSVC编译器
msvc {
    message("Using MSVC compiler")
    DEFINES += USE_MSVC
    QMAKE_CXXFLAGS += /wd4996
}

# MinGW编译器
mingw {
    message("Using MinGW compiler")
    DEFINES += USE_MINGW
}

# GCC编译器
unix:!macx {
    message("Using GCC compiler")
    DEFINES += USE_GCC
}

# Clang编译器
clang {
    message("Using Clang compiler")
    DEFINES += USE_CLANG
}
```

### 复杂条件判断

```qmake
# 表示安卓或者ios平台
android|ios {
    message("Building for mobile platform")
}

# 表示非安卓和非ios平台
!android::!ios {
    message("Building for desktop platform")
}

# 表示非ios系统的mac系统
macx:!ios {
    message("Building for macOS")
}

# 表示非mac系统的unix系统
unix:!macx {
    message("Building for Linux")
}

# 两个条件都要满足
win32::msvc {
    message("Windows with MSVC")
}

# 两个条件满足其一
win32|unix {
    message("Windows or Unix")
}

# 非Windows平台
!win32 {
    message("Not Windows")
}
```

### Qt版本判断

```qmake
# 主版本判断
greaterThan(QT_MAJOR_VERSION, 4) {
    message("Qt version >= 5")
    QT += widgets
}

greaterThan(QT_MAJOR_VERSION, 5) {
    message("Qt version >= 6")
    DEFINES += QT6_VERSION
}

# 子版本判断
greaterThan(QT_MAJOR_VERSION, 4) {
    greaterThan(QT_MINOR_VERSION, 6) {
        message("Qt version >= 5.7")
        DEFINES += QT_VERSION_57
    }
}

# 综合版本判断
greaterThan(QT_MAJOR_VERSION, 4)|greaterThan(QT_MINOR_VERSION, 7) {
    message("Qt version >= 4.8")
}

# 具体版本判断
greaterThan(QT_VERSION, 5.12.3) {
    message("Qt version >= 5.12.3")
}

# 最小版本要求
!minQtVersion(5, 15, 2) {
    message("Cannot build with Qt version $${QT_VERSION}")
    error("Use at least Qt 5.15.2")
}
```

### 模块判断

```qmake
# 判断套件是否存在某个模块
qtHaveModule(multimedia) {
    message("Multimedia module available")
    QT += multimedia
}

# 判断是否引入过某个模块
contains(QT, multimedia) {
    message("Multimedia module enabled")
    DEFINES += QT_MULTIMEDIA_LIB
}

# 条件引入模块
greaterThan(QT_MAJOR_VERSION, 4) {
    greaterThan(QT_MINOR_VERSION, 6) {
        DEFINES += qchart
    }
}

# 根据定义引入模块
contains(DEFINES, qchart) {
    QT += charts
}
```

### 路径处理

```qmake
# 区分不同的系统和位数
win32 {
    path_sys = win
} unix:!macx {
    path_sys = linux
} macx {
    path_sys = mac
}

contains(QT_ARCH, x86_64) {
    path_arch = x64
} else {
    path_arch = x86
}

# 拼接路径
path_lib = $$PWD/lib/$$path_sys/$$path_arch
path_include = $$PWD/include

# 引入库文件
LIBS += -L$$path_lib -llibmylib
INCLUDEPATH += $$path_include

# 32位和64位不同处理
win32 {
    contains(QT_ARCH, x86_64) {
        LIBS += -L$$PWD/lib/x64
    } else {
        LIBS += -L$$PWD/lib/x86
    }
}
```

### 代码中的条件编译

```cpp
// 平台判断
#ifdef Q_OS_WIN
    qDebug() << "Windows platform";
#elif defined(Q_OS_LINUX)
    qDebug() << "Linux platform";
#elif defined(Q_OS_MAC)
    qDebug() << "Mac platform";
#endif

// 编译器判断
#ifdef Q_CC_MSVC
    qDebug() << "MSVC compiler";
#elif defined(Q_CC_GNU)
    qDebug() << "GCC compiler";
#elif defined(Q_CC_CLANG)
    qDebug() << "Clang compiler";
#endif

// 位数判断
#if defined(Q_PROCESSOR_X86_64)
    qDebug() << "64-bit";
#elif defined(Q_PROCESSOR_X86_32)
    qDebug() << "32-bit";
#endif

// Qt版本判断
#if (QT_VERSION >= QT_VERSION_CHECK(5, 0, 0))
    qDebug() << "Qt 5 or later";
#else
    qDebug() << "Qt 4";
#endif

#if (QT_VERSION >= QT_VERSION_CHECK(6, 0, 0))
    qDebug() << "Qt 6 or later";
#endif

// 模块判断
#ifdef QT_MULTIMEDIA_LIB
    qDebug() << "Multimedia module enabled";
#endif

#ifdef QT_SQL_LIB
    qDebug() << "SQL module enabled";
#endif

// 自定义定义
#ifdef DEBUG_MODE
    qDebug() << "Debug build";
#else
    qDebug() << "Release build";
#endif
```

### 总结

**平台判断：**
- win32：Windows平台
- unix：Unix系统（包括Linux和macOS）
- macx：macOS平台
- android：Android平台
- ios：iOS平台

**硬件平台：**
- i386：32位x86
- x86_64：64位x86
- arm：ARM
- arm64：ARM64

**条件运算符：**
- !：非
- |：或
- ::：与
- contains：包含
- greaterThan：大于

**Qt版本判断：**
- QT_MAJOR_VERSION：主版本号
- QT_MINOR_VERSION：次版本号
- QT_VERSION：版本号
- minQtVersion：最小版本要求

**最佳实践：**
- 使用平台宏定义条件编译
- 统一路径管理
- 模块动态引入
- 版本兼容处理
