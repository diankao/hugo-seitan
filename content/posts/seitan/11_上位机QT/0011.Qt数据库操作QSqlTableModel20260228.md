# Qt数据库操作QSqlTableModel

## 题目

QSqlTableModel如何实现数据库的分页查询和增删改查操作？

## 答案

### 基本数据库模型设置

```cpp
class DatabaseManager : public QObject {
    Q_OBJECT

public:
    DatabaseManager(QObject *parent = nullptr) : QObject(parent) {
        //连接数据库
        db = QSqlDatabase::addDatabase("QSQLITE");
        db.setDatabaseName("test.db");
        
        if (db.open()) {
            qDebug() << "Database opened";
            initializeDatabase();
        } else {
            qWarning() << "Failed to open database:" << db.lastError().text();
        }
    }

private:
    void initializeDatabase() {
        QSqlQuery query;
        
        //创建表
        query.exec("CREATE TABLE IF NOT EXISTS students ("
                  "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                  "name TEXT,"
                  "age INTEGER,"
                  "score REAL)");
        
        //插入测试数据
        for (int i = 1; i <= 100; i++) {
            query.prepare("INSERT INTO students (name, age, score) "
                        "VALUES (?, ?, ?)");
            query.addBindValue(QString("Student%1").arg(i));
            query.addBindValue(20 + i % 5);
            query.addBindValue(60.0 + (i % 40));
            query.exec();
        }
    }

private:
    QSqlDatabase db;
};
```

### QSqlTableModel基础使用

```cpp
class TableViewModel : public QObject {
    Q_OBJECT

public:
    TableViewModel(QObject *parent = nullptr) : QObject(parent) {
        model = new QSqlTableModel(this);
        
        //指定表名
        model->setTable("students");
        
        //设置列排序
        model->setSort(0, Qt::AscendingOrder);
        
        //设置提交模式
        //OnFieldChange: 字段改变时立即提交
        //OnRowChange: 行改变时提交
        //OnManualSubmit: 手动提交（推荐）
        model->setEditStrategy(QSqlTableModel::OnManualSubmit);
        
        //立即查询
        model->select();
    }

    QSqlTableModel *getModel() {
        return model;
    }

private:
    QSqlTableModel *model;
};
```

### 分页查询实现

```cpp
class PaginationModel : public QObject {
    Q_OBJECT

public:
    PaginationModel(QObject *parent = nullptr) : QObject(parent) {
        model = new QSqlTableModel(this);
        model->setTable("students");
        model->setEditStrategy(QSqlTableModel::OnManualSubmit);
        
        currentPage = 1;
        pageSize = 10;
    }

    void loadPage(int page) {
        currentPage = page;
        
        //计算偏移量
        int offset = (currentPage - 1) * pageSize;
        
        //使用setFilter实现分页
        QString filter = QString("1=1 order by id asc limit %1,%2;")
                          .arg(offset)
                          .arg(pageSize);
        model->setFilter(filter);
        model->select();
        
        qDebug() << "Loaded page" << currentPage << "offset:" << offset;
    }

    void nextPage() {
        int offset = (currentPage - 1) * pageSize;
        model->select();
        
        if (model->rowCount() == pageSize) {
            loadPage(currentPage + 1);
        }
    }

    void prevPage() {
        if (currentPage > 1) {
            loadPage(currentPage - 1);
        }
    }

    int getCurrentPage() const {
        return currentPage;
    }

    int getPageSize() const {
        return pageSize;
    }

    QSqlTableModel *getModel() {
        return model;
    }

private:
    QSqlTableModel *model;
    int currentPage;
    int pageSize;
};
```

### 条件查询与排序

```cpp
class QueryModel : public QObject {
    Q_OBJECT

public:
    QueryModel(QObject *parent = nullptr) : QObject(parent) {
        model = new QSqlTableModel(this);
        model->setTable("students");
        model->setEditStrategy(QSqlTableModel::OnManualSubmit);
    }

    //简单条件查询
    void queryByName(const QString &name) {
        model->setFilter(QString("name='%1'").arg(name));
        model->select();
    }

    //复合条件查询
    void queryByScoreRange(double minScore, double maxScore) {
        model->setFilter(QString("score >= %1 and score <= %2")
                        .arg(minScore)
                        .arg(maxScore));
        model->select();
    }

    //带排序的分页查询
    void queryWithSort(const QString &fieldName, Qt::SortOrder order) {
        model->setSort(model->fieldIndex(fieldName), order);
        
        int offset = (currentPage - 1) * pageSize;
        model->setFilter(QString("1=1 order by %1 %2 limit %3,%4;")
                        .arg(fieldName)
                        .arg(order == Qt::AscendingOrder ? "asc" : "desc")
                        .arg(offset)
                        .arg(pageSize));
        model->select();
    }

    //复杂条件查询
    void complexQuery(const QString &name, int minAge, int maxAge) {
        model->setFilter(QString("1=1 and name='%1' and age >= %2 and age <= %3;")
                        .arg(name)
                        .arg(minAge)
                        .arg(maxAge));
        model->select();
    }

    QSqlTableModel *getModel() {
        return model;
    }

private:
    QSqlTableModel *model;
    int currentPage = 1;
    int pageSize = 10;
};
```

### 增删改操作

```cpp
class DatabaseOperations : public QObject {
    Q_OBJECT

public:
    DatabaseOperations(QObject *parent = nullptr) : QObject(parent) {
        model = new QSqlTableModel(this);
        model->setTable("students");
        model->setEditStrategy(QSqlTableModel::OnManualSubmit);
        model->select();
    }

    //添加记录
    bool addRecord(const QString &name, int age, double score) {
        int row = model->rowCount();
        
        if (model->insertRow(row)) {
            model->setData(model->index(row, 1), name);
            model->setData(model->index(row, 2), age);
            model->setData(model->index(row, 3), score);
            
            return model->submitAll();
        }
        
        return false;
    }

    //删除记录
    bool deleteRecord(int row) {
        if (model->removeRow(row)) {
            return model->submitAll();
        }
        return false;
    }

    //批量删除
    bool batchDelete(const QList<int> &rows) {
        //从后往前删除，避免索引变化
        for (int i = rows.size() - 1; i >= 0; i--) {
            model->removeRow(rows[i]);
        }
        
        return model->submitAll();
    }

    //更新记录
    bool updateRecord(int row, const QString &name, int age, double score) {
        model->setData(model->index(row, 1), name);
        model->setData(model->index(row, 2), age);
        model->setData(model->index(row, 3), score);
        
        return model->submitAll();
    }

    //撤销更改
    void revertChanges() {
        model->revertAll();
    }

    QSqlTableModel *getModel() {
        return model;
    }

private:
    QSqlTableModel *model;
};
```

class MainWindow : public QMainWindow {

        });
        
        connect(ui->btnPrevPage, &QPushButton::clicked,
                this, [this]() {
            pagination->prevPage();
            ui->tableView->setModel(pagination->getModel());
        });
        
        connect(ui->btnNextPage, &QPushButton::clicked,
                this, [this]() {
            pagination->nextPage();
            ui->tableView->setModel(pagination->getModel());
        });
        
        connect(ui->btnAdd, &QPushButton::clicked,
                this, &MainWindow::onAddRecord);
        
        connect(ui->btnDelete, &QPushButton::clicked,
                this, &MainWindow::onDeleteRecord);
        
        connect(ui->btnUpdate, &QPushButton::clicked,
                this, &MainWindow::onUpdateRecord);
    }

    void onAddRecord() {
        QString name = ui->lineEditName->text();
        int age = ui->spinBoxAge->value();
        double score = ui->doubleSpinBoxScore->value();
        
        if (operations->addRecord(name, age, score)) {
            QMessageBox::information(this, "Success", "Record added successfully");
            ui->tableView->setModel(operations->getModel());
        } else {
            QMessageBox::warning(this, "Error", "Failed to add record");
        }
    }

    void onDeleteRecord() {
        int row = ui->tableView->currentIndex().row();
        
        if (row >= 0) {
            if (operations->deleteRecord(row)) {
                QMessageBox::information(this, "Success", "Record deleted successfully");
                ui->tableView->setModel(operations->getModel());
            }
        }
    }

    void onUpdateRecord() {
        int row = ui->tableView->currentIndex().row();
        
        if (row >= 0) {
            QString name = ui->lineEditName->text();
            int age = ui->spinBoxAge->value();
            double score = ui->doubleSpinBoxScore->value();
            
            if (operations->updateRecord(row, name, age, score)) {
                QMessageBox::information(this, "Success", "Record updated successfully");
                ui->tableView->setModel(operations->getModel());
            }
        }
    }

    DatabaseManager *dbManager;
    TableViewModel *tableModel;

### 总结

**QSqlTableModel要点：**
- 使用OnManualSubmit策略手动提交
- setFilter实现条件查询和分页
- setSort设置排序字段
- 增删改后需要submitAll提交

**分页查询：**
- 使用limit offset, pageSize语法
- offset = (page - 1) * pageSize
- setFilter中可以包含where、order by、limit

**注意事项：**
- 如果setFilter中有排序，不要再使用setSort
- 建议setFilter最前面写1=1，末尾加;
- 批量删除从后往前删除避免索引变化
- revertAll撤销未提交的更改

**最佳实践：**
- 使用OnManualSubmit策略批量操作
- 分页查询在setFilter中实现
- 条件查询时注意SQL注入问题
- 及时提交或撤销更改
