# Qt网络编程TCPUDP详解

## 题目

Qt中TCP和UDP编程有哪些注意事项？如何正确使用网络编程？

## 答案

### TCP客户端绑定端口

```cpp
class TcpClient : public QObject {
    Q_OBJECT

public:
    TcpClient(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
        
        connect(socket, &QTcpSocket::connected, 
                this, &TcpClient::onConnected);
        connect(socket, &QTcpSocket::readyRead, 
                this, &TcpClient::onReadyRead);
        connect(socket, &QTcpSocket::disconnected, 
                this, &TcpClient::onDisconnected);
        connect(socket, QOverload<QAbstractSocket::SocketError>::of(&QTcpSocket::error),
                this, &TcpClient::onError);
    }

    void connectToServer(const QString &host, quint16 port, 
                         const QString &localIp, quint16 localPort) {
        //断开所有连接和操作
        socket->abort();
        
        //绑定本地网卡和端口
        socket->bind(QHostAddress(localIp), localPort);
        
        //连接服务器
        socket->connectToHost(QHostAddress(host), port);
    }

private slots:
    void onConnected() {
        qDebug() << "Connected to server";
        qDebug() << "Local address:" << socket->localAddress() 
                 << "port:" << socket->localPort();
        qDebug() << "Peer address:" << socket->peerAddress() 
                 << "port:" << socket->peerPort();
    }

    void onReadyRead() {
        QByteArray data = socket->readAll();
        qDebug() << "Received:" << data;
        emit dataReceived(data);
    }

    void onDisconnected() {
        qDebug() << "Disconnected from server";
    }

    void onError(QAbstractSocket::SocketError socketError) {
        qWarning() << "Socket error:" << socketError 
                   << socket->errorString();
    }

signals:
    void dataReceived(const QByteArray &data);

private:
    QTcpSocket *socket;
};
```

### UDP客户端绑定端口

```cpp
class UdpClient : public QObject {
    Q_OBJECT

public:
    UdpClient(QObject *parent = nullptr) : QObject(parent) {
        socket = new QUdpSocket(this);
        
        connect(socket, &QUdpSocket::readyRead, 
                this, &UdpClient::onReadyRead);
    }

    void bindPort(const QString &localIp, quint16 localPort) {
        //没有绑定过才需要绑定
        if (socket->localPort() != localPort) {
            socket->abort();
            socket->bind(QHostAddress(localIp), localPort);
            
            qDebug() << "UDP bound to:" << socket->localAddress() 
                     << "port:" << socket->localPort();
        }
    }

    void sendData(const QByteArray &data, 
                  const QString &remoteIp, quint16 remotePort) {
        socket->writeDatagram(data, 
                             QHostAddress(remoteIp), 
                             remotePort);
    }

private slots:
    void onReadyRead() {
        while (socket->hasPendingDatagrams()) {
            QByteArray datagram;
            datagram.resize(socket->pendingDatagramSize());
            
            QHostAddress sender;
            quint16 senderPort;
            
            socket->readDatagram(datagram.data(), 
                                datagram.size(),
                                &sender, 
                                &senderPort);
            
            qDebug() << "Received from:" << sender.toString() 
                     << "port:" << senderPort 
                     << "data:" << datagram;
            
            emit dataReceived(datagram, sender, senderPort);
        }
    }

signals:
    void dataReceived(const QByteArray &data, 
                     const QHostAddress &sender, 
                     quint16 senderPort);

private:
    QUdpSocket *socket;
};
```

### TCP服务端

```cpp
class TcpServer : public QObject {
    Q_OBJECT

public:
    TcpServer(QObject *parent = nullptr) : QObject(parent) {
        server = new QTcpServer(this);
        
        connect(server, &QTcpServer::newConnection,
                this, &TcpServer::onNewConnection);
    }

    void start(const QString &host, quint16 port) {
        if (server->listen(QHostAddress(host), port)) {
            qDebug() << "Server started on:" << host << port;
        } else {
            qWarning() << "Server start failed:" << server->errorString();
        }
    }

private slots:
    void onNewConnection() {
        QTcpSocket *clientSocket = server->nextPendingConnection();
        
        qDebug() << "New connection from:" 
                 << clientSocket->peerAddress().toString()
                 << "port:" << clientSocket->peerPort();
        
        connect(clientSocket, &QTcpSocket::readyRead, 
                this, [this, clientSocket]() {
            QByteArray data = clientSocket->readAll();
            qDebug() << "Received from client:" << data;
            emit dataReceived(clientSocket, data);
        });
        
        connect(clientSocket, &QTcpSocket::disconnected,
                clientSocket, &QTcpSocket::deleteLater);
        
        connect(clientSocket, &QTcpSocket::disconnected, 
                this, [clientSocket]() {
            qDebug() << "Client disconnected";
        });
    }

signals:
    void dataReceived(QTcpSocket *socket, const QByteArray &data);

private:
    QTcpServer *server;
};
```

### TCP保活机制

```cpp
class TcpKeepAlive : public QObject {
    Q_OBJECT

public:
    TcpKeepAlive(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
    }

    void enableKeepAlive() {
        int fd = socket->socketDescriptor();
        
        int keepAlive = 1;
        int keepIdle = 5;
        int keepInterval = 2;
        int keepCount = 2;
        
#ifdef Q_OS_WIN
        setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE,
                   (const char *)&keepAlive, sizeof(keepAlive));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE,
                   (const char *)&keepIdle, sizeof(keepIdle));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL,
                   (const char *)&keepInterval, sizeof(keepInterval));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT,
                   (const char *)&keepCount, sizeof(keepCount));
#else
        setsockopt(fd, SOL_SOCKET, SO_KEEPALIVE,
                   &keepAlive, sizeof(keepAlive));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPIDLE,
                   &keepIdle, sizeof(keepIdle));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPINTVL,
                   &keepInterval, sizeof(keepInterval));
        setsockopt(fd, IPPROTO_TCP, TCP_KEEPCNT,
                   &keepCount, sizeof(keepCount));
#endif
        
        qDebug() << "KeepAlive enabled";
        qDebug() << "keepIdle:" << keepIdle 
                 << "keepInterval:" << keepInterval 
                 << "keepCount:" << keepCount;
    }

    QTcpSocket *getSocket() {
        return socket;
    }

private:
    QTcpSocket *socket;
};
```

### 网络超时设置

```cpp
class NetworkRequest : public QObject {
    Q_OBJECT

public:
    NetworkRequest(QObject *parent = nullptr) : QObject(parent) {
        manager = new QNetworkAccessManager(this);
        timeout = 5000;
    }

    void get(const QString &url, int timeoutMs = 0) {
        if (timeoutMs > 0) {
            timeout = timeoutMs;
        }
        
        QEventLoop eventLoop;
        
        //Qt5.15开始支持超时设置
#if (QT_VERSION >= QT_VERSION_CHECK(5,15,0))
        manager->setTransferTimeout(timeout);
#else
        QTimer timer;
        connect(&timer, &QTimer::timeout, &eventLoop, &QEventLoop::quit);
        timer.setSingleShot(true);
        timer.start(timeout);
#endif
        
        QNetworkReply *reply = manager->get(QNetworkRequest(QUrl(url)));
        
        connect(reply, &QNetworkReply::finished, 
                &eventLoop, &QEventLoop::quit);
        
        eventLoop.exec();
        
        if (reply->bytesAvailable() > 0 && 
            reply->error() == QNetworkReply::NoError) {
            QByteArray data = reply->readAll();
            emit requestFinished(data);
        } else {
            qWarning() << "Request failed:" << reply->errorString();
            emit requestFailed(reply->errorString());
        }
        
        reply->deleteLater();
    }

signals:
    void requestFinished(const QByteArray &data);
    void requestFailed(const QString &error);

private:
    QNetworkAccessManager *manager;
    int timeout;
};
```

### TCP与UDP端口独立性

```cpp
//TCP和UDP可以同时使用相同端口
```

### 客户端端口分配说明

```cpp
class PortExplanation : public QObject {
    Q_OBJECT

public:
    PortExplanation(QObject *parent = nullptr) : QObject(parent) {
        socket = new QTcpSocket(this);
        
        connect(socket, &QTcpSocket::connected, 
                this, &PortExplanation::showPortInfo);
    }

    void connectWithoutBind() {
        //客户端没有绑定，操作系统随机分配端口
        socket->connectToHost("192.168.1.3", 6000);
        
        //端口一般按顺序递增：45110, 45111, 45112...
    }

    void connectWithBind() {
        //客户端绑定指定端口
        socket->abort();
        socket->bind(QHostAddress("192.168.1.2"), 6005);
        socket->connectToHost("192.168.1.3", 6000);
    }

private slots:
    void showPortInfo() {
        qDebug() << "Local address:" << socket->localAddress() 
                 << "port:" << socket->localPort();
        qDebug() << "Peer address:" << socket->peerAddress() 
                 << "port:" << socket->peerPort();
        
        //重要：客户端端口由自己决定，不是服务器分配
        //服务器只能决定自己监听的端口
    }

private:
    QTcpSocket *socket;
};
```

### 总结

**网络编程要点：**
- 客户端可以绑定网卡和端口
- 客户端端口由自己决定，不是服务器分配
- TCP和UDP可以同时使用相同端口（不同网卡或协议）
- 使用TCP保活机制检测连接状态
- 设置合理的超时时间

**注意事项：**
- Qt4中QTcpSocket的bind是protected的，需要继承
- UDP需要使用hasPendingDatagrams检查数据
- 端口随机分配一般按顺序递增
- 注意Qt5.15新增的超时设置函数

**最佳实践：**
- 客户端根据需求绑定特定端口
- 使用保活机制防止连接僵死
- 设置合理的超时时间
- 处理网络错误和异常情况
