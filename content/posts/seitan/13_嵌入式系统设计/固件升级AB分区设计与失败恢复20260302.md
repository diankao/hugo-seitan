# 固件升级AB分区设计与失败恢复

## 题目

固件升级如何设计 AB 分区？升级失败如何保证可恢复？

## 考察点

OTA 升级设计、系统可靠性

## 回答要点

### 1. AB 分区概念

**设计原理**：
- 两个系统分区：A 分区和 B 分区
- 启动计数器（Boot Counter）
- 启动标记（Boot Flag）
- 安全的切换机制

**分区布局**：
```
┌─────────────────────────────────┐
│  Bootloader                    │
├─────────────────────────────────┤
│  配置分区                      │
│  - 当前启动分区                 │
│  - 启动计数器                 │
├─────────────────────────────────┤
│  A 分区（系统 A）              │
│  - 内核                        │
│  - 根文件系统                  │
│  - 应用程序                    │
├─────────────────────────────────┤
│  B 分区（系统 B）              │
│  - 内核                        │
│  - 根文件系统                  │
│  - 应用程序                    │
├─────────────────────────────────┤
│  数据分区                      │
│  - 用户数据                    │
│  - 配置文件                    │
└─────────────────────────────────┘
```

### 2. 升级流程

#### 完整流程
```
当前状态：
- A 分区：正常系统
- B 分区：旧版本或空

升级流程：
1. 检查升级条件（电量、空间）
2. 下载升级包
3. 校验升级包完整性
4. 将升级包写入 B 分区
5. 设置启动标记为 B 分区
6. 重启设备
7. Bootloader 读取标记，启动 B 分区
8. 启动成功，验证 B 分区
9. 更新启动标记，永久切换到 B 分区
10. B 分区成为当前系统
```

#### 配置结构
```c
// 启动配置
typedef struct {
    uint32_t magic;              // 魔数：0x4F544142 (OTAB)
    uint32_t version;            // 配置版本
    uint32_t active_partition;   // 当前启动分区：0=A, 1=B
    uint32_t boot_counter_a;     // A 分区启动计数
    uint32_t boot_counter_b;     // B 分区启动计数
    uint32_t successful_boot_a;  // A 分区成功启动次数
    uint32_t successful_boot_b;  // B 分区成功启动次数
    uint8_t  partition_a_valid; // A 分区有效标志
    uint8_t  partition_b_valid; // B 分区有效标志
    uint8_t  reserved[64];     // 保留字段
    uint32_t crc;               // CRC 校验
} boot_config_t;

// 分区信息
typedef struct {
    uint32_t version_major;
    uint32_t version_minor;
    uint32_t version_patch;
    uint32_t build_number;
    char build_date[32];
    uint32_t size;
    uint32_t checksum;
    uint8_t  reserved[32];
} partition_info_t;
```

```c
// 检查电量
bool check_battery() {
```

#### 步骤 2：下载升级包
```c
// 下载升级包
int download_update(const char* url, const char* output_path) {
    // 1. 创建临时文件
    char temp_path[256];
    sprintf(temp_path, "%s.tmp", output_path);
    FILE* fp = fopen(temp_path, "wb");

    // 2. HTTP 下载
    http_download(url, fp, progress_callback);

    // 3. 完成下载
    fclose(fp);

    // 4. 重命名为正式文件名
    rename(temp_path, output_path);

    return 0;
}
```

#### 步骤 3：校验升级包
```c
// 校验完整性
bool verify_update(const char* path) {
    // 1. 计算 CRC/SHA256
    uint32_t calculated_crc = calc_file_crc(path);

    // 2. 读取头部校验值
    update_header_t header;
    read_update_header(path, &header);

    // 3. 比较校验值
    if (calculated_crc != header.checksum) {
        show_error("Update package corrupted");
        return false;
    }

    return true;
}

// 校验签名（如果有）
bool verify_signature(const char* path, const uint8_t* public_key) {
    // 1. 读取签名
    uint8_t signature[256];
    read_signature(path, signature);

    // 2. 验证签名
    return rsa_verify(path, signature, public_key);
}
```

#### 步骤 4：写入 B 分区
```c
// 写入升级包
int write_to_partition_b(const char* update_path) {
    // 1. 确定 B 分区地址
    uint32_t b_partition_addr = get_partition_addr(PARTITION_B);

    // 2. 擦除 B 分区
    flash_erase(b_partition_addr, b_partition_size);

    // 3. 写入新固件
    FILE* fp = fopen(update_path, "rb");
    uint8_t buffer[4096];
    size_t read_size;

    uint32_t offset = 0;
    while ((read_size = fread(buffer, 1, sizeof(buffer), fp)) > 0) {
        flash_write(b_partition_addr + offset, buffer, read_size);
        offset += read_size;

        // 显示进度
        show_progress(offset, b_partition_size);
    }

    fclose(fp);
    return 0;
}
```

#### 步骤 5：设置启动标记
```c
// 设置启动分区
void set_boot_partition(partition_t part) {
    boot_config_t config;

    // 1. 读取当前配置
    read_boot_config(&config);

    // 2. 更新启动标记
    config.active_partition = (part == PARTITION_A) ? 0 : 1;

    // 3. 重置启动计数器
    if (part == PARTITION_A) {
        config.boot_counter_a = 0;
    } else {
        config.boot_counter_b = 0;
    }

    // 4. 写回配置
    config.crc = calc_crc(&config, sizeof(config) - 4);
    write_boot_config(&config);

    // 5. 同步到存储
    fsync(boot_config_fd);
}
```

#### 步骤 6：重启
```c
// 重启设备
void reboot_for_update() {

    show_message("System will reboot to install update...");

    // 2. 保存状态
    save_update_state(STATE_UPDATING);

    // 3. 同步文件系统
    sync();

    // 4. 重启
    reboot();
}
```

### 4. Bootloader 逻辑

#### 启动选择
```c
// Bootloader 主逻辑
void bootloader_main() {
    // 1. 读取启动配置
    boot_config_t config;
    read_boot_config(&config);

    // 2. 确定启动分区
    partition_t boot_part;
    if (config.active_partition == 0) {
        boot_part = PARTITION_A;
        config.boot_counter_a++;
    } else {
        boot_part = PARTITION_B;
        config.boot_counter_b++;
    }

    // 3. 检查启动计数器
    uint32_t boot_count = (boot_part == PARTITION_A) ?
                        config.boot_counter_a : config.boot_counter_b;

    // 4. 启动计数器超过阈值，回退
    if (boot_count > MAX_BOOT_ATTEMPTS) {
        printf("Boot count exceeded, rollback\n");

        // 回退到另一个分区
        config.active_partition = (boot_part == PARTITION_A) ? 1 : 0;
        config.boot_counter_a = 0;
        config.boot_counter_b = 0;

        write_boot_config(&config);
        boot_part = (config.active_partition == 0) ?
                    PARTITION_A : PARTITION_B;
    }

    // 5. 验证分区有效性
    if (!verify_partition(boot_part)) {
        printf("Partition invalid, try other partition\n");
        config.active_partition = (boot_part == PARTITION_A) ? 1 : 0;
        write_boot_config(&config);
        boot_part = (config.active_partition == 0) ?
                    PARTITION_A : PARTITION_B;
    }

    // 6. 启动系统
    boot_system(boot_part);
}
```

### 5. 失败恢复机制

#### 启动失败检测
```c
// 应用层启动成功通知
void notify_boot_success() {
    // 1. 读取启动配置
    boot_config_t config;
    read_boot_config(&config);

    // 2. 增加成功启动计数
    if (config.active_partition == 0) {
        config.successful_boot_a++;
    } else {
        config.successful_boot_b++;
    }

    // 3. 重置启动计数器
    config.boot_counter_a = 0;
    config.boot_counter_b = 0;

    // 4. 写回配置
    config.crc = calc_crc(&config, sizeof(config) - 4);
    write_boot_config(&config);
}
```

#### 自动回滚
```
触发条件：
1. 连续启动失败超过阈值（如 3 次）
2. 启动超时
3. 系统崩溃

回滚流程：
1. Bootloader 检测到失败
2. 切换到原分区
3. 重置启动计数器
4. 正常启动旧系统
5. 提示用户升级失败
```

```c
// 下载时报告进度
void download_with_progress(const char* url, progress_callback_t cb) {
    int total_size = get_file_size(url);
    int downloaded = 0;

    while (downloaded < total_size) {
        int chunk = download_chunk(url, downloaded);
        downloaded += chunk;

        // 报告进度

        cb(percent);
    }
}
```

### 7. 安全考虑

#### 数字签名验证
```c
// 升级包签名
typedef struct {
    uint32_t magic;
    uint32_t version;
    uint32_t size;
    uint32_t checksum;
    uint8_t  signature[256];  // RSA 签名
    uint8_t  data[0];
} signed_update_t;

// 验证签名
bool verify_signed_update(const char* path) {
    signed_update_t update;
    read_update(path, &update);

    // 使用公钥验证
    return rsa_verify(update.data, update.size,
                   update.signature, public_key);
}
```

### 8. 异常处理

#### 升级中断
```c
// 升级中断恢复
void resume_interrupted_update() {
    // 1. 检查临时文件
    if (file_exists(UPDATE_TEMP_PATH)) {
        // 2. 验证临时文件
        if (verify_update(UPDATE_TEMP_PATH)) {
            // 3. 提示用户恢复
            if (ask_user("Resume interrupted update?")) {
                // 继续升级
                resume_update(UPDATE_TEMP_PATH);
            } else {
                // 删除临时文件
                remove(UPDATE_TEMP_PATH);
            }
        }
    }
}
```

#### 存储空间不足
```c
// 清理空间
void cleanup_for_update(uint64_t required_space) {
    // 1. 检查可用空间
    uint64_t free_space = get_free_space();

    // 2. 空间不足时清理
    while (free_space < required_space * 2) {
        // 删除旧日志
        remove_old_logs();

        // 删除缓存
        clear_cache();

        // 重新检查
        free_space = get_free_space();
    }
}
```
