# 锁设计与死锁风险降低策略

## 题目

如果系统中必须使用锁，如何设计来降低死锁风险？

## 考察点

并发编程、锁机制、死锁预防、工程实践经验

## 回答要点

### 1. 死锁的产生条件（四个必要条件）

| 条件 | 含义 | 破坏方式 |
|------|------|---------|
| 互斥 | 资源同一时刻只能被一个任务持有 | 某些场景可用无锁数据结构替代 |
| 占有并等待 | 持有锁的同时等待另一个锁 | 一次获取所有锁 |
| 不可抢占 | 已获取的锁不能被强制释放 | 设置超时（trylock） |
| 循环等待 | 多个任务形成环形等待链 | 规定加锁顺序 |

**只要破坏其中任何一个条件，死锁就不会发生。**

### 2. 死锁的典型场景

#### 场景一：AB-BA 死锁

```c
// 任务 1
void task1(void) {
    xSemaphoreTake(mutex_A, WAIT_FOREVER);
    // ...
    xSemaphoreTake(mutex_B, WAIT_FOREVER);   // 等待 B
    // ...
    xSemaphoreGive(mutex_B);
    xSemaphoreGive(mutex_A);
}

// 任务 2
void task2(void) {
    xSemaphoreTake(mutex_B, WAIT_FOREVER);
    // ...
    xSemaphoreTake(mutex_A, WAIT_FOREVER);   // 等待 A → 死锁！
    // ...
    xSemaphoreGive(mutex_A);
    xSemaphoreGive(mutex_B);
}

// 任务 1 持有 A 等 B，任务 2 持有 B 等 A → 循环等待
```

#### 场景二：自死锁（同任务重复加锁）

```c
// 使用二值信号量作为互斥锁时
SemaphoreHandle_t bin_sem = xSemaphoreCreateBinary();

void task_a(void) {
    xSemaphoreTake(bin_sem, WAIT_FOREVER);
    func_that_also_takes_lock();    // 再次获取同一把锁 → 死锁！
    xSemaphoreGive(bin_sem);
}
```

#### 场景三：优先级反转导致的隐性死锁

```
低优先级任务 L 持有 Mutex
高优先级任务 H 等待 Mutex
中优先级任务 M 持续运行，L 得不到调度 → H 永远等下去
```

### 3. 降低死锁风险的设计策略

#### 策略一：规定全局加锁顺序

```c
// 为所有锁定义优先级编号，必须按编号从小到大获取
// lock_order.h
#define LOCK_ORDER_SENSOR   1
#define LOCK_ORDER_BUFFER   2
#define LOCK_ORDER_UART     3
#define LOCK_ORDER_FLASH    4
#define LOCK_ORDER_NETWORK  5

// 正确：先低后高
void safe_function(void) {
    xSemaphoreTake(mutex_sensor, WAIT_FOREVER);   // order 1
    xSemaphoreTake(mutex_buffer, WAIT_FOREVER);   // order 2
    xSemaphoreTake(mutex_uart, WAIT_FOREVER);     // order 3
    // ...
    xSemaphoreGive(mutex_uart);
    xSemaphoreGive(mutex_buffer);
    xSemaphoreGive(mutex_sensor);
}

// 编译时/代码审查时检查：不得出现逆序获取
```

**实现辅助**：
```c
typedef struct {
    SemaphoreHandle_t handle;
    uint8_t order;
} ordered_mutex_t;

 BaseType_t ordered_take(ordered_mutex_t *mutex, uint8_t *highest_held) {
    if (mutex->order <= *highest_held) {
        log_error("lock order violation: %d after %d",
                  mutex->order, *highest_held);
        return pdFAIL;
    }
    BaseType_t result = xSemaphoreTake(mutex->handle, WAIT_FOREVER);
    if (result == pdPASS) {
        *highest_held = mutex->order;
    }
    return result;
}
```

#### 策略二：使用 trylock 替代无限等待

```c
// 非阻塞获取锁，失败时释放已持有的锁并重试
void safe_multi_lock(void) {
    int retry = 0;
    while (retry < MAX_RETRY) {
        if (xSemaphoreTake(mutex_A, MAX_WAIT) != pdPASS) {
            retry++;
            continue;
        }
        if (xSemaphoreTake(mutex_B, MAX_WAIT) != pdPASS) {
            xSemaphoreGive(mutex_A);     // 释放 A，避免占着等
            retry++;
            continue;
        }
        break;
    }
}
```

#### 策略三：减少锁的粒度和持有时间

```c
// 错误：锁粒度过大
xSemaphoreTake(mutex, WAIT_FOREVER);
read_sensor();        // I/O 操作，慢
process_data();       // 计算操作，慢
format_output();      // 格式化，慢
xSemaphoreGive(mutex);

// 正确：只锁保护共享数据的部分
sensor_data_t local_copy;
xSemaphoreTake(mutex, WAIT_FOREVER);
local_copy = shared_sensor_data;    // 只复制数据
xSemaphoreGive(mutex);

process_data(&local_copy);          // 不持锁处理
format_output(&local_copy);         // 不持锁格式化
```

#### 策略四：避免嵌套加锁

```c
// 设计原则：一个函数要么获取锁，要么在已持锁状态下被调用
// 不要在函数内部既获取新锁又依赖调用者已持有的锁

// 改造前：嵌套加锁
void update_status(void) {
    xSemaphoreTake(mutex_status, WAIT_FOREVER);
    status.count++;
    log_event();     // 内部又获取 mutex_log
    xSemaphoreGive(mutex_status);
}

// 改造后：分离关注点
void update_status_unlocked(void) {
    status.count++;
}

void update_status(void) {
    xSemaphoreTake(mutex_status, WAIT_FOREVER);
    update_status_unlocked();
    xSemaphoreGive(mutex_status);
    log_event();     // 在锁外调用，log 自己管理自己的锁
}
```

#### 策略五：使用递归互斥锁防止自死锁

```c
// 当同任务可能重复获取同一把锁时，使用递归互斥锁
SemaphoreHandle_t rec_mutex = xSemaphoreCreateRecursiveMutex();

void task_func(void) {
    xSemaphoreTakeRecursive(rec_mutex, WAIT_FOREVER);

    nested_func();    // 内部也获取同一把锁

    xSemaphoreGiveRecursive(rec_mutex);
}

void nested_func(void) {
    xSemaphoreTakeRecursive(rec_mutex, WAIT_FOREVER);   // 同一任务可以重复获取
    // ...
    xSemaphoreGiveRecursive(rec_mutex);                 // 获取几次就释放几次
}
```

### 4. 死锁检测与恢复

#### 运行时检测

```c
// 维护锁的持有矩阵，检测环路
typedef struct {
    TaskHandle_t holder;
    TickType_t acquire_time;
    uint8_t order;
} lock_info_t;

lock_info_t lock_table[MAX_LOCKS];

// 每次获取锁时记录，释放时清除
// 定期扫描 lock_table，检测是否存在环形等待
// 方法：构建等待图，执行拓扑排序，若有环则存在死锁
```

#### 看门狗检测

```c
// 为持锁操作设置超时
void monitored_lock(SemaphoreHandle_t mutex, uint32_t timeout_ms) {
    TickType_t t0 = xTaskGetTickCount();
    if (xSemaphoreTake(mutex, pdMS_TO_TICKS(timeout_ms)) != pdPASS) {
        log_error("potential deadlock: task=%s, lock=%p, wait=%dms",
                  pcTaskGetName(NULL), mutex, timeout_ms);
    }
}
```

### 5. 替代方案：减少锁的使用

| 方案 | 适用场景 | 优点 | 缺点 |
|------|---------|------|------|
| 无锁环形缓冲区 | 单生产者单消费者 | 无死锁风险 | 仅限 SPSC |
| 原子操作 | 简单计数器/标志 | 无锁、低延迟 | 只适用于简单操作 |
| 消息队列 | 任务间通信 | 天然解耦 | 有拷贝开销 |
| 关中断 | 单核极短临界区 | 简单可靠 | 影响中断响应 |
| RCU（Read-Copy-Update） | 读多写少 | 读无锁 | 实现复杂 |

```c
// 无锁 SPSC 环形缓冲区示例
typedef struct {
    volatile uint32_t head;
    volatile uint32_t tail;
    item_t buf[BUF_SIZE];
} spsc_ring_t;

// 生产者（只写 head）
bool produce(spsc_ring_t *ring, item_t *item) {
    uint32_t next = (ring->head + 1) % BUF_SIZE;
    if (next == ring->tail) return false;
    ring->buf[ring->head] = *item;
    __dmb(0xF);
    ring->head = next;
    return true;
}

// 消费者（只写 tail）
bool consume(spsc_ring_t *ring, item_t *item) {
    if (ring->head == ring->tail) return false;
    *item = ring->buf[ring->tail];
    __dmb(0xF);
    ring->tail = (ring->tail + 1) % BUF_SIZE;
    return true;
}
```
