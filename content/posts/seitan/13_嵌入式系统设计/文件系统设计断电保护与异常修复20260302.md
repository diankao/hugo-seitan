### 2. 日志文件系统（Journaling）

#### 工作原理
```
写操作流程：
1. 记录操作到日志（Journal）
2. 更新元数据
3. 更新数据
4. 标记操作完成

断电恢复：
1. 扫描日志
2. 重放未完成的操作
3. 恢复一致性
```

#### 实现示例
```c
// 日志记录
typedef struct {
    uint64_t timestamp;
    uint32_t op_type;
    uint64_t file_offset;
    uint32_t data_len;
    uint8_t  data[0];
} journal_entry_t;

// 写入前记录日志
void safe_write(int fd, void* data, int len, uint64_t offset) {
    // 1. 写入日志
    journal_entry_t entry = {
        .timestamp = get_time(),
        .op_type = OP_WRITE,
        .file_offset = offset,
        .data_len = len
    };
    memcpy(entry.data, data, len);
    write_journal(&entry);

    // 2. 执行写操作
    pwrite(fd, data, len, offset);

    // 3. 标记完成
    mark_journal_complete(&entry);
}

// 恢复时检查日志
void recover_from_journal() {
    journal_entry_t entry;
    while (read_journal(&entry)) {
        if (!is_complete(&entry)) {
            // 重放未完成的操作
            pwrite(fd, entry.data, entry.data_len, entry.file_offset);
        }
    }
}
```

### 3. 原子写操作

#### 临时文件 + 原子重命名
```c
// 原子写入
void atomic_write(const char* filename, void* data, int len) {
    char temp_name[256];
    sprintf(temp_name, "%s.tmp", filename);

    // 1. 写入临时文件
    int fd = open(temp_name, O_WRONLY | O_CREAT | O_TRUNC, 0644);
    write(fd, data, len);
    fsync(fd);  // 确保写入磁盘
    close(fd);

    // 2. 原子重命名
    rename(temp_name, filename);
    fsync(fileno(fopen(filename, "r")));  // 同步目录
}
```

#### 优势
- 重命名是原子操作
- 要么成功要么失败
- 不会出现部分写入

### 5. 数据校验机制

#### CRC 校验
```c
typedef struct {
    uint32_t magic;      // 魔数，用于识别
    uint32_t data_len;
    uint32_t crc32;
    uint8_t  data[0];
} data_chunk_t;

// 写入时计算 CRC
uint32_t calc_crc32(void* data, int len) {
    uint32_t crc = 0xFFFFFFFF;
    for (int i = 0; i < len; i++) {
        crc = crc_table[(crc ^ ((uint8_t*)data)[i]) & 0xFF] ^ (crc >> 8);
    }
    return ~crc;
}

// 写入带校验的数据
void write_with_checksum(int fd, void* data, int len) {
    data_chunk_t chunk = {
        .magic = 0x4D414449,  // "MAGIC"
        .data_len = len,
        .crc32 = calc_crc32(data, len)
    };
    memcpy(chunk.data, data, len);
    write(fd, &chunk, sizeof(chunk) + len);
}

// 读取时验证
int read_with_checksum(int fd, void* data, int max_len) {
    data_chunk_t chunk;
    read(fd, &chunk, sizeof(chunk));

    uint32_t crc = calc_crc32(chunk.data, chunk.data_len);
    if (crc != chunk.crc32) {
        return -1;  // 校验失败
    }

    memcpy(data, chunk.data, chunk.data_len);
    return chunk.data_len;
}
```

### 6. 异常文件修复

#### 损坏检测
```c
// 扫描文件
void scan_and_repair() {
    file_meta_t meta;
    for (int i = 0; i < file_count; i++) {
        read_metadata(i, &meta);

        if (!meta.is_valid) {
            // 标记为无效的文件
            mark_invalid(i);
            continue;
        }

        // 验证文件完整性
        if (!verify_file_integrity(&meta)) {
            // 尝试修复
            repair_file(&meta);
        }
    }
}

// 验证文件
bool verify_file_integrity(file_meta_t* meta) {
    char filename[128];
    sprintf(filename, "video/%04d.mp4", meta->file_id);

    // 1. 检查文件大小
    struct stat st;
    stat(filename, &st);
    if (st.st_size != meta->file_size) {
        return false;
    }

    // 2. 检查 CRC
    uint32_t crc = calc_file_crc(filename);
    if (crc != meta->checksum) {
        return false;
    }

    return true;
}
```

#### 修复策略
```c
// 修复文件
void repair_file(file_meta_t* meta) {
    char filename[128];
    sprintf(filename, "video/%04d.mp4", meta->file_id);

    // 1. 尝试恢复元数据
    recover_metadata(filename, meta);

    // 2. 尝试修复数据
    recover_data(filename, meta);

    // 3. 验证修复结果
    if (verify_file_integrity(meta)) {
        meta->is_valid = 1;
        update_metadata(meta);
    } else {
        // 无法修复，删除文件
        remove(filename);
        meta->is_valid = 0;
        update_metadata(meta);
    }
}

// 恢复元数据
void recover_metadata(const char* filename, file_meta_t* meta) {
    // 分析文件头部
    FILE* fp = fopen(filename, "rb");
    uint8_t header[1024];
    fread(header, 1, sizeof(header), fp);

    // 解析 MP4 盒子
    parse_mp4_boxes(header, sizeof(header), meta);

    fclose(fp);
}

// 恢复数据
void recover_data(const char* filename, file_meta_t* meta) {
    // 1. 找到最后一个完整的帧
    int last_valid_frame = find_last_valid_frame(filename);

    // 2. 截断文件
    truncate(filename, last_valid_frame);

    // 3. 更新元数据
    meta->file_size = last_valid_frame;
    update_metadata(meta);
}
```

### 7. 性能优化

#### 缓冲管理
```c
// 双缓冲
typedef struct {
    uint8_t* buffer[2];
    int current;
    int size;
} double_buffer_t;

// 写入时切换缓冲
void write_frame(double_buffer_t* db, uint8_t* frame) {
    memcpy(db->buffer[db->current], frame, db->size);
    db->current = (db->current + 1) % 2;

    // 异步刷盘
    async_flush(db->buffer[!db->current], db->size);
}
```
