# 线程安全高性能环形缓冲区设计

## 题目

如果需要实现一个线程安全且高性能的环形缓冲区，如何设计？

## 考察点

并发数据结构、无锁编程、内存屏障、工程实践

## 回答要点

### 1. 环形缓冲区的基本原理

```
初始状态（空）：
head = 0
tail = 0
┌───┬───┬───┬───┬───┬───┬───┬───┐
│   │   │   │   │   │   │   │   │
└───┴───┴───┴───┴───┴───┴───┴───┘
  ▲
 head/tail

写入 3 个元素后：
head = 3
tail = 0
┌───┬───┬───┬───┬───┬───┬───┬───┐
│ A │ B │ C │   │   │   │   │   │
└───┴───┴───┴───┴───┴───┴───┴───┘
  ▲           ▲
 tail        head

读取 1 个，再写入 6 个后（环形）：
head = 1
tail = 1
┌───┬───┬───┬───┬───┬───┬───┬───┐
│ H │ B │ C │ D │ E │ F │ G │   │
└───┴───┴───┴───┴───┴───┴───┴───┘
      ▲                       ▲
     tail                    head
```

### 2. 线程安全的挑战

**核心问题**：生产者写 head，消费者写 tail，两者并发操作同一数据结构。

| 场景 | 需要保护的操作 |
|------|--------------|
| 单生产者单消费者（SPSC） | head 和 tail 各只被一方修改 |
| 多生产者单消费者（MPSC） | 多个生产者竞争写 head |
| 多生产者多消费者（MPMC） | 读写两端都有竞争 |

### 3. SPSC 无锁环形缓冲区（最高性能）

#### 3.1 基本实现

```c
#define RING_SIZE  256
#define RING_MASK  (RING_SIZE - 1)

typedef struct {
    item_t buf[RING_SIZE];
    volatile uint32_t head;
    volatile uint32_t tail;
} spsc_ring_t;

bool produce(spsc_ring_t *r, item_t item) {
    uint32_t next = (r->head + 1) & RING_MASK;
    if (next == r->tail) return false;

    r->buf[r->head] = item;

    __dmb(0xF);

    r->head = next;
    return true;
}

bool consume(spsc_ring_t *r, item_t *item) {
    if (r->head == r->tail) return false;

    *item = r->buf[r->tail];

    __dmb(0xF);

    r->tail = (r->tail + 1) & RING_MASK;
    return true;
}
```

#### 3.2 关键设计点

**容量必须为 2 的幂**：
```c
// 用位运算代替取模
// & RING_MASK 比 % RING_SIZE 快一个数量级
uint32_t next = (r->head + 1) & RING_MASK;
```

**内存屏障的位置**：
```c
// 生产者：先写数据，再更新 head
r->buf[r->head] = item;     // 1. 写数据
__dmb(0xF);                  // 2. 确保 1 对消费者可见
r->head = next;              // 3. 消费者看到 head 更新时，数据一定已经就绪

// 消费者：先读数据，再更新 tail
*item = r->buf[r->tail];    // 1. 读数据
__dmb(0xF);                  // 2. 确保数据读完
r->tail = next;              // 3. 生产者看到 tail 更新时，槽位可以复用
```

**volatile 的作用**：
```c
// volatile 防止编译器将 head/tail 缓存到寄存器
// 每次访问都从内存读取最新值
volatile uint32_t head;
volatile uint32_t tail;
```

#### 3.3 批量操作（提升吞吐）

```c
uint32_t produce_batch(spsc_ring_t *r, item_t *items, uint32_t count) {
    uint32_t available = (r->tail - r->head - 1) & RING_MASK;
    uint32_t to_write = (count < available) ? count : available;

    for (uint32_t i = 0; i < to_write; i++) {
        r->buf[(r->head + i) & RING_MASK] = items[i];
    }

    __dmb(0xF);
    r->head = (r->head + to_write) & RING_MASK;
    return to_write;
}

uint32_t consume_batch(spsc_ring_t *r, item_t *items, uint32_t max_count) {
    uint32_t available = (r->head - r->tail) & RING_MASK;
    uint32_t to_read = (max_count < available) ? max_count : available;

    for (uint32_t i = 0; i < to_read; i++) {
        items[i] = r->buf[(r->tail + i) & RING_MASK];
    }

    __dmb(0xF);
    r->tail = (r->tail + to_read) & RING_MASK;
    return to_read;
}
```

### 4. ISR + Task 场景（嵌入式最常见）

```c
// ISR 中写入，任务中读取
// 不需要内存屏障（单核上中断天然有序）
// 但需要 volatile

typedef struct {
    volatile uint32_t head;
    volatile uint32_t tail;
    uint8_t buf[RING_SIZE];
} isr_ring_t;

// ISR 中调用（不使用 FreeRTOS API）
void uart_rx_isr(void) {
    uint8_t data = USART1->DR;
    uint32_t next = (ring.head + 1) & RING_MASK;
    if (next != ring.tail) {
        ring.buf[ring.head] = data;
        ring.head = next;
    }
}

// 任务中读取
void uart_process_task(void *arg) {
    while (1) {
        while (ring.head != ring.tail) {
            uint8_t data = ring.buf[ring.tail];
            ring.tail = (ring.tail + 1) & RING_MASK;
            process(data);
        }
        xSemaphoreTake(uart_sem, portMAX_DELAY);
    }
}
```

### 5. MPSC 无锁环形缓冲区

```c
// 多个生产者竞争 head
// 使用 CAS（Compare-And-Swap）原子操作

bool mpsc_produce(mpsc_ring_t *r, item_t item) {
    uint32_t head, next;

    do {
        head = r->head;
        next = (head + 1) & RING_MASK;
        if (next == r->tail) return false;
    } while (!__atomic_compare_exchange_n(&r->head, &head, next,
                                           0, __ATOMIC_ACQ_REL, __ATOMIC_ACQUIRE));

    r->buf[head] = item;
    __atomic_thread_fence(__ATOMIC_RELEASE);
    return true;
}

// CAS 保证只有一个生产者能成功更新 head
// 失败的生产者重试
```

### 6. 基于信号量的环形缓冲区（通用但较慢）

```c
typedef struct {
    item_t buf[RING_SIZE];
    uint32_t head;
    uint32_t tail;
    SemaphoreHandle_t sem_filled;
    SemaphoreHandle_t sem_empty;
    SemaphoreHandle_t mutex;
} sem_ring_t;

void sem_ring_init(sem_ring_t *r) {
    r->head = 0;
    r->tail = 0;
    r->sem_filled = xSemaphoreCreateCounting(RING_SIZE, 0);
    r->sem_empty = xSemaphoreCreateCounting(RING_SIZE, RING_SIZE);
    r->mutex = xSemaphoreCreateMutex();
}

bool sem_produce(sem_ring_t *r, item_t item, uint32_t timeout_ms) {
    if (xSemaphoreTake(r->sem_empty, pdMS_TO_TICKS(timeout_ms)) != pdPASS)
        return false;

    xSemaphoreTake(r->mutex, portMAX_DELAY);
    r->buf[r->head] = item;
    r->head = (r->head + 1) & RING_MASK;
    xSemaphoreGive(r->mutex);

    xSemaphoreGive(r->sem_filled);
    return true;
}

bool sem_consume(sem_ring_t *r, item_t *item, uint32_t timeout_ms) {
    if (xSemaphoreTake(r->sem_filled, pdMS_TO_TICKS(timeout_ms)) != pdPASS)
        return false;

    xSemaphoreTake(r->mutex, portMAX_DELAY);
    *item = r->buf[r->tail];
    r->tail = (r->tail + 1) & RING_MASK;
    xSemaphoreGive(r->mutex);

    xSemaphoreGive(r->sem_empty);
    return true;
}
```

### 7. 方案对比

| 方案 | 线程安全 | 性能 | 适用场景 |
|------|---------|------|---------|
| SPSC 无锁 | 单生产者单消费者 | 最高 | ISR→Task、流水线 |
| MPSC CAS | 多生产者单消费者 | 高 | 多传感器→单处理 |
| 信号量保护 | 多生产者多消费者 | 中 | 通用场景 |
| 互斥锁保护 | 多生产者多消费者 | 低 | 简单场景 |

```
选择流程：

SPSC？──Yes──▶ 无锁 SPSC（最快）
  │
  No
  │
  ▼
MPSC？──Yes──▶ CAS 无锁 MPSC
  │
  No
  │
  ▼
需要阻塞等待？──Yes──▶ 信号量方案
  │
  No
  │
  ▼
简单互斥锁即可
```

### 8. 性能优化要点

- **容量为 2 的幂**：用 `& mask` 代替 `% size`
- **缓存行对齐**：head 和 tail 分开到不同缓存行，避免 false sharing
- **预取提示**：对即将访问的缓冲区位置发 `__pld` 预取指令
- **批量操作**：减少内存屏障的执行次数
- **避免动态分配**：缓冲区静态分配，大小编译时确定

```c
// 缓存行对齐（Cortex-A 通常 64 字节缓存行）
typedef struct {
    volatile uint32_t head;
    char _pad1[60];              // 填充到独立缓存行
    volatile uint32_t tail;
    char _pad2[60];
    item_t buf[RING_SIZE];
} __attribute__((aligned(64))) spsc_ring_t;
```
