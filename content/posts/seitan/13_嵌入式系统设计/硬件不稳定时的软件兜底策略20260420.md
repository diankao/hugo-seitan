# 硬件不稳定时的软件兜底策略

## 题目

在驱动开发中，如果硬件行为不稳定，如何通过软件做"兜底"？

## 考察点

驱动开发、容错设计、工程经验

## 回答要点

### 1. "硬件不稳定"的常见表现

| 表现 | 可能的硬件原因 |
|------|--------------|
| 偶尔读到错误数据 | 信号完整性差、时序裕量不够 |
| 操作无响应 | 总线挂死、芯片复位 |
| 间歇性通信失败 | 接触不良、EMI 干扰 |
| 数据偶尔错位 | 时钟抖动、信号边沿缓 |
| 上电偶尔初始化失败 | 电源上电时序不满足 |

### 2. 通信类兜底策略

#### 2.1 重试机制

```c
#define MAX_RETRY  3
#define RETRY_DELAY_MS  1

int i2c_read_safe(uint8_t dev_addr, uint8_t reg, uint8_t *data) {
    int retry = 0;
    while (retry < MAX_RETRY) {
        if (HAL_I2C_Mem_Read(&hi2c1, dev_addr, reg,
                             I2C_MEMADD_SIZE_8BIT,
                             data, 1, 100) == HAL_OK) {
            return 0;
        }
        retry++;
        HAL_Delay(RETRY_DELAY_MS);

        // 总线恢复：如果 I2C 总线挂死
        i2c_bus_recovery();
    }
    log_error("I2C read failed after %d retries: dev=0x%02X reg=0x%02X",
              retry, dev_addr, reg);
    return -1;
}
```

#### 2.2 I2C 总线恢复

```c
// I2C 总线挂死：Slave 拉住 SDA 不释放
// 恢复方法：手动 toggle SCL 直到 SDA 释放

void i2c_bus_recovery(void) {
    GPIO_InitTypeDef gpio = {
        .Pin = I2C_SCL_PIN,
        .Mode = GPIO_MODE_OUTPUT_OD,
        .Speed = GPIO_SPEED_FREQ_HIGH,
    };
    HAL_GPIO_Init(I2C_SCL_PORT, &gpio);

    // 发 9 个 SCL 脉冲
    for (int i = 0; i < 9; i++) {
        HAL_GPIO_WritePin(I2C_SCL_PORT, I2C_SCL_PIN, GPIO_PIN_RESET);
        delay_us(5);
        HAL_GPIO_WritePin(I2C_SCL_PORT, I2C_SCL_PIN, GPIO_PIN_SET);
        delay_us(5);

        // 检查 SDA 是否已释放
        if (HAL_GPIO_ReadPin(I2C_SDA_PORT, I2C_SDA_PIN) == GPIO_PIN_SET) {
            break;
        }
    }

    // 产生 STOP 条件
    HAL_GPIO_WritePin(I2C_SDA_PORT, I2C_SDA_PIN, GPIO_PIN_RESET);
    delay_us(5);
    HAL_GPIO_WritePin(I2C_SCL_PORT, I2C_SCL_PIN, GPIO_PIN_SET);
    delay_us(5);
    HAL_GPIO_WritePin(I2C_SDA_PORT, I2C_SDA_PIN, GPIO_PIN_SET);

    // 重新初始化 I2C 外设
    MX_I2C1_Init();
}
```

#### 2.3 数据校验与合理性检查

```c
// 传感器读取 + 合理性验证
bool read_sensor_safe(sensor_data_t *out) {
    sensor_data_t raw;

    // 读取原始数据
    if (sensor_read(&raw) != 0) return false;

    // 校验1：范围检查
    if (raw.temperature < -40 || raw.temperature > 85) {
        log_warning("sensor out of range: %d", raw.temperature);
        return false;
    }

    // 校验2：变化率检查（防突变）
    static int32_t last_temp = 25;
    if (abs(raw.temperature - last_temp) > 20) {
        log_warning("sensor rate-of-change too large: %d -> %d",
                    last_temp, raw.temperature);
        return false;
    }

    // 校验3：多次读取一致性
    sensor_data_t verify;
    if (sensor_read(&verify) != 0) return false;
    if (abs(raw.temperature - verify.temperature) > 2) {
        log_warning("sensor readings inconsistent");
        return false;
    }

    last_temp = raw.temperature;
    *out = raw;
    return true;
}
```

### 3. 外设类兜底策略

#### 3.1 外设复位

```c
// 如果外设有复位引脚，软件控制复位
void sensor_reset(void) {
    HAL_GPIO_WritePin(SENSOR_RST_PORT, SENSOR_RST_PIN, GPIO_PIN_RESET);
    HAL_Delay(1);
    HAL_GPIO_WritePin(SENSOR_RST_PORT, SENSOR_RST_PIN, GPIO_PIN_SET);
    HAL_Delay(10);   // 等待芯片重新初始化
}

// 如果没有复位引脚，通过总线发软复位命令
void sensor_soft_reset(void) {
    uint8_t cmd = SENSOR_CMD_SOFT_RESET;
    i2c_write_safe(SENSOR_ADDR, &cmd, 1);
    HAL_Delay(50);
}
```

#### 3.2 看门狗监控外设

```c
// 定期检查外设是否响应
typedef struct {
    uint8_t  i2c_addr;
    uint8_t  who_am_i_reg;
    uint8_t  expected_id;
    uint32_t check_interval_ms;
    uint32_t last_ok_time;
    uint32_t fail_count;
    void (*reset_fn)(void);
} peripheral_watchdog_t;

void peripheral_monitor_task(void *arg) {
    peripheral_watchdog_t *devs = (peripheral_watchdog_t *)arg;

    while (1) {
        for (int i = 0; i < DEV_COUNT; i++) {
            uint8_t id;
            if (i2c_read_safe(devs[i].i2c_addr,
                              devs[i].who_am_i_reg, &id) == 0 &&
                id == devs[i].expected_id) {
                devs[i].last_ok_time = xTaskGetTickCount();
                devs[i].fail_count = 0;
            } else {
                devs[i].fail_count++;
                if (devs[i].fail_count >= 3) {
                    log_error("device 0x%02X not responding, resetting",
                              devs[i].i2c_addr);
                    if (devs[i].reset_fn) {
                        devs[i].reset_fn();
                    }
                    devs[i].fail_count = 0;
                }
            }
        }
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
```

### 4. 存储类兜底策略

#### 4.1 Flash 写入验证

```c
// 写入后回读验证
int flash_write_verify(uint32_t addr, const uint8_t *data, uint32_t len) {
    // 写入
    flash_unlock();
    flash_write(addr, data, len);
    flash_lock();

    // 回读校验
    for (uint32_t i = 0; i < len; i++) {
        uint8_t read_val = *(volatile uint8_t *)(addr + i);
        if (read_val != data[i]) {
            log_error("flash verify failed at 0x%08X: wrote 0x%02X read 0x%02X",
                      addr + i, data[i], read_val);
            return -1;
        }
    }
    return 0;
}
```

### 5. 启动类兜底策略

#### 5.1 上电检测与延迟等待

```c
// 某些外设上电需要稳定时间
// 如果立即访问可能失败

void board_init(void) {
    // 给外设上电
    power_on_external_devices();

    // 等待电源稳定
    HAL_Delay(100);

    // 等待外设就绪（轮询状态寄存器）
    int retry = 0;
    while (retry < 100) {
        uint8_t status;
        if (sensor_read_status(&status) == 0 && (status & STATUS_READY)) {
            break;
        }
        HAL_Delay(10);
        retry++;
    }
}
```

#### 5.2 外设初始化降级

```c
// 如果某个外设初始化失败，不阻塞系统启动
// 降级运行，标记该功能不可用

typedef struct {
    bool sensor_ok;
    bool display_ok;
    bool network_ok;
    bool storage_ok;
} system_health_t;

static system_health_t health = {0};

void system_init(void) {
    if (sensor_init() == 0) {
        health.sensor_ok = true;
    } else {
        log_warning("sensor init failed, running without sensor");
    }

    if (display_init() == 0) {
        health.display_ok = true;
    } else {
        log_warning("display init failed");
    }
}

// 应用代码检查
float get_temperature(void) {
    if (!health.sensor_ok) return DEFAULT_TEMPERATURE;
    // ...
}
```

### 6. 兜底策略总结

```
硬件不稳定时的软件兜底层级：

第一层：重试
  通信失败 → 重试 N 次
  最简单最常用

第二层：校验
  数据读取 → 范围检查 / CRC / 一致性检查
  过滤掉异常数据

第三层：恢复
  外设无响应 → 软复位 / 硬复位 / 总线恢复
  尝试恢复正常

第四层：降级
  恢复失败 → 标记不可用，使用默认值
  保证核心功能继续运行

第五层：上报
  记录故障日志
  上报给上位机 / 云端
  触发运维流程
```
