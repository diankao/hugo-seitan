# 相机过热处理与热管理策略分级

## 题目

相机过热如何处理？热管理策略如何分级？

## 考察点

热管理设计、系统可靠性

## 回答要点

### 1. 热源分析

#### 主要热源
```
SoC（ARM + DSP）：
- 高频运行产生大量热量
- 编码计算密集
- 功耗 2-5W

ISP（图像信号处理）：
- 实时图像处理
- 功耗 1-2W

Sensor（图像传感器）：
- 高帧率采集
- 功耗 0.5-1W

WiFi/BT 模块：
- 高功率发射
- 功耗 0.5-1W

显示屏：
- 高亮度背光
- 功耗 0.5-2W
```

### 2. 温度监测

#### 温度传感器布局
```c
// 温度传感器位置
typedef struct {
    float soc_temp;           // SoC 温度
    float sensor_temp;         // Sensor 温度
    float battery_temp;         // 电池温度
    float ambient_temp;         // 环境温度
    float wifi_temp;           // WiFi 模块温度
} temperature_t;

// 读取温度
temperature_t read_all_temperatures() {
    temperature_t temp;

    temp.soc_temp = read_sensor(SENSOR_SOC);
    temp.sensor_temp = read_sensor(SENSOR_SENSOR);
    temp.battery_temp = read_sensor(SENSOR_BATTERY);
    temp.ambient_temp = read_sensor(SENSOR_AMBIENT);
    temp.wifi_temp = read_sensor(SENSOR_WIFI);

    return temp;
}
```

#### 温度阈值
```c
// 温度级别定义
typedef enum {
    TEMP_NORMAL = 0,      // < 50°C
    TEMP_WARM = 1,         // 50-55°C
    TEMP_HOT = 2,           // 55-60°C
    TEMP_CRITICAL = 3,      // 60-65°C
    TEMP_EMERGENCY = 4      // > 65°C
} temp_level_t;

// 判断温度级别
temp_level_t get_temp_level(float temp) {
    if (temp < 50) {
        return TEMP_NORMAL;
    } else if (temp < 55) {
        return TEMP_WARM;
    } else if (temp < 60) {
        return TEMP_HOT;
    } else if (temp < 65) {
        return TEMP_CRITICAL;
    } else {
        return TEMP_EMERGENCY;
    }
}
```

### 3. 热管理策略分级

#### 一级热管理（50-55°C）
```c
// 温度预警
void handle_warm_temperature(temperature_t* temp) {
    // 1. 降低频率
    reduce_dvfs_frequency(PERFORMANCE_LEVEL_MEDIUM);

    // 2. 降低帧率
    set_frame_rate(24);  // 从 30fps 降到 24fps

    // 3. 提示用户
    show_warning("Device is warm, performance reduced");

    // 4. 降低屏幕亮度
    reduce_brightness(20);

    // 5. 关闭 WiFi（如果不需要）
    if (!is_wifi_active()) {
        turn_off_wifi();
    }
}
```

#### 二级热管理（55-60°C）
```c
// 热度处理
void handle_hot_temperature(temperature_t* temp) {
    // 1. 进一步降低频率
    reduce_dvfs_frequency(PERFORMANCE_LEVEL_LOW);

    // 2. 降低分辨率
    set_resolution(1080);  // 从 4K 降到 1080p

    // 3. 降低帧率
    set_frame_rate(15);

    // 4. 降低编码质量
    set_encoding_bitrate(LOW_BITRATE);

    // 5. 关闭不必要功能
    disable_advanced_features();

    // 6. 强制降低屏幕亮度
    reduce_brightness(50);

    // 7. 显示热警告
    show_hot_warning("Device is hot, recording may stop");
}
```

#### 三级热管理（60-65°C）
```c
// 关键热处理
void handle_critical_temperature(temperature_t* temp) {
    // 1. 最低频率
    reduce_dvfs_frequency(PERFORMANCE_LEVEL_MIN);

    // 2. 停止录制
    stop_recording();
    save_recording_state();

    // 3. 限制功能
    enter_thermal_mode();

    // 4. 降低屏幕亮度到最低
    reduce_brightness(100);

    // 5. 关闭所有无线
    turn_off_wifi();
    turn_off_bluetooth();

    // 6. 显示严重警告
    show_critical_warning("Critical temperature, system limited");
}
```

#### 四级热管理（>65°C）
```c
// 紧急热保护
void handle_emergency_temperature(temperature_t* temp) {
    // 1. 立即停止所有操作
    emergency_shutdown();

    // 2. 保存状态
    save_critical_state();

    // 3. 冷却模式
    enter_cooling_mode();

    // 4. 关闭所有外设
    shutdown_all_peripherals();

    // 5. 显示错误
    show_emergency_error("Temperature critical, device shutdown");
}
```

### 5. 动态频率调节

#### 基于温度的 DVFS
```c
// 温度与频率映射表
typedef struct {
    float temp_min;
    float temp_max;
    int freq_mhz;
} temp_freq_map_t;

const temp_freq_map_t freq_map[] = {
    { .temp_min = 0,    .temp_max = 50,  .freq_mhz = 1500 },
    { .temp_min = 50,   .temp_max = 55,  .freq_mhz = 1200 },
    { .temp_min = 55,   .temp_max = 60,  .freq_mhz = 800  },
    { .temp_min = 60,   .temp_max = 65,  .freq_mhz = 500  },
    { .temp_min = 65,   .temp_max = 100, .freq_mhz = 300  }
};

// 根据温度设置频率
void set_freq_by_temperature(float temp) {
    for (int i = 0; i < sizeof(freq_map) / sizeof(freq_map[0]); i++) {
        if (temp >= freq_map[i].temp_min && temp < freq_map[i].temp_max) {
            set_cpu_frequency(freq_map[i].freq_mhz);
            return;
        }
    }
}
```

```c
// 温度趋势预测
typedef struct {
    float temps[10];  // 10 秒温度历史
    int index;
} temp_history_t;

// 预测未来温度
float predict_temperature(temp_history_t* hist) {
    // 计算温度变化率
    float rate = (hist->temps[hist->index] -
                 hist->temps[(hist->index + 9) % 10]) / 10;

    // 预测 5 秒后温度
    return hist->temps[hist->index] + rate * 5;
}
```

### 6. 热管理状态机

#### 状态机设计
```c
// 热管理状态
typedef enum {
    THERMAL_STATE_NORMAL,
    THERMAL_STATE_WARM,
    THERMAL_STATE_HOT,
    THERMAL_STATE_CRITICAL,
    THERMAL_STATE_EMERGENCY,
    THERMAL_STATE_COOLING
} thermal_state_t;

// 状态机主循环
void thermal_control_loop() {
    thermal_state_t state = THERMAL_STATE_NORMAL;
    temperature_t temp;

    while (1) {
        temp = read_all_temperatures();
        temp_level_t level = get_temp_level(temp.soc_temp);

        switch (state) {
            case THERMAL_STATE_NORMAL:
                if (level == TEMP_WARM) {
                    state = THERMAL_STATE_WARM;
                    handle_warm_temperature(&temp);
                }
                break;

            case THERMAL_STATE_WARM:
                if (level == TEMP_HOT) {
                    state = THERMAL_STATE_HOT;
                    handle_hot_temperature(&temp);
                } else if (level == TEMP_NORMAL) {
                    state = THERMAL_STATE_NORMAL;
                    restore_normal_performance();
                }
                break;

            case THERMAL_STATE_HOT:
                if (level == TEMP_CRITICAL) {
                    state = THERMAL_STATE_CRITICAL;
                    handle_critical_temperature(&temp);
                } else if (level == TEMP_WARM) {
                    state = THERMAL_STATE_WARM;
                    restore_warm_performance();
                }
                break;

            case THERMAL_STATE_CRITICAL:
                if (level == TEMP_EMERGENCY) {
                    state = THERMAL_STATE_EMERGENCY;
                    handle_emergency_temperature(&temp);
                } else if (level == TEMP_HOT) {
                    state = THERMAL_STATE_HOT;
                    restore_hot_performance();
                }
                break;

            case THERMAL_STATE_EMERGENCY:
                if (level < TEMP_CRITICAL) {
                    state = THERMAL_STATE_COOLING;
                    start_cooling();
                }
                break;

            case THERMAL_STATE_COOLING:
                if (level == TEMP_NORMAL) {
                    state = THERMAL_STATE_NORMAL;
                    exit_cooling();
                    restore_normal_performance();
                }
                break;
        }

        sleep(1);  // 1 秒检查一次
    }
}
```

### 7. 散热优化

#### 预热策略
```c
// 开机预热
void warmup_phase() {
    // 1. 低功率预热
    set_low_power_mode();

    // 2. 逐渐提升功率
    for (int i = 0; i < 10; i++) {
        gradual_power_increase(i);
        sleep(1);
    }

    // 3. 达到稳定温度后正常工作
    enter_normal_mode();
}
```

#### 冷却策略
```c
// 智能冷却
void intelligent_cooling() {
    // 1. 检测散热能力
    float cooling_rate = measure_cooling_rate();

    // 2. 调整冷却策略
    if (cooling_rate < expected_rate) {
        // 增强散热带
        increase_fan_speed();
        enable_additional_vents();
    }

    // 3. 温度恢复后恢复正常
    while (get_temperature() > 50) {
        sleep(1);
    }

    restore_normal_mode();
}
```

### 8. 用户体验

#### 温度提示
```c
// 渐进式提示
void show_temperature_warning(float temp) {
    if (temp < 55) {
        // 正常，不提示
    } else if (temp < 60) {
        show_toast("Device is warm");
    } else if (temp < 65) {
        show_alert("Device is hot, performance reduced");
    } else {
        show_critical("Critical temperature, recording stopped");
    }
}
```

#### 录制保护
```c
// 录制温度保护
void recording_thermal_protection() {
    float temp = get_temperature();

    // 温度过高时停止录制
    if (temp > 65 && is_recording()) {
        // 1. 保存当前录制
        save_recording();

        // 2. 停止录制
        stop_recording();

        // 3. 提示用户
        show_message("Recording stopped due to overheating");

        // 4. 自动保存
        auto_save();
    }
}
```
