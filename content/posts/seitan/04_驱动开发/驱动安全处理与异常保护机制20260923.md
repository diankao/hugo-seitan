# 驱动安全处理与异常保护机制

## 题目

了解驱动的安全处理吗？

## 考察点

驱动安全、内核保护

## 回答要点

### 1. 驱动安全的重要性

#### 驱动在系统中的位置
```
用户空间
    ↓ 系统调用
内核空间
    ↓
驱动程序（Ring 0）
    ↓
硬件设备
```

#### 驱动安全风险
- 恶意代码访问内核
- 权限提升漏洞
- 拒绝服务攻击
- 数据泄露

### 2. 驱动安全机制

#### 权限控制
```c
// 检查用户权限
bool check_user_permission(int operation) {
    uid_t uid = current_uid();

    switch (operation) {
        case OP_READ:
            return check_read_permission(uid);
        case OP_WRITE:
            return check_write_permission(uid);
        case OP_IOCTL:
            return check_ioctl_permission(uid);
        default:
            return false;
    }
}

// 文件操作权限检查
static int device_open(struct inode *inode, struct file *file) {
    if (!check_user_permission(OP_READ)) {
        pr_err("Permission denied: user %d\n", current_uid());
        return -EPERM;
    }

    return 0;
}
```

#### 参数验证
```c
// 严格的参数验证
static int device_ioctl(struct file *file, unsigned int cmd, unsigned long arg) {
    // 1. 验证命令是否合法
    if (_IOC_TYPE(cmd) != DRIVER_MAGIC) {
        pr_err("Invalid command type\n");
        return -ENOTTY;
    }

    // 2. 验证参数大小
    if (_IOC_SIZE(cmd) > MAX_IOCTL_SIZE) {
        pr_err("Parameter size too large\n");
        return -EINVAL;
    }

    // 3. 验证用户空间指针
    if (!access_ok(arg, _IOC_SIZE(cmd))) {
        pr_err("Invalid user pointer\n");
        return -EFAULT;
    }

    // 4. 执行具体操作
    switch (cmd) {
        case DRIVER_IOCTL_SET_CONFIG:
            return handle_set_config(arg);
        case DRIVER_IOCTL_GET_STATUS:
            return handle_get_status(arg);
        default:
            return -ENOTTY;
    }
}
```

### 3. 内存安全

#### 防止缓冲区溢出
```c
// 安全的内存拷贝
static int device_read(struct file *file, char __user *buf,
                      size_t count, loff_t *ppos) {
    // 1. 验证用户缓冲区
    if (!access_ok(buf, count)) {
        pr_err("Invalid user buffer\n");
        return -EFAULT;
    }

    // 2. 检查数据大小
    if (count > MAX_DATA_SIZE) {
        count = MAX_DATA_SIZE;
    }

    // 3. 安全拷贝
    if (copy_to_user(buf, driver_data, count)) {
        pr_err("Failed to copy to user\n");
        return -EFAULT;
    }

    return count;
}

// 安全的内存分配
void* secure_alloc(size_t size) {
    // 1. 检查大小
    if (size > MAX_ALLOC_SIZE) {
        return NULL;
    }

    // 2. 使用 GFP_KERNEL 分配
    void* ptr = kmalloc(size, GFP_KERNEL);
    if (!ptr) {
        return NULL;
    }

    // 3. 清零内存
    memset(ptr, 0, size);

    return ptr;
}
```

#### 防止空指针解引用
```c
// 空指针检查
static int device_write(struct file *file, const char __user *buf,
                       size_t count, loff_t *ppos) {
    // 1. 检查设备上下文
    if (!device_ctx) {
        pr_err("Device context is NULL\n");
        return -ENODEV;
    }

    // 2. 检查设备状态
    if (!device_ctx->is_ready) {
        pr_err("Device not ready\n");
        return -EBUSY;
    }

    // 3. 验证缓冲区
    if (!buf || count == 0) {
        pr_err("Invalid buffer\n");
        return -EINVAL;
    }

    // 执行写入操作
    return do_device_write(device_ctx, buf, count);
}
```

### 4. 异常处理

#### 异常捕获
```c
// 使用 try-catch 机制（伪代码）
static int risky_operation(void) {
    int result;

    // 检查前置条件
    if (!check_preconditions()) {
        return -EINVAL;
    }

    // 执行操作
    result = perform_operation();

    // 检查结果
    if (result < 0) {
        pr_err("Operation failed: %d\n", result);
        handle_error(result);
        return result;
    }

    return 0;
}

// 错误恢复
static void handle_error(int error) {
    switch (error) {
        case ERR_TIMEOUT:
            // 重试
            retry_operation();
            break;
        case ERR_HW_FAULT:
            // 硬件故障，需要重置
            reset_hardware();
            break;
        case ERR_INVALID_PARAM:
            // 参数错误，记录日志
            log_invalid_param();
            break;
        default:
            // 未知错误
            report_error(error);
    }
}
```

#### 资源清理
```c
// 确保资源释放
static int complex_operation(void) {
    void* buffer = NULL;
    struct resource* res = NULL;
    int ret = 0;

    // 分配资源 1
    buffer = kmalloc(BUFFER_SIZE, GFP_KERNEL);
    if (!buffer) {
        ret = -ENOMEM;
        goto err_out;
    }

    // 分配资源 2
    res = acquire_resource();
    if (!res) {
        ret = -EBUSY;
        goto err_free_buffer;
    }

    // 执行操作
    ret = do_operation(buffer, res);
    if (ret < 0) {
        goto err_release_res;
    }

    // 成功
    return 0;

// 错误处理路径
err_release_res:
    release_resource(res);
err_free_buffer:
    kfree(buffer);
err_out:
    return ret;
}
```

### 5. 并发安全

#### 自旋锁保护
```c
// 使用自旋锁保护共享资源
static DEFINE_SPINLOCK(driver_lock);

static int device_read(struct file *file, char __user *buf,
                      size_t count, loff_t *ppos) {
    unsigned long flags;
    int ret;

    // 获取锁
    spin_lock_irqsave(&driver_lock, flags);

    // 访问共享资源
    ret = do_read(buf, count);

    // 释放锁
    spin_unlock_irqrestore(&driver_lock, flags);

    return ret;
}

// 尝试获取锁（避免死锁）
static int device_write(struct file *file, const char __user *buf,
                       size_t count, loff_t *ppos) {
    unsigned long flags;
    int ret;

    // 尝试获取锁
    if (!spin_trylock_irqsave(&driver_lock, flags)) {
        pr_warn("Failed to acquire lock\n");
        return -EBUSY;
    }

    ret = do_write(buf, count);

    spin_unlock_irqrestore(&driver_lock, flags);
    return ret;
}
```

#### 互斥锁保护
```c
// 使用互斥锁保护长临界区
static DEFINE_MUTEX(driver_mutex);

static long device_ioctl(struct file *file, unsigned int cmd,
                        unsigned long arg) {
    int ret;

    // 获取互斥锁
    if (mutex_lock_interruptible(&driver_mutex)) {
        return -ERESTARTSYS;
    }

    // 执行操作
    ret = handle_ioctl(cmd, arg);

    // 释放锁
    mutex_unlock(&driver_mutex);

    return ret;
}
```

### 6. 防御性编程

#### 断言检查
```c
// 调试断言
#ifdef DEBUG
#define DRIVER_ASSERT(cond) \
    do { \
        if (!(cond)) { \
            pr_crit("Assertion failed: %s\n", #cond); \
            BUG(); \
        } \
    } while (0)
#else
#define DRIVER_ASSERT(cond) do {} while (0)
#endif

// 使用断言
static void driver_function(void) {
    DRIVER_ASSERT(device_ctx != NULL);
    DRIVER_ASSERT(device_ctx->state >= 0);

    do_something();
}
```

#### 边界检查
```c
// 数组边界检查
static int driver_array_access(int index) {
    // 检查边界
    if (index < 0 || index >= MAX_ARRAY_SIZE) {
        pr_err("Array index out of bounds: %d\n", index);
        return -EINVAL;
    }

    return driver_array[index];
}

// 循环边界检查
static void driver_loop(void) {
    int i;

    // 限制循环次数
    for (i = 0; i < MAX_ITERATIONS; i++) {
        if (i >= MAX_SAFE_ITERATIONS) {
            pr_warn("Loop approaching safety limit\n");
            break;
        }

        do_iteration(i);
    }
}
```

### 8. 安全最佳实践

#### 最小权限原则
```c
// 只授予必要的权限
static int set_permissions(struct file *file) {
    // 1. 默认只读权限
    file->f_mode &= ~FMODE_WRITE;

    // 2. 根据用户授予写权限
    if (is_admin_user(current_uid())) {
        file->f_mode |= FMODE_WRITE;
    }

    // 3. 始终禁止执行权限
    file->f_mode &= ~FMODE_EXEC;

    return 0;
}
```

#### 深度防御
```c
// 多层防御
static int secure_operation(void) {
    // 第一层：参数验证
    if (!validate_parameters()) {
        return -EINVAL;
    }

    // 第二层：权限检查
    if (!check_permission()) {
        return -EPERM;
    }

    // 第三层：资源检查
    if (!check_resources()) {
        return -EBUSY;
    }

    // 第四层：执行操作
    return do_operation();
}
```
