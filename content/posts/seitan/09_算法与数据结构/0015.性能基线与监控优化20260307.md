#### 2. 基线测试

**基准测试框架**：
```c
typedef struct {
    const char *test_name;
    uint32_t (*test_func)(void);
    uint32_t iterations;
    uint32_t baseline_time_us;
    uint32_t current_time_us;
    float deviation_percent;
} Benchmark_t;

#define BENCHMARK_COUNT 10

Benchmark_t benchmarks[BENCHMARK_COUNT] = {
    {"CPU Performance", Test_CPU_Performance, 1000, 0, 0, 0},
    {"Memory Access", Test_Memory_Access, 10000, 0, 0, 0},
    {"Task Switch", Test_Task_Switch, 1000, 0, 0, 0},
    {"Interrupt Response", Test_Interrupt_Response, 1000, 0, 0, 0},
    {"Context Switch", Test_Context_Switch, 1000, 0, 0, 0},
    {"Memory Allocation", Test_Memory_Allocation, 1000, 0, 0, 0},
    {"String Operations", Test_String_Operations, 1000, 0, 0, 0},
    {"Math Operations", Test_Math_Operations, 1000, 0, 0, 0},
    {"IO Operations", Test_IO_Operations, 100, 0, 0, 0},
    {"Network Operations", Test_Network_Operations, 100, 0, 0, 0}
};

uint32_t Run_Benchmark(Benchmark_t *benchmark) {
    uint32_t start_time = GetMicroseconds();
    
    for (uint32_t i = 0; i < benchmark->iterations; i++) {
        benchmark->test_func();
    }
    
    uint32_t end_time = GetMicroseconds();
    benchmark->current_time_us = end_time - start_time;
    
    if (benchmark->baseline_time_us > 0) {
        benchmark->deviation_percent = 
            (float)(benchmark->current_time_us - benchmark->baseline_time_us) / 
            benchmark->baseline_time_us * 100;
    }
    
    return benchmark->current_time_us;
}

void Establish_Baseline(void) {
    printf("Establishing performance baseline...\n");
    
    for (int i = 0; i < BENCHMARK_COUNT; i++) {
        printf("Running benchmark: %s\n", benchmarks[i].test_name);
        
        uint32_t total_time = 0;
        uint32_t samples = 10;
        
        运行多次取平均值
        for (int j = 0; j < samples; j++) {
            total_time += Run_Benchmark(&benchmarks[i]);
            vTaskDelay(100);
        }
        
        benchmarks[i].baseline_time_us = total_time / samples;
        benchmarks[i].current_time_us = benchmarks[i].baseline_time_us;
        benchmarks[i].deviation_percent = 0;
        
        printf("  Baseline: %d us (%.2f us/op)\n", 
               benchmarks[i].baseline_time_us,
               (float)benchmarks[i].baseline_time_us / benchmarks[i].iterations);
    }
    
    printf("Baseline established successfully\n");
}
```

#### 3. 性能对比

```c
void Compare_Performance(void) {
    printf("Comparing current performance with baseline...\n");
    
    for (int i = 0; i < BENCHMARK_COUNT; i++) {
        printf("%s:\n", benchmarks[i].test_name);
        printf("  Baseline: %d us\n", benchmarks[i].baseline_time_us);
        printf("  Current: %d us\n", benchmarks[i].current_time_us);
        printf("  Deviation: %.2f%%\n", benchmarks[i].deviation_percent);
        
        if (abs(benchmarks[i].deviation_percent) > 10) {
            printf("  WARNING: Performance deviation > 10%%\n");
        }
    }
}

bool Performance_Is_Acceptable(void) {
    for (int i = 0; i < BENCHMARK_COUNT; i++) {
        if (abs(benchmarks[i].deviation_percent) > 20) {
            return false;
        }
    }
    return true;
}
```

### 持续性能监控

#### 1. 系统监控

**CPU使用率监控**：
```c
typedef struct {
    uint32_t total_ticks;
    uint32_t idle_ticks;
    float cpu_usage;
} CPU_Monitor_t;

CPU_Monitor_t cpu_monitor;

void CPU_Monitor_Init(void) {
    cpu_monitor.total_ticks = 0;
    cpu_monitor.idle_ticks = 0;
    cpu_monitor.cpu_usage = 0;
}

void CPU_Monitor_Update(void) {
    static uint32_t last_total_ticks = 0;
    static uint32_t last_idle_ticks = 0;
    
    uint32_t current_total_ticks = GetSystemTicks();
    uint32_t current_idle_ticks = GetIdleTaskTicks();
    
    uint32_t total_delta = current_total_ticks - last_total_ticks;
    uint32_t idle_delta = current_idle_ticks - last_idle_ticks;
    
    if (total_delta > 0) {
        cpu_monitor.cpu_usage = 
            (float)(total_delta - idle_delta) / total_delta * 100;
    }
    
    cpu_monitor.total_ticks = current_total_ticks;
    cpu_monitor.idle_ticks = current_idle_ticks;
    
    last_total_ticks = current_total_ticks;
    last_idle_ticks = current_idle_ticks;
}

float CPU_Get_Usage(void) {
    return cpu_monitor.cpu_usage;
}
```

**内存监控**：
```c
typedef struct {
    uint32_t total_heap;
    uint32_t free_heap;
    uint32_t min_free_heap;
    uint32_t allocated_blocks;
    uint32_t max_allocated;
    uint32_t fragmentation_percent;
} Memory_Monitor_t;

Memory_Monitor_t memory_monitor;

void Memory_Monitor_Update(void) {
    memory_monitor.total_heap = configTOTAL_HEAP_SIZE;
    memory_monitor.free_heap = xPortGetFreeHeapSize();
    memory_monitor.min_free_heap = xPortGetMinimumEverFreeHeapSize();
    memory_monitor.allocated_blocks = Get_Allocated_Blocks_Count();
    memory_monitor.max_allocated = memory_monitor.total_heap - memory_monitor.min_free_heap;
    
    计算内存碎片率
    if (memory_monitor.allocated_blocks > 0) {
        uint32_t avg_block_size = 
            (memory_monitor.total_heap - memory_monitor.free_heap) / 
            memory_monitor.allocated_blocks;
        
        memory_monitor.fragmentation_percent = 
            (float)(memory_monitor.free_heap - avg_block_size) / 
            memory_monitor.free_heap * 100;
    }
}

void Memory_Monitor_Print(void) {
    printf("Memory Monitor:\n");
    printf("  Total: %d bytes\n", memory_monitor.total_heap);
    printf("  Free: %d bytes (%.1f%%)\n", 
           memory_monitor.free_heap,
           (float)memory_monitor.free_heap / memory_monitor.total_heap * 100);
    printf("  Min Free: %d bytes\n", memory_monitor.min_free_heap);
    printf("  Allocated Blocks: %d\n", memory_monitor.allocated_blocks);
    printf("  Max Allocated: %d bytes\n", memory_monitor.max_allocated);
    printf("  Fragmentation: %.1f%%\n", memory_monitor.fragmentation_percent);
}
```

```c
typedef struct {
    char name[16];

    uint32_t switch_count;
    uint32_t priority;

Task_Monitor_t task_monitors[MAX_TASKS];

void Task_Monitor_Update(void) {
    UBaseType_t task_count = uxTaskGetNumberOfTasks();
    
    TaskStatus_t *task_status = pvPortMalloc(task_count * sizeof(TaskStatus_t));
    if (task_status == NULL) {
        return;
    }
    
    uxTaskGetSystemState(task_status, task_count, NULL);
    
    for (UBaseType_t i = 0; i < task_count; i++) {
        strncpy(task_monitors[i].name, task_status[i].pcTaskName, 15);
        task_monitors[i].name[15] = '\0';
        task_monitors[i].stack_size = task_status[i].usStackHighWaterMark;
        task_monitors[i].stack_high_water_mark = task_status[i].usStackHighWaterMark;
        task_monitors[i].run_time = task_status[i].ulRunTimeCounter;
        task_monitors[i].switch_count = 0;
        task_monitors[i].priority = task_status[i].uxCurrentPriority;
        task_monitors[i].state = task_status[i].eCurrentState;
    }
    
    vPortFree(task_status);
}

void Task_Monitor_Print(void) {
    UBaseType_t task_count = uxTaskGetNumberOfTasks();
    
    printf("Task Monitor:\n");
    printf("  %-16s %8s %8s %10s %6s %8s\n",

    printf("  %-16s %8s %8s %10s %6s %8s\n", 
           "----", "-----", "----------", "-------", "----", "-----");
    
    for (UBaseType_t i = 0; i < task_count; i++) {
        printf("  %-16s %8d %8d %10d %6d %8d\n",
               task_monitors[i].name,

               task_monitors[i].stack_high_water_mark,
               task_monitors[i].run_time,
               task_monitors[i].priority,

    }
}
```

### 内存带宽优化

#### 1. 缓存优化

```c
// 不好的做法：随机内存访问
void Bad_Memory_Access(uint32_t *array, uint32_t size) {
    for (uint32_t i = 0; i < size; i++) {
        uint32_t index = Get_Random_Index(size);
        array[index] = i;  // 随机访问，缓存命中率低
    }
}

// 好的做法：顺序内存访问
void Good_Memory_Access(uint32_t *array, uint32_t size) {
    for (uint32_t i = 0; i < size; i++) {
        array[i] = i;  // 顺序访问，缓存命中率高
    }
}
```

#### 2. 数据结构优化

```c
// 不好的做法：链表遍历
typedef struct Node {
    int data;
    struct Node *next;
} Node_t;

int Sum_Linked_List(Node_t *head) {
    int sum = 0;
    Node_t *current = head;
    
    while (current != NULL) {
        sum += current->data;
        current = current->next;  // 不连续内存访问
    }
    
    return sum;
}

// 好的做法：数组遍历
int Sum_Array(int *array, int size) {
    int sum = 0;
    
    for (int i = 0; i < size; i++) {
        sum += array[i];  // 连续内存访问
    }
    
    return sum;
}
```

#### 3. DMA使用

```c
// 不好的做法：CPU复制数据
void Bad_Data_Copy(uint8_t *src, uint8_t *dst, uint32_t size) {
    for (uint32_t i = 0; i < size; i++) {
        dst[i] = src[i];  // CPU参与，效率低
    }
}

// 好的做法：DMA复制数据
void Good_Data_Copy(uint8_t *src, uint8_t *dst, uint32_t size) {
    HAL_DMA_Start(&hdma_memtomem, (uint32_t)src, (uint32_t)dst, size);
    
    while (__HAL_DMA_GET_FLAG(&hdma_memtomem, DMA_FLAG_TC) == RESET) {
        等待DMA完成
    }
    
    __HAL_DMA_CLEAR_FLAG(&hdma_memtomem, DMA_FLAG_TC);
}
```

#### 4. 内存对齐

```c
// 不好的做法：未对齐访问
void Unaligned_Access(void) {
    uint8_t buffer[32];
    uint32_t *unaligned_ptr = (uint32_t*)&buffer[1];  // 未对齐
    
    *unaligned_ptr = 0x12345678;  // 效率低，某些架构不支持

// 好的做法：对齐访问
void Aligned_Access(void) {
    uint32_t buffer[8] __attribute__((aligned(16)));  // 16字节对齐

}
```

```c
// 不好的做法：长时间临界区
void Bad_Critical_Section(void) {
    taskENTER_CRITICAL();
    
    Complex_Processing();  // 长时间处理
    More_Complex_Processing();  // 更多处理

// 好的做法：短时间临界区
void Good_Critical_Section(void) {

    taskEXIT_CRITICAL();

    Complex_Processing();
```

#### 2. 优先级优化

```c
// 不好的做法：高优先级任务长时间运行
void HighPriority_Task(void *param) {
    while(1) {
        Long_Computation();  // 长时间计算，阻塞其他任务
        vTaskDelay(10);
    }
}

// 好的做法：高优先级任务短时间运行
void HighPriority_Task(void *param) {
    while(1) {
        Quick_Response();  // 快速响应
        xSemaphoreGive(xLowPriorityTaskSemaphore);  // 通知低优先级任务
        vTaskDelay(10);
    }
}

void LowPriority_Task(void *param) {
    while(1) {
        xSemaphoreTake(xLowPriorityTaskSemaphore, portMAX_DELAY);
        Long_Computation();  // 低优先级任务执行长时间计算
    }
}
```

```c
// 不好的做法：中断中做复杂处理
void UART_IRQHandler(void) {
    uint8_t data = UART_ReceiveByte();
    
    Process_Protocol(data);  // 复杂处理
    Update_State_Machine(data);  // 状态机更新

// 好的做法：中断只做必要处理
void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    uint8_t data = UART_ReceiveByte();
    
    xQueueSendFromISR(xQueue, &data, &xHigherPriorityTaskWoken);
    UART_ClearFlag();
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    uint8_t data;
    while(1) {
        if (xQueueReceive(xQueue, &data, portMAX_DELAY) == pdTRUE) {
            Process_Protocol(data);
            Update_State_Machine(data);
            Check_Checksum(data);
            Log_Data(data);
        }
    }
}
```

### 性能分析工具

#### 1. 性能统计

```c
typedef struct {
    const char *name;
    uint32_t call_count;
    uint32_t total_time_us;
    uint32_t min_time_us;
    uint32_t max_time_us;
    uint32_t avg_time_us;
} Performance_Stat_t;

#define MAX_STATS 32
Performance_Stat_t perf_stats[MAX_STATS];
volatile uint8_t stat_count = 0;

Performance_Stat_t *Get_Or_Create_Stat(const char *name) {
    for (int i = 0; i < stat_count; i++) {
        if (strcmp(perf_stats[i].name, name) == 0) {
            return &perf_stats[i];
        }
    }
    
    if (stat_count < MAX_STATS) {
        perf_stats[stat_count].name = name;
        perf_stats[stat_count].call_count = 0;
        perf_stats[stat_count].total_time_us = 0;
        perf_stats[stat_count].min_time_us = UINT32_MAX;
        perf_stats[stat_count].max_time_us = 0;
        perf_stats[stat_count].avg_time_us = 0;
        
        return &perf_stats[stat_count++];
    }
    
    return NULL;
}

void Record_Performance(const char *name, uint32_t time_us) {
    Performance_Stat_t *stat = Get_Or_Create_Stat(name);
    
    if (stat != NULL) {
        stat->call_count++;
        stat->total_time_us += time_us;
        
        if (time_us < stat->min_time_us) {
            stat->min_time_us = time_us;
        }
        
        if (time_us > stat->max_time_us) {
            stat->max_time_us = time_us;
        }
        
        stat->avg_time_us = stat->total_time_us / stat->call_count;
    }
}

#define PERF_SCOPE(name) \
    PerfScope __scope__(name);

class PerfScope {
public:
    PerfScope(const char *name) : name_(name) {
        start_time_ = GetMicroseconds();
    }
    
    ~PerfScope() {
        uint32_t end_time = GetMicroseconds();
        uint32_t elapsed = end_time - start_time_;
        Record_Performance(name_, elapsed);
    }
    
private:
    const char *name_;
    uint32_t start_time_;
};

void Some_Function(void) {
    PERF_SCOPE("Some_Function");
    
    Do_Something();
    Do_Something_Else();
}

void Print_Performance_Stats(void) {
    printf("Performance Statistics:\n");
    printf("  %-20s %8s %10s %10s %10s %10s\n", 
           "Name", "Calls", "Total(us)", "Min(us)", "Max(us)", "Avg(us)");
    printf("  %-20s %8s %10s %10s %10s %10s\n", 
           "----", "-----", "---------", "-------", "-------", "-------");
    
    for (int i = 0; i < stat_count; i++) {
        printf("  %-20s %8d %10d %10d %10d %10d\n",
               perf_stats[i].name,
               perf_stats[i].call_count,
               perf_stats[i].total_time_us,
               perf_stats[i].min_time_us,
               perf_stats[i].max_time_us,
               perf_stats[i].avg_time_us);
    }
}
```

### 注意事项

1. **基准测试**：确保测试环境稳定和可重复
2. **监控开销**：监控本身会影响性能，需要权衡
3. **优化权衡**：优化代码可读性和可维护性
4. **持续改进**：建立持续性能监控和改进机制
5. **数据驱动**：基于数据做出优化决策，而不是猜测
