# MCU低功耗模式

## 标签
#嵌入式 #面试题 #低功耗 #STM32 #电源管理

## 问题描述
MCU常见低功耗模式有哪些？如何实现深度睡眠与唤醒？

## 参考答案

### 低功耗模式概述

#### 1. 低功耗模式分类

**STM32低功耗模式**：
- **睡眠模式**：仅CPU停止，外设继续运行
- **停止模式**：CPU和大多数外设停止，SRAM数据保留
- **待机模式**：几乎整个芯片停止，仅保留备份寄存器
- **停机模式**：类似停止模式但功耗更低

**功耗对比**：
```
运行模式：     100%  (10-100mA)
睡眠模式：     10-50% (1-5mA)
停止模式：     1-5%   (10-100μA)
待机模式：     <1%    (1-10μA)
```

### 睡眠模式

#### 1. 睡眠模式特点

**特性**：
- CPU时钟停止
- 外设继续运行
- SRAM和寄存器内容保留
- 唤醒时间短（<10μs）

**适用场景**：
- 等待外部事件
- 定时器中断周期
- DMA传输等待

#### 2. 睡眠模式实现

```c
// 进入睡眠模式
void Enter_SleepMode(void) {
    __WFI();  // Wait For Interrupt
    // 或
    __WFE();  // Wait For Event
}

// 配置睡眠模式
void Configure_SleepMode(void) {
    设置SLEEPONEXIT位，从中断返回后进入睡眠
    SCB->SCR |= SCB_SCR_SLEEPONEXIT_Msk;
}

// 睡眠模式示例
void SleepMode_Example(void) {
    while(1) {
        执行任务
        ProcessTask();
        
        进入睡眠模式
        Enter_SleepMode();
    }
}

void EXTI_IRQHandler(void) {
    清除中断标志
    __HAL_GPIO_EXTI_CLEAR_IT(GPIO_PIN_0);
    
    处理中断事件
    ProcessEvent();
}
```

#### 3. 睡眠模式优化

```c
// 关闭未使用的外设时钟
void Disable_Unused_Peripherals(void) {
    __HAL_RCC_GPIOA_CLK_DISABLE();
    __HAL_RCC_GPIOB_CLK_DISABLE();
    __HAL_RCC_USART1_CLK_DISABLE();
}

// 降低CPU时钟频率
void Reduce_CPU_Clock(void) {
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};
    
    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | 
                                  RCC_CLOCKTYPE_SYSCLK |
                                  RCC_CLOCKTYPE_PCLK1 |
                                  RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV8;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    
    HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0);
}
```

### 停止模式

#### 1. 停止模式特点

**特性**：
- CPU和所有外设时钟停止
- SRAM和寄存器内容保留
- 可通过外部中断或特定事件唤醒
- 唤醒时间中等（~10-100μs）

**适用场景**：
- 长时间休眠
- 低功耗数据记录
- 电池供电设备

#### 2. 停止模式实现

```c
// 进入停止模式
void Enter_StopMode(void) {
    配置唤醒源
    HAL_PWR_EnableWakeUpPin(PWR_WAKEUP_PIN1);
    
    进入停止模式
    HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
    
    唤醒后恢复时钟
    SystemClock_Config();
}

// 停止模式示例
void StopMode_Example(void) {
    uint8_t data[100];
    
    while(1) {
        采集数据
        CollectData(data, 100);
        
        保存数据到Flash
        Flash_Write(data, 100);
        
        进入停止模式
        Enter_StopMode();
    }
}

void RTC_IRQHandler(void) {
    if (__HAL_RTC_WAKEUPTIMER_GET_FLAG(&hrtc, RTC_FLAG_WUTF)) {
        __HAL_RTC_WAKEUPTIMER_CLEAR_FLAG(&hrtc, RTC_FLAG_WUTF);
        
        唤醒后执行任务
        WakeupTask();
    }
}
```

#### 3. 停止模式优化

```c
// 配置RTC作为唤醒源
void Configure_RTC_Wakeup(void) {
    RTC_TimeTypeDef sTime = {0};
    RTC_DateTypeDef sDate = {0};
    
    HAL_RTC_SetTime(&hrtc, &sTime, RTC_FORMAT_BIN);
    HAL_RTC_SetDate(&hrtc, &sDate, RTC_FORMAT_BIN);
    
    配置唤醒定时器
    HAL_RTCEx_SetWakeUpTimer_IT(&hrtc, 10, RTC_WAKEUPCLOCK_CK_SPRE_16BITS);
    
    使能RTC唤醒中断
    HAL_NVIC_SetPriority(RTC_WKUP_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(RTC_WKUP_IRQn);
}

// 关闭未使用的电压调节器
void Disable_Regulator(void) {
    HAL_PWR_DisableBkUpAccess();
    HAL_PWR_EnableBkUpAccess();
    
    选择低功耗调节器
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE3);
}
```

### 待机模式

#### 1. 待机模式特点

**特性**：
- 整个芯片断电（除备份域）
- SRAM和寄存器内容丢失
- 仅备份寄存器和RTC保留
- 可通过特定引脚或RTC唤醒
- 唤醒时间长（~1-10ms）

**适用场景**：
- 超低功耗应用
- 长时间休眠
- 按键唤醒设备

#### 2. 待机模式实现

```c
// 进入待机模式
void Enter_StandbyMode(void) {
    使能 wakeup 引脚
    HAL_PWR_EnableWakeUpPin(PWR_WAKEUP_PIN1);
    
    清除唤醒标志
    __HAL_PWR_CLEAR_FLAG(PWR_FLAG_WU);
    
    进入待机模式
    HAL_PWR_EnterSTANDBYMode();
}

// 待机模式示例
void StandbyMode_Example(void) {
    配置唤醒引脚
    Configure_WakeupPin();
    
    设置RTC闹钟
    Set_RTC_Alarm();
    
    进入待机模式
    Enter_StandbyMode();
}

void HAL_RTCEx_WakeUpTimerEventCallback(RTC_HandleTypeDef *hrtc) {
    RTC唤醒，但不会执行，因为待机模式会复位
}
```

#### 3. 待机模式优化

```c
// 使用备份寄存器保存状态
void Save_State_To_Backup(void) {
    HAL_PWR_EnableBkUpAccess();
    
    保存状态到备份寄存器
    RTC->BKP0R = system_state.counter;
    RTC->BKP1R = system_state.mode;
    RTC->BKP2R = system_state.flags;
    
    HAL_PWR_DisableBkUpAccess();
}

void Restore_State_From_Backup(void) {
    HAL_PWR_EnableBkUpAccess();
    
    从备份寄存器恢复状态
    system_state.counter = RTC->BKP0R;
    system_state.mode = RTC->BKP1R;
    system_state.flags = RTC->BKP2R;
    
    HAL_PWR_DisableBkUpAccess();
}

void main(void) {
    HAL_Init();
    
    检查复位原因
    if (__HAL_PWR_GET_FLAG(PWR_FLAG_SB) != RESET) {
        __HAL_PWR_CLEAR_FLAG(PWR_FLAG_SB);
        
        从待机模式唤醒
        Restore_State_From_Backup();
        Resume_From_Standby();
    } else {
        正常启动
        Normal_Startup();
    }
}
```

### 深度睡眠与唤醒

#### 1. 深度睡眠配置

```c
typedef enum {
    SLEEP_MODE_LIGHT,
    SLEEP_MODE_DEEP,
    SLEEP_MODE_STOP,
    SLEEP_MODE_STANDBY
} SleepMode_t;

typedef struct {
    SleepMode_t mode;
    uint32_t sleep_duration_ms;
    uint8_t wakeup_source;
} SleepConfig_t;

// 配置深度睡眠
void Configure_DeepSleep(SleepConfig_t *config) {
    switch (config->mode) {
        case SLEEP_MODE_LIGHT:
            配置睡眠模式
            SCB->SCR &= ~SCB_SCR_SLEEPDEEP_Msk;
            break;
            
        case SLEEP_MODE_DEEP:
            配置深度睡眠模式
            SCB->SCR |= SCB_SCR_SLEEPDEEP_Msk;
            break;
            
        case SLEEP_MODE_STOP:
            配置停止模式
            HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
            break;
            
        case SLEEP_MODE_STANDBY:
            配置待机模式
            HAL_PWR_EnterSTANDBYMode();
            break;
    }
}
```

#### 2. 唤醒源配置

```c
void Configure_Exti_Wakeup(void) {
```

```c
// 唤醒处理函数
void Wakeup_Handler(void) {
    uint32_t reset_source = RCC->CSR;
    
    if (reset_source & RCC_CSR_WWDGRSTF) {
        printf("WWDG reset\n");
    }
    
    if (reset_source & RCC_CSR_IWDGRSTF) {
        printf("IWDG reset\n");
    }
    
    if (reset_source & RCC_CSR_SFTRSTF) {
        printf("Software reset\n");
    }
    
    if (reset_source & RCC_CSR_PORRSTF) {
        printf("POR/PDR reset\n");
    }
    
    if (reset_source & RCC_CSR_PINRSTF) {
        printf("PIN reset\n");
    }
    
    if (reset_source & RCC_CSR_BORRSTF) {
        printf("BOR reset\n");
    }
    
    清除复位标志
    RCC->CSR |= RCC_CSR_RMVF;
}

void EXTI0_IRQHandler(void) {
    if (__HAL_GPIO_EXTI_GET_IT(GPIO_PIN_0)) {
        __HAL_GPIO_EXTI_CLEAR_IT(GPIO_PIN_0);
        
        printf("Wakeup by GPIO\n");
        Wakeup_Handler();
    }
}
```

### 低功耗设计示例

#### 智能传感器节点

```c
typedef struct {
    uint32_t interval_ms;
    uint32_t active_time_ms;
    uint32_t sleep_time_ms;
} PowerProfile_t;

PowerProfile_t power_profile = {
    .interval_ms = 60000,      // 1分钟周期
    .active_time_ms = 100,     // 活跃时间100ms
    .sleep_time_ms = 59900     // 睡眠时间59.9秒
};

void Sensor_Node_Task(void *param) {
    Sensor_Init();
    Configure_RTC_Wakeup(power_profile.interval_ms);
    
    while(1) {
        记录活跃开始时间
        uint32_t start_time = HAL_GetTick();
        
        唤醒传感器
        Sensor_Wakeup();
        
        采集数据
        SensorData_t data;
        data.temperature = ReadTemperature();
        data.humidity = ReadHumidity();
        data.battery = ReadBattery();
        
        处理数据
        ProcessData(&data);
        
        保存数据
        Flash_Write(&data, sizeof(data));
        
        传感器进入睡眠
        Sensor_Sleep();
        
        计算实际活跃时间
        uint32_t active_time = HAL_GetTick() - start_time;
        printf("Active time: %d ms\n", active_time);
        
        进入停止模式
        Enter_StopMode();
    }
}

void RTC_WKUP_IRQHandler(void) {
    if (__HAL_RTC_WAKEUPTIMER_GET_FLAG(&hrtc, RTC_FLAG_WUTF)) {
        __HAL_RTC_WAKEUPTIMER_CLEAR_FLAG(&hrtc, RTC_FLAG_WUTF);
        
        唤醒后重新配置时钟
        SystemClock_Config();
    }
}
```

### 功耗优化技巧

#### 1. 时钟优化

```c
// 动态调整时钟频率
void Adjust_Clock_Frequency(uint8_t cpu_load) {
    if (cpu_load > 80) {
        设置最高时钟频率
        Set_Clock_Frequency(72);
    } else if (cpu_load > 50) {
        设置中等时钟频率
        Set_Clock_Frequency(36);
    } else {
        设置最低时钟频率
        Set_Clock_Frequency(8);
    }
}

void Set_Clock_Frequency(uint32_t freq_mhz) {
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};
    
    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSE;
    RCC_OscInitStruct.HSEState = RCC_HSE_ON;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
    RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSE;
    RCC_OscInitStruct.PLL.PLLMUL = RCC_PLL_MUL9;
    
    if (freq_mhz <= 8) {
        RCC_OscInitStruct.PLL.PLLState = RCC_PLL_OFF;
    }
    
    HAL_RCC_OscConfig(&RCC_OscInitStruct);
    
    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK |
                                    RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;
    
    HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_2);
}
```

```c
void Configure_Clock_Gating(void) {
    在不使用时关闭外设时钟
    ADC1->CR2 &= ~ADC_CR2_ADON;
    DAC->CR &= ~DAC_CR_EN1;
    TIM2->CR1 &= ~TIM_CR1_CEN;
    
    需要使用时再开启
    ADC1->CR2 |= ADC_CR2_ADON;
}
```

### 注意事项

1. **唤醒时间**：考虑唤醒时间对实时性的影响
2. **数据保留**：确保重要数据在低功耗模式下的保留
3. **外设状态**：唤醒后需要重新配置外设
4. **功耗测量**：实际测量功耗以验证优化效果
5. **温度影响**：温度变化会影响功耗和唤醒时间
