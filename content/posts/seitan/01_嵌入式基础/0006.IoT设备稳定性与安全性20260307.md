#### 3. 状态机设计

```c
typedef enum {
    STATE_INIT,
    STATE_IDLE,
    STATE_CONNECTING,
    STATE_CONNECTED,
    STATE_PROCESSING,
    STATE_ERROR,
    STATE_RECOVERING
} SystemState_t;

typedef struct {
    SystemState_t current_state;
    SystemState_t next_state;
    uint32_t state_enter_time;
    uint32_t error_count;
} StateMachine_t;

StateMachine_t state_machine;

void StateMachine_Init(void) {
    state_machine.current_state = STATE_INIT;
    state_machine.next_state = STATE_INIT;
    state_machine.error_count = 0;
}

void StateMachine_Run(void) {
    switch (state_machine.current_state) {
        case STATE_INIT:
            System_Init();
            state_machine.next_state = STATE_IDLE;
            break;
            
        case STATE_IDLE:
            if (Need_Connect()) {
                state_machine.next_state = STATE_CONNECTING;
            }
            break;
            
        case STATE_CONNECTING:
            if (Connect_To_Server()) {
                state_machine.next_state = STATE_CONNECTED;
            } else {
                state_machine.error_count++;
                if (state_machine.error_count > MAX_RETRY) {
                    state_machine.next_state = STATE_ERROR;
                } else {
                    state_machine.next_state = STATE_IDLE;
                }
            }
            break;
            
        case STATE_CONNECTED:
            if (Has_Data_To_Process()) {
                state_machine.next_state = STATE_PROCESSING;
            } else if (!Is_Connected()) {
                state_machine.next_state = STATE_IDLE;
            }
            break;
            
        case STATE_PROCESSING:
            Process_Data();
            state_machine.next_state = STATE_CONNECTED;
            break;
            
        case STATE_ERROR:
            Handle_Error();
            state_machine.next_state = STATE_RECOVERING;
            break;
            
        case STATE_RECOVERING:
            if (Recover_From_Error()) {
                state_machine.error_count = 0;
                state_machine.next_state = STATE_IDLE;
            } else {
                System_Reset();
            }
            break;
    }
    
    if (state_machine.current_state != state_machine.next_state) {
        printf("State transition: %d -> %d\n", 
               state_machine.current_state, state_machine.next_state);
        state_machine.current_state = state_machine.next_state;
        state_machine.state_enter_time = GetTick();
    }
}
```

```c
#include "mbedtls/sha256.h"
#include "mbedtls/hmac.h"

typedef struct {
    uint8_t device_id[16];
    uint8_t device_secret[32];
} DeviceCredentials_t;

DeviceCredentials_t credentials;

bool Generate_Device_ID(void) {
    使用设备唯一ID生成设备ID
    uint8_t uid[12];
    HAL_GetUID(uid);
    
    mbedtls_sha256_context ctx;
    mbedtls_sha256_init(&ctx);
    mbedtls_sha256_starts(&ctx, 0);
    mbedtls_sha256_update(&ctx, uid, sizeof(uid));
    mbedtls_sha256_finish(&ctx, credentials.device_id);
    mbedtls_sha256_free(&ctx);
    
    return true;
}

bool Authenticate_Device(const uint8_t *challenge, uint8_t challenge_len,
                          uint8_t *response, uint8_t *response_len) {
    计算HMAC-SHA256响应
    mbedtls_md_context_t ctx;
    mbedtls_md_init(&ctx);
    mbedtls_md_setup(&ctx, mbedtls_md_info_from_type(MBEDTLS_MD_SHA256), 1);
    
    mbedtls_md_hmac_starts(&ctx, credentials.device_secret, 32);
    mbedtls_md_hmac_update(&ctx, challenge, challenge_len);
    mbedtls_md_hmac_finish(&ctx, response);
    
    *response_len = 32;
    mbedtls_md_free(&ctx);
    
    return true;
}
```

#### 2. 通信加密

```c
#include "mbedtls/aes.h"
#include "mbedtls/ctr_drbg.h"
#include "mbedtls/entropy.h"

typedef struct {
    uint8_t key[32];
    uint8_t iv[16];
} CryptoContext_t;

CryptoContext_t crypto_ctx;

bool Init_Crypto(void) {
    mbedtls_entropy_context entropy;
    mbedtls_ctr_drbg_context ctr_drbg;
    
    mbedtls_entropy_init(&entropy);
    mbedtls_ctr_drbg_init(&ctr_drbg);
    
    mbedtls_ctr_drbg_seed(&ctr_drbg, mbedtls_entropy_func, &entropy,
                          NULL, 0);
    
    生成随机密钥和IV
    mbedtls_ctr_drbg_random(&ctr_drbg, crypto_ctx.key, 32);
    mbedtls_ctr_drbg_random(&ctr_drbg, crypto_ctx.iv, 16);
    
    mbedtls_ctr_drbg_free(&ctr_drbg);
    mbedtls_entropy_free(&entropy);
    
    return true;
}

bool Encrypt_Data(const uint8_t *plaintext, size_t plaintext_len,
                  uint8_t *ciphertext, size_t *ciphertext_len) {
    mbedtls_aes_context aes;
    mbedtls_aes_init(&aes);
    mbedtls_aes_setkey_enc(&aes, crypto_ctx.key, 256);
    
    使用AES-CBC加密
    size_t block_size = 16;
    size_t padding_len = block_size - (plaintext_len % block_size);
    size_t total_len = plaintext_len + padding_len;
    
    uint8_t *padded_plaintext = malloc(total_len);
    memcpy(padded_plaintext, plaintext, plaintext_len);
    memset(padded_plaintext + plaintext_len, padding_len, padding_len);
    
    mbedtls_aes_crypt_cbc(&aes, MBEDTLS_AES_ENCRYPT, total_len,
                         crypto_ctx.iv, padded_plaintext, ciphertext);
    
    *ciphertext_len = total_len;
    
    free(padded_plaintext);
    mbedtls_aes_free(&aes);
    
    return true;
}

bool Decrypt_Data(const uint8_t *ciphertext, size_t ciphertext_len,
                  uint8_t *plaintext, size_t *plaintext_len) {
    mbedtls_aes_context aes;
    mbedtls_aes_init(&aes);
    mbedtls_aes_setkey_dec(&aes, crypto_ctx.key, 256);
    
    使用AES-CBC解密
    uint8_t iv[16];
    memcpy(iv, crypto_ctx.iv, 16);
    
    mbedtls_aes_crypt_cbc(&aes, MBEDTLS_AES_DECRYPT, ciphertext_len,
                         iv, ciphertext, plaintext);
    
    移除PKCS7填充
    uint8_t padding_len = plaintext[ciphertext_len - 1];
    *plaintext_len = ciphertext_len - padding_len;
    
    mbedtls_aes_free(&aes);
    
    return true;
}
```

```c
#include "mbedtls/rsa.h"
#include "mbedtls/sha256.h"

bool Verify_Boot_Firmware(void) {
    读取固件
    uint32_t firmware_addr = Get_Active_App_Addr();
    uint32_t firmware_size = Get_Firmware_Size(firmware_addr);
    
    uint8_t *firmware = malloc(firmware_size);
    Flash_Read(firmware_addr, firmware, firmware_size);
    
    计算固件哈希
    uint8_t hash[32];
    mbedtls_sha256_context ctx;
    mbedtls_sha256_init(&ctx);
    mbedtls_sha256_starts(&ctx, 0);
    mbedtls_sha256_update(&ctx, firmware, firmware_size);
    mbedtls_sha256_finish(&ctx, hash);
    mbedtls_sha256_free(&ctx);
    
    读取签名
    uint8_t signature[256];
    Flash_Read(firmware_addr + firmware_size, signature, sizeof(signature));
    
    验证签名
    mbedtls_pk_context pk;
    mbedtls_pk_init(&pk);
    mbedtls_pk_parse_public_key(&pk, public_key, public_key_len);
    
    int ret = mbedtls_pk_verify(&pk, MBEDTLS_MD_SHA256,
                                hash, sizeof(hash),
                                signature, sizeof(signature));
    
    mbedtls_pk_free(&pk);
    free(firmware);
    
    return (ret == 0);
}
```

#### 4. 数据完整性

```c
uint32_t Calculate_CRC32(const uint8_t *data, uint32_t len) {
    uint32_t crc = 0xFFFFFFFF;
    
    for (uint32_t i = 0; i < len; i++) {
        crc ^= data[i];
        
        for (uint8_t j = 0; j < 8; j++) {
            if (crc & 1) {
                crc = (crc >> 1) ^ 0xEDB88320;
            } else {
                crc >>= 1;
            }
        }
    }
    
    return ~crc;
}

bool Verify_Data_Integrity(const uint8_t *data, uint32_t len,
                           uint32_t expected_crc) {
    uint32_t calculated_crc = Calculate_CRC32(data, len);
    
    return (calculated_crc == expected_crc);
}

void Add_CRC_To_Data(uint8_t *data, uint32_t len) {
    uint32_t crc = Calculate_CRC32(data, len);
    memcpy(data + len, &crc, sizeof(crc));
}
```

```c
    mbedtls_entropy_context entropy;

    mbedtls_entropy_init(&entropy);
    mbedtls_ctr_drbg_init(&ctr_drbg);
    mbedtls_ctr_drbg_seed(&ctr_drbg, mbedtls_entropy_func, &entropy, NULL, 0);
    
    mbedtls_ssl_conf_rng(&conf, mbedtls_ctr_drbg_random, &ctr_drbg);
    
    连接服务器
    int ret = mbedtls_net_connect(&server_fd, host, port, MBEDTLS_NET_PROTO_TCP);

        return false;
    }

    mbedtls_ssl_setup(&ssl, &conf);
    mbedtls_ssl_set_bio(&ssl, &server_fd, mbedtls_net_send, mbedtls_net_recv, NULL);

    if (ret != 0) {
        return false;
    }
    
    验证证书
    const uint32_t flags = mbedtls_ssl_get_verify_result(&ssl);
    if (flags != 0) {
        return false;
    }
    
    return true;
}
```

```c
bool Generate_Secure_Message(SecureMessage_t *msg, 
                               const uint8_t *data, uint32_t len) {
    msg->timestamp = GetTimestamp();
    msg->nonce = Get_Random_Nonce();
    memcpy(msg->data, data, len);
    
    计算HMAC
    mbedtls_md_context_t ctx;
    mbedtls_md_init(&ctx);
    mbedtls_md_setup(&ctx, mbedtls_md_info_from_type(MBEDTLS_MD_SHA256), 1);
    
    mbedtls_md_hmac_starts(&ctx, shared_secret, 32);
    mbedtls_md_hmac_update(&ctx, (uint8_t*)&msg->timestamp, sizeof(msg->timestamp));
    mbedtls_md_hmac_update(&ctx, (uint8_t*)&msg->nonce, sizeof(msg->nonce));
    mbedtls_md_hmac_update(&ctx, msg->data, len);
    mbedtls_md_hmac_finish(&ctx, msg->hmac);
    
    mbedtls_md_free(&ctx);

}

        return false;

    mbedtls_md_context_t ctx;
```
