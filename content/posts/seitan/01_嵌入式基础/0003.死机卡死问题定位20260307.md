# 死机卡死问题定位

## 标签
#嵌入式 #面试题 #调试 #死机 #卡死 #故障排查

## 问题描述
如何定位和解决嵌入式系统死机或卡死问题？

## 参考答案

### 死机卡死的常见原因

#### 1. 软件原因

**栈溢出**：
- 任务栈空间不足
- 递归调用过深
- 局部变量过大

**内存问题**：
- 野指针访问
- 内存泄漏
- 越界访问
- 双重释放

**死锁**：
- 资源竞争
- 优先级反转
- 错误的同步机制

**中断问题**：
- 中断嵌套过深
- 中断栈溢出
- 中断优先级设置错误

#### 2. 硬件原因

**电源问题**：
- 电压不稳定
- 电流不足
- 电源纹波大

**时钟问题**：
- 时钟配置错误
- 时钟源失效
- 时钟频率不稳定

**外设问题**：
- 外设配置错误
- 外设故障
- 外设冲突

**温度问题**：
- 过热保护
- 温度过高导致不稳定

**野指针检测**：
```c
#define MAGIC_NUMBER 0xDEADBEEF

typedef struct {
    void *ptr;
    uint32_t magic;
    size_t size;
} SafePtr_t;

SafePtr_t *SafeMalloc(size_t size) {
    SafePtr_t *safe_ptr = malloc(sizeof(SafePtr_t) + size);
    if (safe_ptr) {
        safe_ptr->ptr = (void *)(safe_ptr + 1);
        safe_ptr->magic = MAGIC_NUMBER;
        safe_ptr->size = size;
        memset(safe_ptr->ptr, 0xAA, size);  // 填充魔数
    }
    return safe_ptr;
}

void SafeFree(SafePtr_t *safe_ptr) {
    if (safe_ptr && safe_ptr->magic == MAGIC_NUMBER) {
        memset(safe_ptr->ptr, 0xDD, safe_ptr->size);  // 填充魔数
        safe_ptr->magic = 0;
        free(safe_ptr);
    }
}
```

**内存泄漏检测**：
```c
typedef struct {
    void *ptr;
    size_t size;
    const char *file;
    int line;
} AllocInfo_t;

#define MAX_ALLOCS 256
AllocInfo_t alloc_info[MAX_ALLOCS];
volatile uint32_t alloc_count = 0;

void *DebugMalloc(size_t size, const char *file, int line) {
    void *ptr = malloc(size);
    if (ptr && alloc_count < MAX_ALLOCS) {
        alloc_info[alloc_count].ptr = ptr;
        alloc_info[alloc_count].size = size;
        alloc_info[alloc_count].file = file;
        alloc_info[alloc_count].line = line;
        alloc_count++;
    }
    return ptr;
}

void DebugFree(void *ptr) {
    for (uint32_t i = 0; i < alloc_count; i++) {
        if (alloc_info[i].ptr == ptr) {
            alloc_info[i].ptr = NULL;
            break;
        }
    }
    free(ptr);
}

void PrintMemoryLeaks(void) {
    printf("Memory Leaks:\n");
    for (uint32_t i = 0; i < alloc_count; i++) {
        if (alloc_info[i].ptr != NULL) {
            printf("  Leak: %p, size=%d, %s:%d\n",
                   alloc_info[i].ptr,
                   alloc_info[i].size,
                   alloc_info[i].file,
                   alloc_info[i].line);
        }
    }
}

#define malloc(size) DebugMalloc(size, __FILE__, __LINE__)
#define free(ptr) DebugFree(ptr)
```

#### 3. 死锁检测

**死锁检测**：
```c
#define DEADLOCK_TIMEOUT_MS 10000

typedef struct {
    SemaphoreHandle_t mutex;
    uint32_t lock_time;
    const char *task_name;
    const char *file;
    int line;
} MutexInfo_t;

MutexInfo_t mutex_info[10];

bool SafeMutexTake(SemaphoreHandle_t mutex, TickType_t timeout,
                   const char *file, int line) {
    uint32_t start_time = GetTick();
    bool result = (xSemaphoreTake(mutex, timeout) == pdTRUE);
    
    if (result) {
        mutex_info[index].mutex = mutex;
        mutex_info[index].lock_time = start_time;
        mutex_info[index].task_name = pcTaskGetName(NULL);
        mutex_info[index].file = file;
        mutex_info[index].line = line;
    } else {
        printf("Potential deadlock: %s waiting %s:%d for %d ms\n",
               pcTaskGetName(NULL), file, line,
               GetTick() - start_time);
    }
    
    return result;
}

void CheckDeadlocks(void) {
    uint32_t current_time = GetTick();
    
    for (int i = 0; i < 10; i++) {
        if (mutex_info[i].mutex != NULL) {
            uint32_t lock_duration = current_time - mutex_info[i].lock_time;
            if (lock_duration > DEADLOCK_TIMEOUT_MS) {
                printf("Possible deadlock detected:\n");
                printf("  Mutex held by %s\n", mutex_info[i].task_name);
                printf("  Locked for %d ms\n", lock_duration);
                printf("  Locked at %s:%d\n", mutex_info[i].file, mutex_info[i].line);
            }
        }
    }
}

#define xSemaphoreTake(m, t) SafeMutexTake(m, t, __FILE__, __LINE__)
```
