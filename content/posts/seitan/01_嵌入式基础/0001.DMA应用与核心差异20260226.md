# DMA应用与核心差异

## 标签
#嵌入式 #面试题 #DMA #数据传输 #性能优化

## 问题描述
1. DMA哪里用到了？（ADC、数据收发）
2. DMA使用与否的核心差异点

## 参考答案

### 1. DMA应用场景

**DMA（Direct Memory Access）**直接存储器访问，允许外设直接与内存交换数据，无需CPU干预。

#### 1.1 ADC采样中使用DMA

**传统方式（CPU搬运）：**
```c
// 传统ADC采样
uint16_t adc_values[100];

void adc_sample_without_dma() {
    for (int i = 0; i < 100; i++) {
        ADC_StartConversion(ADC1);
        while (!ADC_GetFlagStatus(ADC1, ADC_FLAG_EOC));
        adc_values[i] = ADC_GetConversionValue(ADC1);
    }
}
```

**DMA方式：**
```c
// ADC + DMA采样
uint16_t adc_values[100];

void adc_sample_with_dma() {
    // 配置DMA
    DMA_InitTypeDef DMA_InitStruct;
    DMA_DeInit(DMA1_Channel1);
    
    DMA_InitStruct.DMA_PeripheralBaseAddr = (uint32_t)&ADC1->DR;
    DMA_InitStruct.DMA_MemoryBaseAddr = (uint32_t)adc_values;
    DMA_InitStruct.DMA_DIR = DMA_DIR_PeripheralSRC;
    DMA_InitStruct.DMA_BufferSize = 100;
    DMA_InitStruct.DMA_PeripheralInc = DMA_PeripheralInc_Disable;
    DMA_InitStruct.DMA_MemoryInc = DMA_MemoryInc_Enable;
    DMA_InitStruct.DMA_PeripheralDataSize = DMA_PeripheralDataSize_HalfWord;
    DMA_InitStruct.DMA_MemoryDataSize = DMA_MemoryDataSize_HalfWord;
    DMA_InitStruct.DMA_Mode = DMA_Mode_Normal;
    DMA_InitStruct.DMA_Priority = DMA_Priority_High;
    DMA_InitStruct.DMA_M2M = DMA_M2M_Disable;
    
    DMA_Init(DMA1_Channel1, &DMA_InitStruct);
    DMA_Cmd(DMA1_Channel1, ENABLE);
    
    // 配置ADC使用DMA
    ADC_DMACmd(ADC1, ENABLE);
    
    // 启动ADC
    ADC_SoftwareStartConvCmd(ADC1, ENABLE);
    
    // CPU可以执行其他任务
    do_other_tasks();
    
    // 等待DMA完成
    while (!DMA_GetFlagStatus(DMA1_FLAG_TC1));
    DMA_ClearFlag(DMA1_FLAG_TC1);
}
```

#### 1.2 串口数据收发中使用DMA

**发送数据：**
```c
#define UART_TX_BUFFER_SIZE 256
uint8_t uart_tx_buffer[UART_TX_BUFFER_SIZE];

void uart_send_dma(const uint8_t *data, uint16_t len) {
    // 复制数据到发送缓冲区
    memcpy(uart_tx_buffer, data, len);
    
    // 配置DMA发送
    DMA_Cmd(DMA1_Channel4, DISABLE);
    DMA1_Channel4->CNDTR = len;
    DMA1_Channel4->CMAR = (uint32_t)uart_tx_buffer;
    DMA_Cmd(DMA1_Channel4, ENABLE);
    
    // 等待发送完成
    while (!DMA_GetFlagStatus(DMA1_FLAG_TC4));
    DMA_ClearFlag(DMA1_FLAG_TC4);
}
```

**接收数据：**
```c
#define UART_RX_BUFFER_SIZE 256
uint8_t uart_rx_buffer[UART_RX_BUFFER_SIZE];
volatile uint16_t rx_received = 0;

void uart_receive_dma_init() {
    // 配置DMA接收
    DMA_InitTypeDef DMA_InitStruct;
    DMA_DeInit(DMA1_Channel5);
    
    DMA_InitStruct.DMA_PeripheralBaseAddr = (uint32_t)&USART1->DR;
    DMA_InitStruct.DMA_MemoryBaseAddr = (uint32_t)uart_rx_buffer;
    DMA_InitStruct.DMA_DIR = DMA_DIR_PeripheralSRC;
    DMA_InitStruct.DMA_BufferSize = UART_RX_BUFFER_SIZE;
    DMA_InitStruct.DMA_PeripheralInc = DMA_PeripheralInc_Disable;
    DMA_InitStruct.DMA_MemoryInc = DMA_MemoryInc_Enable;
    DMA_InitStruct.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte;
    DMA_InitStruct.DMA_MemoryDataSize = DMA_MemoryDataSize_Byte;
    DMA_InitStruct.DMA_Mode = DMA_Mode_Circular;  // 循环模式
    DMA_InitStruct.DMA_Priority = DMA_Priority_High;
    DMA_InitStruct.DMA_M2M = DMA_M2M_Disable;
    
    DMA_Init(DMA1_Channel5, &DMA_InitStruct);
    DMA_Cmd(DMA1_Channel5, ENABLE);
    
    // 使能串口DMA接收
    USART_DMACmd(USART1, USART_DMAReq_Rx, ENABLE);
}

void DMA1_Channel5_IRQHandler(void) {
    if (DMA_GetITStatus(DMA1_IT_TC5)) {
        DMA_ClearITPendingBit(DMA1_IT_TC5);
        
        // 处理接收到的数据
        process_received_data(uart_rx_buffer, UART_RX_BUFFER_SIZE);
    }
}
```

#### 1.3 SPI/I2C数据传输中使用DMA

**SPI DMA发送：**
```c
void spi_send_dma(const uint8_t *data, uint16_t len) {
    // 配置SPI DMA发送
    DMA_DeInit(DMA2_Channel3);
    
    DMA_InitTypeDef DMA_InitStruct;
    DMA_InitStruct.DMA_PeripheralBaseAddr = (uint32_t)&SPI1->DR;
    DMA_InitStruct.DMA_MemoryBaseAddr = (uint32_t)data;
    DMA_InitStruct.DMA_DIR = DMA_DIR_PeripheralDST;
    DMA_InitStruct.DMA_BufferSize = len;
    DMA_InitStruct.DMA_PeripheralInc = DMA_PeripheralInc_Disable;
    DMA_InitStruct.DMA_MemoryInc = DMA_MemoryInc_Enable;
    DMA_InitStruct.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte;
    DMA_InitStruct.DMA_MemoryDataSize = DMA_MemoryDataSize_Byte;
    DMA_InitStruct.DMA_Mode = DMA_Mode_Normal;
    DMA_InitStruct.DMA_Priority = DMA_Priority_High;
    DMA_InitStruct.DMA_M2M = DMA_M2M_Disable;
    
    DMA_Init(DMA2_Channel3, &DMA_InitStruct);
    DMA_Cmd(DMA2_Channel3, ENABLE);
    
    // 使能SPI DMA
    SPI_DMACmd(SPI1, SPI_DMAReq_Tx, ENABLE);
    
    // 等待传输完成
    while (!DMA_GetFlagStatus(DMA2_FLAG_TC3));
    DMA_ClearFlag(DMA2_FLAG_TC3);
}
```

### 2. DMA使用与否的核心差异

#### 2.1 CPU利用率

**不使用DMA：**
```
数据搬运：CPU逐字节/字搬运
CPU占用率：高（100%搬运期间无法执行其他任务）
适用场景：数据量小、CPU空闲
```

**使用DMA：**
```
数据搬运：DMA控制器自动搬运
CPU占用率：低（仅需配置DMA，搬运期间可执行其他任务）
适用场景：数据量大、CPU任务繁重
```

**对比示例：**
```c
// 不使用DMA：CPU持续占用
void copy_without_dma(uint8_t *src, uint8_t *dst, uint32_t size) {
    for (uint32_t i = 0; i < size; i++) {
        dst[i] = src[i];  // CPU执行搬运
        // 无法执行其他任务
    }
}

// 使用DMA：CPU空闲
void copy_with_dma(uint8_t *src, uint8_t *dst, uint32_t size) {
    // 配置DMA（CPU耗时很短）
    DMA_Config(src, dst, size);
    DMA_Start();
    
    // CPU可以执行其他任务
    process_other_tasks();
    
    // 等待DMA完成
    DMA_WaitComplete();
}
```

#### 2.2 传输效率

**不使用DMA：**
- CPU指令周期有限
- 每次搬运需要取指、译码、执行
- 传输速度受CPU主频限制

**使用DMA：**
- 专用DMA控制器
- 突发传输模式
- 传输速度接近总线速度

**性能对比（以STM32为例）：**
```c
// 测试：传输1KB数据
// 不使用DMA：约1ms（72MHz CPU）
// 使用DMA：约0.1ms（总线带宽）

#define TEST_SIZE 1024
uint8_t src[TEST_SIZE];
uint8_t dst[TEST_SIZE];

void performance_test() {
    uint32_t start_time, end_time;
    
    // 测试不使用DMA
    start_time = get_tick();
    copy_without_dma(src, dst, TEST_SIZE);
    end_time = get_tick();
    printf("CPU拷贝耗时: %lu us\n", end_time - start_time);
    
    // 测试使用DMA
    start_time = get_tick();
    copy_with_dma(src, dst, TEST_SIZE);
    end_time = get_tick();
    printf("DMA拷贝耗时: %lu us\n", end_time - start_time);
}
```

#### 2.3 系统实时性

**不使用DMA：**
- CPU被数据搬运占用
- 实时任务响应延迟增加
- 可能导致任务调度问题

**使用DMA：**
- CPU空闲处理实时任务
- 系统响应性更好
- 支持多任务并发

#### 2.4 功耗优化

**不使用DMA：**
- CPU持续运行搬运数据
- 功耗较高

**使用DMA：**
- CPU可进入低功耗模式
- DMA搬运期间CPU休眠
- 功耗显著降低

**低功耗示例：**
```c
void low_power_adc_sample() {
    // 配置ADC + DMA
    adc_dma_config();
    
    // CPU进入休眠
    __WFI();  // Wait For Interrupt
    
    // DMA完成唤醒CPU
    while (!DMA_GetFlagStatus(DMA_FLAG_TC));
    DMA_ClearFlag(DMA_FLAG_TC);
    
    // 处理数据
    process_adc_data();
}
```

#### 2.5 内存访问模式

**不使用DMA：**
- CPU访问内存
- 可能影响缓存命中率
- 访问模式由CPU指令决定

**使用DMA：**
- DMA直接访问内存
- 支持对齐访问
- 支持突发传输

### 3. DMA配置要点

**关键配置项：**
```c
DMA_InitStruct.DMA_PeripheralInc = DMA_PeripheralInc_Disable;  // 外设地址不递增
DMA_InitStruct.DMA_MemoryInc = DMA_MemoryInc_Enable;          // 内存地址递增
DMA_InitStruct.DMA_Mode = DMA_Mode_Circular;                  // 循环模式（ADC常用）
DMA_InitStruct.DMA_Priority = DMA_Priority_VeryHigh;          // 优先级设置
```

**中断配置：**
```c
// 使能DMA传输完成中断
DMA_ITConfig(DMA1_Channel1, DMA_IT_TC, ENABLE);
NVIC_EnableIRQ(DMA1_Channel1_IRQn);

// 中断服务函数
void DMA1_Channel1_IRQHandler(void) {
    if (DMA_GetITStatus(DMA1_IT_TC1)) {
        DMA_ClearITPendingBit(DMA1_IT_TC1);
        // 处理完成事件
    }
}
```

### 4. 总结

| 特性 | 不使用DMA | 使用DMA |
|------|----------|---------|
| CPU利用率 | 高（搬运期间） | 低 |
| 传输效率 | 低 | 高 |
| 实时性 | 差 | 好 |
| 功耗 | 高 | 低 |
| 复杂度 | 简单 | 中等 |
| 适用场景 | 小数据量 | 大数据量 |
