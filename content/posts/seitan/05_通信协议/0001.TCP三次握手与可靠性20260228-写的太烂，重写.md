# TCP三次握手与可靠性保证

## 题目

TCP三次握手的过程，为什么不可以两次握手？TCP是如何保证可靠性的？

## 答案

### TCP三次握手过程

**三次握手步骤：**

```
客户端                              服务器
  |                                   |
  |-----------SYN,seq=x------------->|
  |                                   |
  |<---SYN,ACK,seq=y,ack=x+1---------|
  |                                   |
  |-----------ACK,seq=x+1,ack=y+1--->|
  |                                   |
  |      建立连接完成                   |
```

**代码示例：**

```cpp
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <iostream>
#include <cstring>

int createTcpClient(const char* ip, int port) {
    int sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) {
        std::cerr << "Socket creation failed" << std::endl;
        return -1;
    }
    
    struct sockaddr_in server_addr;
    memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(port);
    inet_pton(AF_INET, ip, &server_addr.sin_addr);
    
    std::cout << "Sending SYN..." << std::endl;
    if (connect(sockfd, (struct sockaddr*)&server_addr, sizeof(server_addr)) < 0) {
        std::cerr << "Connection failed" << std::endl;
        close(sockfd);
        return -1;
    }
    
    std::cout << "Three-way handshake completed" << std::endl;
    return sockfd;
}

int createTcpServer(int port) {
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) {
        return -1;
    }
    
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));
    
    struct sockaddr_in address;
    memset(&address, 0, sizeof(address));
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(port);
    
    bind(server_fd, (struct sockaddr*)&address, sizeof(address));
    listen(server_fd, 3);
    
    std::cout << "Server listening on port " << port << std::endl;
    return server_fd;
}

void handleTcpConnection(int server_fd) {
    struct sockaddr_in client_addr;
    socklen_t addr_len = sizeof(client_addr);
    
    std::cout << "Waiting for SYN..." << std::endl;
    int client_fd = accept(server_fd, (struct sockaddr*)&client_addr, &addr_len);
    if (client_fd < 0) {
        std::cerr << "Accept failed" << std::endl;
        return;
    }
    
    std::cout << "Three-way handshake completed" << std::endl;
    std::cout << "Connection accepted from: " 
              << inet_ntoa(client_addr.sin_addr) << std::endl;
    
    close(client_fd);
}
```

### 为什么需要三次握手

**1. 防止已失效的连接请求突然又传到服务端**

```cpp
class TcpStateDiagram {
public:
    enum State {
        CLOSED,
        LISTEN,
        SYN_SENT,
        SYN_RECEIVED,
        ESTABLISHED,
        FIN_WAIT_1,
        FIN_WAIT_2,
        TIME_WAIT,

        CLOSING
    };

        State from;
        State to;
        const char* event;
        const char* action;
    };
    
    static void printTransitions() {
        std::vector<Transition> transitions = {
            {CLOSED, LISTEN, "passive OPEN", "create socket, bind, listen"},
            {CLOSED, SYN_SENT, "active OPEN, send SYN", "connect()"},
            {LISTEN, SYN_RECEIVED, "recv SYN, send SYN+ACK", "accept()"},
            {SYN_SENT, ESTABLISHED, "recv SYN+ACK, send ACK", "connection established"},
            {SYN_RECEIVED, ESTABLISHED, "recv ACK", "connection established"},
            {ESTABLISHED, FIN_WAIT_1, "close, send FIN", "shutdown(SHUT_WR)"},
            {FIN_WAIT_1, FIN_WAIT_2, "recv ACK", "half-closed"},
            {FIN_WAIT_2, TIME_WAIT, "recv FIN, send ACK", "wait 2MSL"},

            {CLOSE_WAIT, LAST_ACK, "close, send FIN", "application closed"},
            {TIME_WAIT, CLOSED, "timeout 2MSL", "connection terminated"}
        };

            std::cout << stateToString(t.from) << " -> " 
                      << stateToString(t.to) << ": " 
                      << t.event << " (" << t.action << ")" << std::endl;
        }
    }
    
private:
    static const char* stateToString(State state) {
        switch (state) {
            case CLOSED: return "CLOSED";
            case LISTEN: return "LISTEN";
            case SYN_SENT: return "SYN_SENT";
            case SYN_RECEIVED: return "SYN_RECEIVED";
            case ESTABLISHED: return "ESTABLISHED";
            case FIN_WAIT_1: return "FIN_WAIT_1";
            case FIN_WAIT_2: return "FIN_WAIT_2";
            case TIME_WAIT: return "TIME_WAIT";
            case CLOSE_WAIT: return "CLOSE_WAIT";
            case LAST_ACK: return "LAST_ACK";
            case CLOSING: return "CLOSING";
            default: return "UNKNOWN";
        }
    }
};
```

### TCP可靠性保证机制

**1. 序列号和确认应答**

```cpp
class TcpReliability {
public:
    struct TcpHeader {
        uint16_t source_port;
        uint16_t dest_port;
        uint32_t seq_num;
        uint32_t ack_num;
        uint8_t data_offset;
        uint8_t flags;
        uint16_t window_size;
        uint16_t checksum;
        uint16_t urgent_ptr;
    };
    
    class ReliableSender {
    private:
        uint32_t next_seq_num;
        std::map<uint32_t, std::string> unacknowledged_data;
        int max_retransmissions;
        
    public:
        ReliableSender() : next_seq_num(1000), max_retransmissions(3) {}
        
        void sendData(const std::string& data) {
            uint32_t seq = next_seq_num;
            unacknowledged_data[seq] = data;
            next_seq_num += data.length();
            
            std::cout << "Sending data, seq=" << seq 
                      << ", length=" << data.length() << std::endl;
        }
        
        bool receiveAck(uint32_t ack_num) {
            auto it = unacknowledged_data.find(ack_num);
            if (it != unacknowledged_data.end()) {
                unacknowledged_data.erase(it);
                std::cout << "ACK received for seq=" << ack_num 
                          << ", unacknowledged=" << unacknowledged_data.size() << std::endl;
                return true;
            }
            return false;
        }
        
        void retransmit() {
            for (auto& pair : unacknowledged_data) {
                std::cout << "Retransmitting data, seq=" << pair.first << std::endl;
            }
        }
    };
    
    class ReliableReceiver {
    private:
        uint32_t expected_seq_num;
        
    public:
        ReliableReceiver() : expected_seq_num(1001) {}
        
        bool receiveData(uint32_t seq_num, const std::string& data) {
            if (seq_num == expected_seq_num) {
                std::cout << "Received data, seq=" << seq_num << std::endl;
                expected_seq_num += data.length();
                return true;
            } else if (seq_num < expected_seq_num) {
                std::cout << "Duplicate packet, seq=" << seq_num << std::endl;
                return false;
            } else {
                std::cout << "Out of order packet, seq=" << seq_num << std::endl;
                return false;
            }
        }
        
        uint32_t getAckNum() const {
            return expected_seq_num;
        }
    };
};
```

```cpp
class RetransmissionTimer {
private:
    std::map<uint32_t, std::chrono::steady_clock::time_point> packet_timestamps;
    std::chrono::milliseconds rto;
    int max_retransmissions;
    
public:
    RetransmissionTimer(int rto_ms = 1000, int max_retrans = 3) 
        : rto(rto_ms), max_retransmissions(max_retrans) {}
    
    void recordPacket(uint32_t seq_num) {
        packet_timestamps[seq_num] = std::chrono::steady_clock::now();
        std::cout << "Recorded packet seq=" << seq_num << std::endl;
    }
    
    void acknowledgePacket(uint32_t seq_num) {
        packet_timestamps.erase(seq_num);
        std::cout << "Acknowledged packet seq=" << seq_num << std::endl;
    }
    
    void checkTimeouts() {
        auto now = std::chrono::steady_clock::now();
        
        for (auto it = packet_timestamps.begin(); it != packet_timestamps.end(); ) {
            auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
                now - it->second).count();
            
            if (elapsed >= rto.count()) {
                std::cout << "Timeout for packet seq=" << it->first 
                          << ", elapsed=" << elapsed << "ms" << std::endl;
                std::cout << "Need to retransmit" << std::endl;
                it = packet_timestamps.erase(it);
            } else {
                ++it;
            }
        }
    }
    
    void updateRTO(int new_rto_ms) {
        rto = std::chrono::milliseconds(new_rto_ms);
        std::cout << "Updated RTO to " << new_rto_ms << "ms" << std::endl;
    }
};
```

**3. 滑动窗口**

```cpp
class SlidingWindow {
private:
    uint32_t send_base;
    uint32_t next_seq_num;
    size_t window_size;
    std::map<uint32_t, std::string> window_buffer;
    
public:
    SlidingWindow(size_t window_size = 10) 
        : send_base(1000), next_seq_num(1000), window_size(window_size) {}
    
    bool canSend() {
        return (next_seq_num - send_base) < window_size;
    }
    
    void sendData(const std::string& data) {
        if (!canSend()) {
            std::cout << "Window full, cannot send" << std::endl;
            return;
        }
        
        window_buffer[next_seq_num] = data;
        std::cout << "Sent data, seq=" << next_seq_num 
                  << ", window=[" << send_base << ", " << next_seq_num << ")" << std::endl;
        next_seq_num += data.length();
    }
    
    void receiveAck(uint32_t ack_num) {
        auto it = window_buffer.begin();
        while (it != window_buffer.end() && it->first < ack_num) {
            std::cout << "Acked packet seq=" << it->first << std::endl;
            it = window_buffer.erase(it);
        }
        send_base = ack_num;
        std::cout << "Window advanced, send_base=" << send_base << std::endl;
    }
    
    void printWindowStatus() {
        std::cout << "Window status: [" << send_base << ", " << next_seq_num << ")" << std::endl;
        std::cout << "Buffered packets: " << window_buffer.size() << std::endl;
    }
};
```

```cpp
class CongestionControl {
private:
    size_t cwnd;
    size_t ssthresh;
    enum Phase { SLOW_START, CONGESTION_AVOIDANCE, FAST_RECOVERY };
    Phase current_phase;
    
public:
    CongestionControl() : cwnd(1), ssthresh(65535), current_phase(SLOW_START) {
        std::cout << "Initialized: cwnd=" << cwnd << ", ssthresh=" << ssthresh << std::endl;
    }
    
    void onAckReceived() {
        switch (current_phase) {
            case SLOW_START:
                cwnd *= 2;
                std::cout << "Slow start: cwnd doubled to " << cwnd << std::endl;
                if (cwnd >= ssthresh) {
                    current_phase = CONGESTION_AVOIDANCE;
                    std::cout << "Entered congestion avoidance phase" << std::endl;
                }
                break;
                
            case CONGESTION_AVOIDANCE:
                cwnd++;
                std::cout << "Congestion avoidance: cwnd increased to " << cwnd << std::endl;
                break;
                
            case FAST_RECOVERY:
                cwnd++;
                std::cout << "Fast recovery: cwnd increased to " << cwnd << std::endl;
                break;
        }
    }
    
    void onPacketLoss() {
        ssthresh = cwnd / 2;
        cwnd = 1;
        current_phase = SLOW_START;
        std::cout << "Packet loss detected: ssthresh=" << ssthresh 
                  << ", cwnd reset to " << cwnd << std::endl;
    }
    
    void onTripleDuplicateAck() {
        ssthresh = cwnd / 2;
        cwnd = ssthresh + 3;
        current_phase = FAST_RECOVERY;
        std::cout << "Triple duplicate ACK: ssthresh=" << ssthresh 
                  << ", cwnd set to " << cwnd << std::endl;
    }
    
    size_t getWindowSize() const {
        return cwnd;
    }
};
```

```cpp
class FlowControl {
private:
    size_t receive_window;
    size_t receive_buffer_size;
    size_t receive_buffer_used;
    
public:
    FlowControl(size_t buffer_size = 65535) 
        : receive_window(buffer_size), receive_buffer_size(buffer_size), 
          receive_buffer_used(0) {}
    
    size_t getWindowSize() const {
        return receive_window;
    }
    
    void receiveData(size_t data_size) {
        receive_buffer_used += data_size;
        receive_window = receive_buffer_size - receive_buffer_used;
        std::cout << "Received " << data_size << " bytes, window=" << receive_window << std::endl;
    }
    
    void consumeData(size_t data_size) {
        receive_buffer_used = (receive_buffer_used > data_size) ? 
                                receive_buffer_used - data_size : 0;
        receive_window = receive_buffer_size - receive_buffer_used;
        std::cout << "Consumed " << data_size << " bytes, window=" << receive_window << std::endl;
    }
    
    void advertiseWindow(size_t new_window) {
        receive_window = new_window;
        std::cout << "Advertising window size: " << receive_window << std::endl;
    }
};
```

### 总结

**TCP三次握手原因：**
1. 防止已失效的连接请求突然传到服务端
2. 确认双方的接收和发送能力
3. 同步初始序列号

**TCP可靠性保证机制：**
1. 序列号和确认应答：确保数据有序到达
2. 超时重传：处理丢包情况
3. 滑动窗口：提高传输效率
4. 拥塞控制：避免网络拥塞
5. 流量控制：防止接收端缓冲区溢出

**为什么不能两次握手：**
- 无法确认双方的接收能力
- 无法防止历史连接的干扰
- 可能导致资源浪费和数据不一致
