# epoll工作原理与select/poll区别

## 标签
#嵌入式 #面试题 #epoll #IO多路复用 #网络编程

## 问题描述
epoll的工作原理，与select/poll的区别？ET和LT模式的区别及应用场景？

## 参考答案

### 1. epoll工作原理

#### 1.1 epoll核心概念

**epoll**是Linux特有的I/O事件通知机制，用于高效监控大量文件描述符。

**三个核心函数：**

**1. epoll_create：创建epoll实例**
```c
#include <sys/epoll.h>

int epfd = epoll_create1(0);
if (epfd < 0) {
    perror("epoll_create1 failed");
    return -1;
}
// 返回epoll文件描述符
```

**2. epoll_ctl：添加/修改/删除监控的文件描述符**
```c
struct epoll_event ev;
ev.events = EPOLLIN;  // 监控可读事件
ev.data.fd = sockfd;

// 添加fd到epoll
if (epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd, &ev) < 0) {
    perror("epoll_ctl failed");
    close(epfd);
    return -1;
}

// 修改fd的事件
ev.events = EPOLLIN | EPOLLOUT;
epoll_ctl(epfd, EPOLL_CTL_MOD, sockfd, &ev);

// 删除fd
epoll_ctl(epfd, EPOLL_CTL_DEL, sockfd, NULL);
```

**3. epoll_wait：等待事件发生**
```c
#define MAX_EVENTS 1024
struct epoll_event events[MAX_EVENTS];

int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
if (nfds < 0) {
    perror("epoll_wait failed");
    return -1;
}

// 处理就绪的fd
for (int i = 0; i < nfds; i++) {
    if (events[i].events & EPOLLIN) {
        // 可读事件
        int fd = events[i].data.fd;
        handle_read_event(fd);
    }
    if (events[i].events & EPOLLOUT) {
        // 可写事件
        int fd = events[i].data.fd;
        handle_write_event(fd);
    }
}
```

#### 1.2 epoll底层实现

**数据结构：**

**1. 红黑树（rbtree）：管理所有监控的fd**
```c
struct eventpoll {
    struct rb_root rbr;  // 红黑树根节点
    struct list_head rdllist;  // 就绪链表
    spinlock_t lock;  // 自旋锁保护
    wait_queue_head_t wq;  // 等待队列
};

struct epitem {
    struct rb_node rbn;  // 红黑树节点
    struct list_head rdllink;  // 就绪链表节点
    struct epoll_filefd ffd;  // 文件描述符信息
    struct eventpoll *ep;  // 所属epoll实例
    struct epoll_event event;  // 关注的事件
};
```

**工作流程：**

1. **添加fd（epoll_ctl）：**
```c
epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd, &ev);

// 内部流程：
1. 创建epitem结构
2. 设置epitem->event = ev
3. 将epitem插入红黑树
4. 注册回调函数ep_poll_callback
5. 时间复杂度：O(log n)
```

2. **fd就绪时：**
```c
// 当fd有事件时，内核调用回调函数
ep_poll_callback(wait_queue_entry_t *wait, unsigned mode, int sync, void *key) {
    // 1. 查找对应的epitem
    struct epitem *epi = ep_item_from_wait(wait);
    
    // 2. 将epitem添加到就绪链表rdllist
    list_add_tail(&epi->rdllink, &epi->ep->rdllist);
    
    // 3. 唤醒等待的进程
    wake_up(&epi->ep->wq);
}
```

3. **等待事件（epoll_wait）：**
```c
epoll_wait(epfd, events, MAX_EVENTS, timeout);

// 内部流程：
1. 检查就绪链表rdllist是否为空
2. 如果不为空，直接返回就绪的fd
3. 如果为空，进程进入睡眠
4. 被唤醒时，从rdllist取出就绪的fd
5. 时间复杂度：O(1)
```

### 2. epoll vs select/poll

#### 2.1 对比表格

| 特性 | select | poll | epoll |
|------|--------|------|-------|
| 最大连接数 | 1024（FD_SETSIZE） | 无限制 | 无限制 |
| fd集合拷贝 | 每次调用拷贝 | 每次调用拷贝 | 仅注册时拷贝 |
| 查找就绪fd | O(n)遍历 | O(n)遍历 | O(1)直接获取 |
| 内存效率 | 固定大小（1024位） | 固定大小数组 | 动态分配 |
| 工作方式 | 轮询 | 轮询 | 事件驱动 |
| 跨平台 | 是 | 是 | 仅Linux |
| 性能 | 差（连接多时） | 差（连接多时） | 优 |

#### 2.2 select详解

**工作原理：**
```c
fd_set readfds;
FD_ZERO(&readfds);
FD_SET(sockfd, &readfds);

int max_fd = sockfd;

int ret = select(max_fd + 1, &readfds, NULL, NULL, NULL);
if (ret < 0) {
    perror("select failed");
    return -1;
}

if (FD_ISSET(sockfd, &readfds)) {
    // sockfd可读
    handle_read(sockfd);
}
```

**缺点：**
```c
1. fd数量限制：
   - FD_SETSIZE默认为1024
   - 修改需要重新编译内核

2. 性能问题：
   - 每次调用需要将fd_set从用户态拷贝到内核态
   - 内核需要遍历所有fd，O(n)复杂度
   - 连接越多，性能越差

3. 内存开销：
   - 固定大小的位图
   - 即使只有少量连接，也分配1024位

4. 接口不友好：
   - 每次调用需要重新设置fd_set
   - 无法知道哪些fd就绪，需要遍历检查
```

#### 2.3 poll详解

**工作原理：**
```c
struct pollfd fds[MAX_FDS];
fds[0].fd = sockfd;
fds[0].events = POLLIN;  // 监控可读事件

int ret = poll(fds, 1, -1);
if (ret < 0) {
    perror("poll failed");
    return -1;
}

if (fds[0].revents & POLLIN) {
    handle_read(fds[0].fd);
}
```

**缺点：**
```c
1. 性能问题：
   - 每次调用需要将pollfd数组拷贝到内核态
   - 内核需要遍历所有fd，O(n)复杂度
   - 连接越多，性能越差

2. 内存开销：
   - 固定大小的数组
   - 需要预估最大连接数

3. 接口不友好：
   - 需要维护pollfd数组
   - 每次调用需要重新设置events
```

#### 2.4 epoll优势详解

**优势1：红黑树管理，无fd数量限制**
```c
// select/poll：固定大小
fd_set readfds;  // 最多1024个fd
struct pollfd fds[10000];  // 需要预估大小

// epoll：动态管理
int epfd = epoll_create1(0);
// 可以添加任意数量的fd（受系统资源限制）
for (int i = 0; i < 100000; i++) {
    epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd[i], &ev);
}
```

**优势2：事件驱动，只返回就绪的fd**
```c
// select/poll：需要遍历所有fd
int ret = select(max_fd + 1, &readfds, NULL, NULL, NULL);
for (int i = 0; i <= max_fd; i++) {
    if (FD_ISSET(i, &readfds)) {  // O(n)遍历
        handle_read(i);
    }
}

// epoll：只处理就绪的fd
int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
for (int i = 0; i < nfds; i++) {  // 只遍历就绪的
    handle_read(events[i].data.fd);
}
```

**优势3：mmap共享内存，减少拷贝**
```c
// select/poll：每次调用都拷贝
select(max_fd + 1, &readfds, NULL, NULL, NULL);
// 拷贝readfds从用户态到内核态

// epoll：仅注册时拷贝
epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd, &ev);
// 拷贝ev到内核态，之后不需要重复拷贝
epoll_wait(epfd, events, MAX_EVENTS, -1);
// 直接读取就绪事件，无需拷贝
```

### 3. ET和LT模式

#### 3.1 模式定义

**LT（Level Trigger，水平触发）：**
- 默认模式
- 只要fd就绪，就会通知
- 可能多次通知同一事件
- 编程简单，兼容性好

**ET（Edge Trigger，边缘触发）：**
- 需要显式设置（EPOLLET）
- 只在状态变化时通知一次
- 效率更高，但编程复杂

#### 3.2 工作方式对比

**LT模式：**
```c
struct epoll_event ev;
ev.events = EPOLLIN;  // 默认LT模式
ev.data.fd = sockfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd, &ev);

int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
for (int i = 0; i < nfds; i++) {
    if (events[i].events & EPOLLIN) {
        int fd = events[i].data.fd;
        
        // 可以不读完数据
        char buffer[100];
        int len = recv(fd, buffer, sizeof(buffer), 0);
        
        // 下次epoll_wait仍会通知可读
    }
}
```

**ET模式：**
```c
struct epoll_event ev;
ev.events = EPOLLIN | EPOLLET;  // ET模式
ev.data.fd = sockfd;
epoll_ctl(epfd, EPOLL_CTL_ADD, sockfd, &ev);

int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
for (int i = 0; i < nfds; i++) {
    if (events[i].events & EPOLLIN) {
        int fd = events[i].data.fd;
        
        // 必须读完所有数据，否则下次不会通知
        while (1) {
            char buffer[100];
            int len = recv(fd, buffer, sizeof(buffer), 0);
            if (len <= 0) {
                if (len < 0 && errno == EAGAIN) {
                    break;  // 数据读完
                }
                break;
            }
            handle_data(buffer, len);
        }
    }
}
```

#### 3.3 完整ET模式服务器示例

```c
#define MAX_EVENTS 1024
#define BUFFER_SIZE 4096

void set_nonblocking(int fd) {
    int flags = fcntl(fd, F_GETFL, 0);
    fcntl(fd, F_SETFL, flags | O_NONBLOCK);
}

int epoll_et_server(int port) {
    int listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    set_nonblocking(listen_fd);
    
    // 绑定和监听
    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(port);
    bind(listen_fd, (struct sockaddr*)&addr, sizeof(addr));
    listen(listen_fd, 128);
    
    // 创建epoll实例
    int epfd = epoll_create1(0);
    
    // 添加监听fd，使用ET模式
    struct epoll_event ev;
    ev.events = EPOLLIN | EPOLLET;
    ev.data.fd = listen_fd;
    epoll_ctl(epfd, EPOLL_CTL_ADD, listen_fd, &ev);
    
    struct epoll_event events[MAX_EVENTS];
    
    while (1) {
        int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
        if (nfds < 0) {
            perror("epoll_wait failed");
            break;
        }
        
        for (int i = 0; i < nfds; i++) {
            int fd = events[i].data.fd;
            
            if (fd == listen_fd) {
                // 接受新连接（ET模式需要循环accept）
                while (1) {
                    struct sockaddr_in client_addr;
                    socklen_t addr_len = sizeof(client_addr);
                    int conn_fd = accept(listen_fd, 
                                       (struct sockaddr*)&client_addr, 
                                       &addr_len);
                    if (conn_fd < 0) {
                        if (errno == EAGAIN || errno == EWOULDBLOCK) {
                            break;  // 没有更多连接
                        }
                        perror("accept failed");
                        break;
                    }
                    
                    set_nonblocking(conn_fd);
                    
                    // 添加到epoll，ET模式
                    ev.events = EPOLLIN | EPOLLET;
                    ev.data.fd = conn_fd;
                    epoll_ctl(epfd, EPOLL_CTL_ADD, conn_fd, &ev);
                    
                    printf("新连接: %d\n", conn_fd);
                }
            } else {
                // 处理可读事件（ET模式需要循环读取）
                if (events[i].events & EPOLLIN) {
                    char buffer[BUFFER_SIZE];
                    while (1) {
                        int len = recv(fd, buffer, sizeof(buffer), 0);
                        if (len < 0) {
                            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                                break;  // 数据读完
                            }
                            perror("recv failed");
                            close(fd);
                            epoll_ctl(epfd, EPOLL_CTL_DEL, fd, NULL);
                            break;
                        } else if (len == 0) {
                            printf("连接关闭: %d\n", fd);
                            close(fd);
                            epoll_ctl(epfd, EPOLL_CTL_DEL, fd, NULL);
                            break;
                        } else {
                            // 处理数据
                            handle_data(buffer, len);
                        }
                    }
                }
            }
        }
    }
    
    close(listen_fd);
    close(epfd);
    return 0;
}
```

### 4. 应用场景选择

**LT模式适用场景：**
- 简单的应用程序
- 数据量较小的场景
- 不需要极致性能的场景
- 快速原型开发

**ET模式适用场景：**
- 高并发服务器
- 需要高性能的场景
- 连接数上万的大型系统
- Nginx、Redis等高性能服务器
