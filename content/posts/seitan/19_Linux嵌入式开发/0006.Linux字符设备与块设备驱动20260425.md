# Linux字符设备与块设备驱动

## 题目

Linux设备驱动中字符设备和块设备的区别是什么？file_operations 结构体你实现过哪些操作？有没有踩过坑？

> 来源：零跑汽车嵌入式岗面试

## 考察点

Linux设备驱动模型、字符设备与块设备的本质区别、file_operations结构体、驱动开发实践

## 回答要点

### 1. 字符设备与块设备的本质区别

Linux内核将设备分为三大类：字符设备、块设备和网络设备。其中字符设备和块设备最核心的区别在于**数据访问方式**和**内核是否提供缓存**。

**字符设备（Character Device）**：以字节流为单位进行顺序或随机访问，应用程序可以直接通过read/write系统调用与设备交互，数据不经过内核缓冲。典型设备包括串口（UART）、I2C设备、SPI设备、LED、GPIO、传感器等。

**块设备（Block Device）**：以固定大小的块（通常为512字节或4KB）为单位进行访问，数据经过内核的页缓存（Page Cache）和I/O调度器，支持随机访问。典型设备包括硬盘、SD卡、eMMC、NAND Flash等。

下面从多个维度进行详细对比：

| 对比维度 | 字符设备 | 块设备 |
|---------|---------|--------|
| 访问单位 | 字节/字符流 | 固定大小的块（512B/4KB） |
| 内核缓冲 | 无缓冲，直接读写 | 有页缓存（Page Cache） |
| I/O调度 | 无I/O调度器 | 有I/O调度器（电梯算法等） |
| 随机访问 | 通常不支持或有限支持 | 支持高效随机访问 |
| 数据流向 | 应用程序 ↔ 设备 | 应用程序 ↔ 页缓存 ↔ 设备 |
| 核心结构体 | cdev + file_operations | gendisk + block_device_operations |
| 注册接口 | register_chrdev_region / alloc_chrdev_region | register_blkdev / blk_mq_alloc_disk |
| 设备文件类型 | 字符设备文件（c） | 块设备文件（b） |
| 典型设备 | 串口、LED、I2C、SPI、传感器 | 硬盘、SD卡、eMMC、NAND |
| read/write语义 | 直接操作硬件 | 经过页缓存，可能不立即写盘 |
| 支持mmap | 部分支持 | 完整支持 |
| 系统调用路径 | VFS → 字符设备驱动 | VFS → 页缓存 → I/O调度 → 块设备驱动 |
| 驱动复杂度 | 相对简单 | 较复杂（需处理请求队列、bio等） |
| 示例主设备号 | 串口(4/204), I2C(89) | SATA(8), SD(179), loop(7) |

**关键理解**：字符设备像"管道"，数据一个字节一个字节地流过；块设备像"仓库"，数据按固定大小的"箱子"（块）存取。内核为块设备提供了缓存和调度机制，以提高I/O性能，而字符设备则追求低延迟的即时响应。

### 2. 字符设备驱动框架

字符设备驱动的核心是 **cdev 结构体** 和 **file_operations 结构体**。整个注册流程如下：

```
alloc_chrdev_region()  →  分配设备号
        ↓
cdev_init()            →  初始化cdev，绑定file_operations
        ↓
cdev_add()             →  将cdev添加到系统
        ↓
class_create()         →  创建设备类（/sys/class/xxx）
        ↓
device_create()        →  创建设备节点（/dev/xxx）
```

**设备号的组成**：

```
dev_t dev = MKDEV(major, minor);
unsigned int major = MAJOR(dev);
unsigned int minor = MINOR(dev);
```

高12位为主设备号（标识驱动程序），低20位为次设备号（标识具体设备实例）。

**完整的字符设备驱动框架代码**：

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>
#include <linux/slab.h>

#define DEVICE_NAME     "mychar"
#define BUF_SIZE        1024

static dev_t dev_num;
static struct cdev my_cdev;
static struct class *my_class;
static struct device *my_device;
static char kernel_buf[BUF_SIZE];
static DEFINE_MUTEX(buf_mutex);

static int my_open(struct inode *inode, struct file *filp)
{
    pr_info("%s: device opened\n", DEVICE_NAME);
    return 0;
}

static int my_release(struct inode *inode, struct file *filp)
{
    pr_info("%s: device closed\n", DEVICE_NAME);
    return 0;
}

static ssize_t my_read(struct file *filp, char __user *buf,
                       size_t count, loff_t *f_pos)
{
    int ret;

    if (mutex_lock_interruptible(&buf_mutex))
        return -ERESTARTSYS;

    if (*f_pos >= BUF_SIZE) {
        mutex_unlock(&buf_mutex);
        return 0;
    }

    if (count > BUF_SIZE - *f_pos)
        count = BUF_SIZE - *f_pos;

    ret = copy_to_user(buf, kernel_buf + *f_pos, count);
    if (ret) {
        mutex_unlock(&buf_mutex);
        return -EFAULT;
    }

    *f_pos += count;
    mutex_unlock(&buf_mutex);

    pr_info("%s: read %zu bytes at offset %lld\n",
            DEVICE_NAME, count, *f_pos - count);
    return count;
}

static ssize_t my_write(struct file *filp, const char __user *buf,
                        size_t count, loff_t *f_pos)
{
    int ret;

    if (mutex_lock_interruptible(&buf_mutex))
        return -ERESTARTSYS;

    if (*f_pos >= BUF_SIZE) {
        mutex_unlock(&buf_mutex);
        return -ENOSPC;
    }

    if (count > BUF_SIZE - *f_pos)
        count = BUF_SIZE - *f_pos;

    ret = copy_from_user(kernel_buf + *f_pos, buf, count);
    if (ret) {
        mutex_unlock(&buf_mutex);
        return -EFAULT;
    }

    *f_pos += count;
    mutex_unlock(&buf_mutex);

    pr_info("%s: write %zu bytes at offset %lld\n",
            DEVICE_NAME, count, *f_pos - count);
    return count;
}

static const struct file_operations my_fops = {
    .owner   = THIS_MODULE,
    .open    = my_open,
    .release = my_release,
    .read    = my_read,
    .write   = my_write,
};

static int __init my_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        pr_err("alloc_chrdev_region failed: %d\n", ret);
        return ret;
    }

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev_num, 1);
    if (ret < 0) {
        pr_err("cdev_add failed: %d\n", ret);
        goto err_unregister;
    }

    my_class = class_create(DEVICE_NAME);
    if (IS_ERR(my_class)) {
        pr_err("class_create failed\n");
        ret = PTR_ERR(my_class);
        goto err_cdev_del;
    }

    my_device = device_create(my_class, NULL, dev_num, NULL, DEVICE_NAME);
    if (IS_ERR(my_device)) {
        pr_err("device_create failed\n");
        ret = PTR_ERR(my_device);
        goto err_class_destroy;
    }

    pr_info("%s: registered, major=%d, minor=%d\n",
            DEVICE_NAME, MAJOR(dev_num), MINOR(dev_num));
    return 0;

err_class_destroy:
    class_destroy(my_class);
err_cdev_del:
    cdev_del(&my_cdev);
err_unregister:
    unregister_chrdev_region(dev_num, 1);
    return ret;
}

static void __exit my_exit(void)
{
    device_destroy(my_class, dev_num);
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    pr_info("%s: unregistered\n", DEVICE_NAME);
}

module_init(my_init);
module_exit(my_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("seitan");
MODULE_DESCRIPTION("A simple character device driver");
```

### 3. file_operations 结构体详解

`file_operations` 是字符设备驱动的核心，定义了VFS（虚拟文件系统）与设备驱动之间的接口。当用户空间程序调用 `open()`、`read()`、`write()` 等系统调用时，内核通过VFS找到对应的 `file_operations` 并调用驱动中注册的函数。

以下是常用操作的详细说明：

| 操作 | 函数签名 | 作用 |
|------|---------|------|
| open | `int (*open)(struct inode *, struct file *)` | 打开设备文件，初始化设备，申请资源 |
| release | `int (*release)(struct inode *, struct file *)` | 关闭设备文件，释放资源（注意：不是每次close都调用） |
| read | `ssize_t (*read)(struct file *, char __user *, size_t, loff_t *)` | 从设备读取数据到用户空间 |
| write | `ssize_t (*write)(struct file *, const char __user *, size_t, loff_t *)` | 将用户空间数据写入设备 |
| ioctl | `long (*unlocked_ioctl)(struct file *, unsigned int, unsigned long)` | 设备控制命令，配置设备参数 |
| mmap | `int (*mmap)(struct file *, struct vm_area_struct *)` | 将设备内存映射到用户空间，实现零拷贝 |
| poll | `__poll_t (*poll)(struct file *, struct poll_table_struct *)` | 实现非阻塞I/O，配合select/poll/epoll使用 |
| llseek | `loff_t (*llseek)(struct file *, loff_t, int)` | 修改文件的读写偏移量 |
| fsync | `int (*fsync)(struct file *, loff_t, loff_t, int datasync)` | 将缓存数据同步到设备 |

**open/release 的注意事项**：

`open` 在用户空间调用 `open()` 时触发，用于初始化设备、分配资源。`release` 在最后一个引用关闭时调用（不是每次 `close()` 都触发，因为 `fork`/`dup` 会增加引用计数）。在 `open` 中申请的资源必须在 `release` 中释放，否则会导致资源泄漏。

**ioctl 命令的构造规范**：

ioctl命令码由四个部分组成，必须使用内核提供的宏来构造：

```c
/* _IO(type, nr)           - 无参数命令 */
/* _IOR(type, nr, dtype)   - 从驱动读取数据到用户空间 */
/* _IOW(type, nr, dtype)   - 从用户空间写入数据到驱动 */
/* _IOWR(type, nr, dtype)  - 双向传输 */

#define MY_MAGIC  'M'

#define MY_CMD_RESET       _IO(MY_MAGIC, 0)
#define MY_CMD_GET_STATUS  _IOR(MY_MAGIC, 1, int)
#define MY_CMD_SET_SPEED   _IOW(MY_MAGIC, 2, int)
#define MY_CMD_GET_INFO    _IOWR(MY_MAGIC, 3, struct my_info)
```

**mmap 的典型应用场景**：

在嵌入式开发中，mmap常用于将设备的寄存器空间或DMA缓冲区映射到用户空间，避免频繁的系统调用开销。例如，将帧缓冲区（Framebuffer）映射到用户空间实现图形绘制，或将ADC的DMA缓冲区映射到用户空间实现高速数据采集。

**poll 的实现要点**：

poll操作使设备支持非阻塞I/O。驱动中通常使用 `wait_queue_head_t` 等待队列来实现：

```c
static DECLARE_WAIT_QUEUE_HEAD(my_waitq);
static unsigned int my_poll_flag = 0;

static __poll_t my_poll(struct file *filp, struct poll_table_struct *wait)
{
    __poll_t mask = 0;

    poll_wait(filp, &my_waitq, wait);

    if (my_poll_flag)
        mask |= EPOLLIN | EPOLLRDNORM;

    return mask;
}
```

当数据就绪时，在中断处理函数或工作队列中调用 `wake_up_interruptible(&my_waitq)` 唤醒等待的进程。

### 4. 块设备驱动框架

块设备驱动相比字符设备驱动要复杂得多，核心在于处理请求队列（Request Queue）和块I/O（bio）。

**block_device_operations 结构体**：

块设备的操作接口比 `file_operations` 简单得多，因为大部分I/O操作由内核的通用块层（Generic Block Layer）处理：

```c
struct block_device_operations {
    int (*open)(struct block_device *, fmode_t);
    void (*release)(struct gendisk *, fmode_t);
    int (*ioctl)(struct block_device *, fmode_t, unsigned int, unsigned long);
    int (*compat_ioctl)(struct block_device *, fmode_t, unsigned int, unsigned long);
    int (*direct_access)(struct block_device *, sector_t, void **, pfn_t *);
    unsigned int (*check_events)(struct gendisk *, unsigned int clearing);
    void (*unlock_native_capacity)(struct gendisk *);
    int (*revalidate_disk)(struct gendisk *);
    int (*getgeo)(struct block_device *, struct hd_geometry *);
    void (*swap_slot_free_notify)(struct block_device *, unsigned long);
    int (*report_zones)(struct block_device *, sector_t, unsigned int,
                        void *, report_zones_cb, void *);
    char *(*devnode)(struct gendisk *, umode_t *);
    struct module *owner;
};
```

**块设备注册流程**（基于blk-mq框架，现代内核推荐）：

```
blk_mq_alloc_sq_tag_set()  →  分配tag set
        ↓
blk_mq_alloc_disk()        →  分配gendisk和请求队列
        ↓
设置 gendisk 参数（容量、逻辑块大小等）
        ↓
add_disk()                 →  注册块设备
```

**块设备与字符设备的关键差异**：

| 差异点 | 字符设备 | 块设备 |
|--------|---------|--------|
| 数据传输 | 驱动直接处理read/write | 通过bio和请求队列，由通用块层调度 |
| 缓存机制 | 无 | 页缓存（Page Cache） |
| I/O合并 | 无 | 内核自动合并相邻请求 |
| 数据对齐 | 无要求 | 必须按块大小对齐 |
| 支持分区 | 不支持 | 天然支持（通过分区表） |
| 支持文件系统 | 不支持 | 支持挂载文件系统（ext4/fat等） |
| 驱动入口 | file_operations.read/write | blk_mq_ops.queue_rq |

**块设备驱动框架代码**：

```c
#include <linux/module.h>
#include <linux/blkdev.h>
#include <linux/blk-mq.h>
#include <linux/hdreg.h>
#include <linux/slab.h>

#define MY_BLKDEV_NAME    "myblkdev"
#define SECTOR_SIZE       512
#define NR_SECTORS        2048

static struct gendisk *my_disk;
static struct blk_mq_tag_set tag_set;
static struct request_queue *my_queue;
static u8 *my_disk_data;

static blk_status_t my_queue_rq(struct blk_mq_hw_ctx *hctx,
                                 const struct blk_mq_queue_data *bd)
{
    struct request *req = bd->rq;
    struct bio_vec bvec;
    struct bvec_iter iter;
    sector_t sector = blk_rq_pos(req);

    blk_mq_start_request(req);

    bio_for_each_segment(bvec, req->bio, iter) {
        void *dst = my_disk_data + (sector * SECTOR_SIZE);
        unsigned int len = bvec.bv_len;

        if (op_is_write(req_op(req)))
            memcpy_from_bvec(dst, &bvec);
        else
            memcpy_to_bvec(&bvec, dst);

        sector += len / SECTOR_SIZE;
    }

    blk_mq_end_request(req, BLK_STS_OK);
    return BLK_STS_OK;
}

static struct blk_mq_ops my_mq_ops = {
    .queue_rq = my_queue_rq,
};

static int __init my_blk_init(void)
{
    int ret;

    my_disk_data = kzalloc(NR_SECTORS * SECTOR_SIZE, GFP_KERNEL);
    if (!my_disk_data)
        return -ENOMEM;

    ret = blk_mq_alloc_sq_tag_set(&tag_set, &my_mq_ops, 128,
                                  BLK_MQ_F_SHOULD_MERGE);
    if (ret)
        goto err_free_data;

    my_disk = blk_mq_alloc_disk(&tag_set, NULL);
    if (IS_ERR(my_disk)) {
        ret = PTR_ERR(my_disk);
        goto err_free_tag_set;
    }

    my_queue = my_disk->queue;
    my_disk->major = register_blkdev(0, MY_BLKDEV_NAME);
    if (my_disk->major < 0) {
        ret = my_disk->major;
        goto err_put_disk;
    }

    strscpy(my_disk->disk_name, MY_BLKDEV_NAME, DISK_NAME_LEN);
    set_capacity(my_disk, NR_SECTORS);
    blk_queue_logical_block_size(my_queue, SECTOR_SIZE);

    ret = add_disk(my_disk);
    if (ret)
        goto err_unregister;

    pr_info("%s: registered, major=%d, capacity=%d sectors\n",
            MY_BLKDEV_NAME, my_disk->major, NR_SECTORS);
    return 0;

err_unregister:
    unregister_blkdev(my_disk->major, MY_BLKDEV_NAME);
err_put_disk:
    put_disk(my_disk);
err_free_tag_set:
    blk_mq_free_tag_set(&tag_set);
err_free_data:
    kfree(my_disk_data);
    return ret;
}

static void __exit my_blk_exit(void)
{
    del_gendisk(my_disk);
    put_disk(my_disk);
    blk_mq_free_tag_set(&tag_set);
    unregister_blkdev(my_disk->major, MY_BLKDEV_NAME);
    kfree(my_disk_data);
    pr_info("%s: unregistered\n", MY_BLKDEV_NAME);
}

module_init(my_blk_init);
module_exit(my_blk_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("seitan");
MODULE_DESCRIPTION("A simple block device driver");
```

### 5. 驱动开发中的常见坑

在实际驱动开发中，以下问题非常常见，面试中如果能结合实际经验讲出来会非常加分。

#### 5.1 并发问题：多进程同时访问设备

**问题描述**：多个进程同时打开同一个设备文件并进行读写操作，导致数据竞争。例如进程A正在写入数据，进程B同时读取，可能读到半写状态的数据。

**解决方案**：使用互斥锁（mutex）或自旋锁（spinlock）保护共享资源。

```c
static DEFINE_MUTEX(dev_mutex);

static ssize_t my_write(struct file *filp, const char __user *buf,
                        size_t count, loff_t *f_pos)
{
    if (mutex_lock_interruptible(&dev_mutex))
        return -ERESTARTSYS;

    /* 临界区操作 */

    mutex_unlock(&dev_mutex);
    return count;
}
```

**选择锁的原则**：如果临界区可能睡眠（如调用copy_to_user、kmalloc(GFP_KERNEL)），必须用mutex；如果临界区在中断上下文或极短，用spinlock。在嵌入式场景中，大部分字符设备的read/write都会涉及用户空间拷贝，因此mutex更常用。

#### 5.2 资源泄漏：open/release不配对

**问题描述**：在 `open` 中申请了内存或注册了中断，但在 `release` 中没有正确释放。由于 `fork`/`dup` 机制，`close()` 不一定会触发 `release`，导致资源持续泄漏。

**典型案例**：

```c
static int my_open(struct inode *inode, struct file *filp)
{
    struct my_dev *dev = kmalloc(sizeof(*dev), GFP_KERNEL);
    if (!dev)
        return -ENOMEM;
    filp->private_data = dev;
    return 0;
}

static int my_release(struct inode *inode, struct file *filp)
{
    /* 忘记释放 filp->private_data 导致内存泄漏 */
    return 0;
}
```

**解决方案**：在 `release` 中释放 `open` 中申请的所有资源，并使用引用计数确保配对：

```c
static int my_release(struct inode *inode, struct file *filp)
{
    struct my_dev *dev = filp->private_data;
    if (dev) {
        kfree(dev);
        filp->private_data = NULL;
    }
    return 0;
}
```

#### 5.3 错误处理：copy_to_user/copy_from_user返回值未检查

**问题描述**：`copy_to_user` 和 `copy_from_user` 可能因为用户空间地址无效而部分拷贝或完全失败，如果不检查返回值，会导致数据丢失或内核安全问题。

**错误写法**：

```c
copy_to_user(buf, kernel_buf, count);
return count;
```

**正确写法**：

```c
ret = copy_to_user(buf, kernel_buf, count);
if (ret)
    return -EFAULT;
return count - ret;
```

**更严重的问题**：直接使用 `memcpy` 代替 `copy_to_user`。用户空间指针可能随时失效（页面被换出、地址无效等），必须使用内核提供的 `copy_to_user`/`copy_from_user`，它们会进行安全检查。

#### 5.4 设备号冲突

**问题描述**：使用固定设备号（`register_chrdev(MAJOR_NUM, ...)`）时，如果主设备号已被其他驱动占用，注册会失败。在产品开发中，不同模块的驱动可能由不同团队开发，容易出现冲突。

**解决方案**：使用动态分配设备号：

```c
/* 推荐：动态分配 */
ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);

/* 不推荐：硬编码主设备号 */
ret = register_chrdev(240, DEVICE_NAME, &my_fops);
```

动态分配后可以通过 `/proc/devices` 查看分配的设备号。在产品发布时，可以通过udev规则创建固定的设备节点名称。

#### 5.5 ioctl命令编号不规范

**问题描述**：手动构造ioctl命令码（如 `#define MY_CMD 1`）可能导致不同驱动的命令码冲突，且无法通过 `ioctl` 的方向检查机制保证数据安全。

**错误写法**：

```c
#define MY_CMD_SET_VAL  1
#define MY_CMD_GET_VAL  2
```

**正确写法**：使用内核提供的宏构造命令码，并包含 Documentation/ioctl/ioctl-number.txt 中记录的magic number：

```c
#include <linux/ioctl.h>

#define MY_IOC_MAGIC  'm'

#define MY_IOC_RESET       _IO(MY_IOC_MAGIC, 0)
#define MY_IOC_SET_VAL     _IOW(MY_IOC_MAGIC, 1, int)
#define MY_IOC_GET_VAL     _IOR(MY_IOC_MAGIC, 2, int)
#define MY_IOC_RDWR_VAL    _IOWR(MY_IOC_MAGIC, 3, struct my_data)
```

这些宏会自动编码方向、类型、序号和数据大小，内核在调用驱动前会进行参数校验，防止用户空间传入非法参数。

### 6. 完整代码示例：带ioctl的字符设备驱动

下面给出一个更完整的字符设备驱动示例，包含 open、read、write、ioctl、release 的完整实现，并加入了并发保护和错误处理：

```c
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>
#include <linux/slab.h>
#include <linux/mutex.h>
#include <linux/ioctl.h>

#define DEVICE_NAME     "mydev"
#define BUF_SIZE        4096
#define MY_IOC_MAGIC    'k'

struct my_device_data {
    char buffer[BUF_SIZE];
    size_t data_len;
    int config;
};

#define MY_IOC_RESET       _IO(MY_IOC_MAGIC, 0)
#define MY_IOC_SET_CONFIG  _IOW(MY_IOC_MAGIC, 1, int)
#define MY_IOC_GET_CONFIG  _IOR(MY_IOC_MAGIC, 2, int)
#define MY_IOC_CLEAR_BUF   _IO(MY_IOC_MAGIC, 3)

static dev_t dev_num;
static struct cdev my_cdev;
static struct class *my_class;
static DEFINE_MUTEX(dev_mutex);

static int my_open(struct inode *inode, struct file *filp)
{
    struct my_device_data *data;

    data = kzalloc(sizeof(*data), GFP_KERNEL);
    if (!data)
        return -ENOMEM;

    filp->private_data = data;
    pr_info("%s: opened\n", DEVICE_NAME);
    return 0;
}

static int my_release(struct inode *inode, struct file *filp)
{
    struct my_device_data *data = filp->private_data;

    if (data) {
        kfree(data);
        filp->private_data = NULL;
    }

    pr_info("%s: released\n", DEVICE_NAME);
    return 0;
}

static ssize_t my_read(struct file *filp, char __user *buf,
                       size_t count, loff_t *f_pos)
{
    struct my_device_data *data = filp->private_data;
    ssize_t ret;

    if (mutex_lock_interruptible(&dev_mutex))
        return -ERESTARTSYS;

    if (*f_pos >= data->data_len) {
        ret = 0;
        goto out;
    }

    if (count > data->data_len - *f_pos)
        count = data->data_len - *f_pos;

    if (copy_to_user(buf, data->buffer + *f_pos, count)) {
        ret = -EFAULT;
        goto out;
    }

    *f_pos += count;
    ret = count;

out:
    mutex_unlock(&dev_mutex);
    return ret;
}

static ssize_t my_write(struct file *filp, const char __user *buf,
                        size_t count, loff_t *f_pos)
{
    struct my_device_data *data = filp->private_data;
    ssize_t ret;

    if (mutex_lock_interruptible(&dev_mutex))
        return -ERESTARTSYS;

    if (*f_pos + count > BUF_SIZE) {
        ret = -ENOSPC;
        goto out;
    }

    if (copy_from_user(data->buffer + *f_pos, buf, count)) {
        ret = -EFAULT;
        goto out;
    }

    *f_pos += count;
    if (*f_pos > data->data_len)
        data->data_len = *f_pos;
    ret = count;

out:
    mutex_unlock(&dev_mutex);
    return ret;
}

static long my_ioctl(struct file *filp, unsigned int cmd, unsigned long arg)
{
    struct my_device_data *data = filp->private_data;
    int ret = 0;
    int val;

    if (_IOC_TYPE(cmd) != MY_IOC_MAGIC)
        return -ENOTTY;

    if (mutex_lock_interruptible(&dev_mutex))
        return -ERESTARTSYS;

    switch (cmd) {
    case MY_IOC_RESET:
        data->data_len = 0;
        data->config = 0;
        pr_info("%s: device reset\n", DEVICE_NAME);
        break;

    case MY_IOC_SET_CONFIG:
        if (copy_from_user(&val, (int __user *)arg, sizeof(val))) {
            ret = -EFAULT;
            break;
        }
        data->config = val;
        pr_info("%s: config set to %d\n", DEVICE_NAME, val);
        break;

    case MY_IOC_GET_CONFIG:
        val = data->config;
        if (copy_to_user((int __user *)arg, &val, sizeof(val))) {
            ret = -EFAULT;
            break;
        }
        pr_info("%s: config get = %d\n", DEVICE_NAME, val);
        break;

    case MY_IOC_CLEAR_BUF:
        memset(data->buffer, 0, BUF_SIZE);
        data->data_len = 0;
        pr_info("%s: buffer cleared\n", DEVICE_NAME);
        break;

    default:
        ret = -ENOTTY;
        break;
    }

    mutex_unlock(&dev_mutex);
    return ret;
}

static loff_t my_llseek(struct file *filp, loff_t offset, int whence)
{
    struct my_device_data *data = filp->private_data;
    loff_t new_pos;

    switch (whence) {
    case SEEK_SET:
        new_pos = offset;
        break;
    case SEEK_CUR:
        new_pos = filp->f_pos + offset;
        break;
    case SEEK_END:
        new_pos = data->data_len + offset;
        break;
    default:
        return -EINVAL;
    }

    if (new_pos < 0 || new_pos > BUF_SIZE)
        return -EINVAL;

    filp->f_pos = new_pos;
    return new_pos;
}

static const struct file_operations my_fops = {
    .owner   = THIS_MODULE,
    .open    = my_open,
    .release = my_release,
    .read    = my_read,
    .write   = my_write,
    .unlocked_ioctl = my_ioctl,
    .llseek  = my_llseek,
};

static int __init my_init(void)
{
    int ret;

    ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        pr_err("alloc_chrdev_region failed: %d\n", ret);
        return ret;
    }

    cdev_init(&my_cdev, &my_fops);
    my_cdev.owner = THIS_MODULE;

    ret = cdev_add(&my_cdev, dev_num, 1);
    if (ret < 0) {
        pr_err("cdev_add failed: %d\n", ret);
        goto err_unregister;
    }

    my_class = class_create(DEVICE_NAME);
    if (IS_ERR(my_class)) {
        ret = PTR_ERR(my_class);
        goto err_cdev_del;
    }

    if (IS_ERR(device_create(my_class, NULL, dev_num, NULL, DEVICE_NAME))) {
        ret = PTR_ERR(my_device);
        goto err_class_destroy;
    }

    pr_info("%s: registered, major=%d\n", DEVICE_NAME, MAJOR(dev_num));
    return 0;

err_class_destroy:
    class_destroy(my_class);
err_cdev_del:
    cdev_del(&my_cdev);
err_unregister:
    unregister_chrdev_region(dev_num, 1);
    return ret;
}

static void __exit my_exit(void)
{
    device_destroy(my_class, dev_num);
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    pr_info("%s: unregistered\n", DEVICE_NAME);
}

module_init(my_init);
module_exit(my_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("seitan");
MODULE_DESCRIPTION("Character device driver with ioctl support");
```

**用户空间测试程序**：

```c
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <sys/ioctl.h>

#define MY_IOC_MAGIC    'k'
#define MY_IOC_RESET       _IO(MY_IOC_MAGIC, 0)
#define MY_IOC_SET_CONFIG  _IOW(MY_IOC_MAGIC, 1, int)
#define MY_IOC_GET_CONFIG  _IOR(MY_IOC_MAGIC, 2, int)
#define MY_IOC_CLEAR_BUF   _IO(MY_IOC_MAGIC, 3)

int main(void)
{
    int fd;
    char buf[128];
    int config;

    fd = open("/dev/mydev", O_RDWR);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    write(fd, "Hello, seitan driver!", 20);

    lseek(fd, 0, SEEK_SET);
    read(fd, buf, 20);
    buf[20] = '\0';
    printf("Read: %s\n", buf);

    ioctl(fd, MY_IOC_SET_CONFIG, 42);
    ioctl(fd, MY_IOC_GET_CONFIG, &config);
    printf("Config: %d\n", config);

    ioctl(fd, MY_IOC_RESET);
    close(fd);
    return 0;
}
```

## 总结

回答这道面试题时，建议按照以下结构展开：

1. **先讲本质区别**：从访问单位、缓冲机制、I/O调度三个核心维度区分字符设备和块设备，再展开到更多对比维度
2. **再讲驱动框架**：展示字符设备的注册流程和关键数据结构（cdev、file_operations、设备号）
3. **重点讲file_operations**：结合实际项目经验，说明自己实现过哪些操作，以及每个操作的使用场景
4. **补充块设备知识**：简要说明block_device_operations和请求队列，展示知识面
5. **最后讲踩坑经验**：这是加分项，结合实际项目讲遇到的并发、资源泄漏、ioctl规范等问题

面试官最看重的是**实际经验**，所以讲file_operations时一定要结合自己做过的项目，比如"在XX项目中我为传感器写了字符设备驱动，实现了read来读取数据、ioctl来配置采样率"这样的具体描述。
