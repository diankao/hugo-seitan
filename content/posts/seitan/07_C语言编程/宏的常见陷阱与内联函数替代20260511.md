# 宏的常见陷阱与内联函数替代

## 题目

C语言宏（`#define`）有哪些常见陷阱？有些情况下为什么必须用内联函数替代？

## 考察点

宏的预处理本质、宏的副作用、内联函数的优势、宏不可替代的场景。

## 回答要点

### 1. 宏的本质：文本替换

宏是**预处理阶段**的纯文本替换，编译器根本看不到宏的名字，只看到替换后的结果：

```
源代码 → 预处理器（展开宏）→ 编译器（编译替换后的代码）
```

```c
#define SQUARE(x)  ((x) * (x))

int a = SQUARE(3);
// 预处理后变成：
int a = ((3) * (3));  // 编译器只看到这个
```

**这个"文本替换"的本质，是一切陷阱的根源。**

### 2. 常见陷阱

#### 2.1 参数副作用：多次求值

```c
#define SQUARE(x)  ((x) * (x))

int i = 3;
int result = SQUARE(i++);
// 展开后：((i++) * (i++))
// i 被自增了两次！result 可能是 9 或 12（未定义行为）

// 内联函数没有这个问题
static inline int square(int x) {
    return x * x;
}

int j = 3;
int result2 = square(j++);  // j 只自增一次，result2 = 9，确定
```

#### 2.2 优先级陷阱：少加括号

```c
// 陷阱一：参数没加括号
#define DOUBLE(x)  x + x

int a = DOUBLE(3) * 2;
// 展开为：3 + 3 * 2 = 9，不是 12

// 修复：给参数加括号
#define DOUBLE(x)  ((x) + (x))

// 陷阱二：整个表达式没加括号
#define SHIFT(x)   (x) << 2

int b = SHIFT(3) | 0x01;
// 展开为：(3) << 2 | 0x01
// 优先级：<< 高于 |，所以是 (3 << 2) | 0x01 = 13
// 但如果意图是 (3 << 2) | 0x01 = 13 那就对了
// 如果意图是 (3 << (2 | 0x01)) 就错了

// 安全写法：外层再加括号
#define SHIFT(x)   ((x) << 2)
```

**规则：宏定义中，每个参数都要加括号，整个表达式也要加括号。**

#### 2.3 if-else 悬空问题

```c
#define SAFE_FREE(p)  if (p) { free(p); p = NULL; }

// 正常使用没问题
SAFE_FREE(ptr);

// 放在 if-else 中就出问题了
if (condition)
    SAFE_FREE(ptr);
else
    do_something();

// 展开后：
if (condition)
    if (ptr) { free(ptr); ptr = NULL; }
else        // 这个 else 匹配了内层的 if！
    do_something();

// 修复：用 do { ... } while(0) 包裹
#define SAFE_FREE(p)  do { if (p) { free(p); (p) = NULL; } } while(0)
```

#### 2.4 类型不安全

```c
#define MAX(a, b)  ((a) > (b) ? (a) : (b))

// 可以比较任意类型，编译器不会警告类型不匹配
int x = 10;
float y = 3.14f;
double z = MAX(x, y);  // int 和 float 混用，隐式转换

// 更危险的：指针和整数比较
int* p = NULL;
int q = 42;
auto r = MAX(p, q);  // 指针和 int 比较，编译器可能只给警告

// 内联函数有类型检查
static inline int max_int(int a, int b) {
    return a > b ? a : b;
}

float m = max_int(3.14f, 2.71f);  // float → int 隐式截断，编译器警告
```

#### 2.5 调试困难

```c
#define CALC(x)  ((x) * (x) + 2 * (x) + 1)

int result = CALC(get_value());

// 预处理后：
int result = ((get_value()) * (get_value()) + 2 * (get_value()) + 1);

// 调试器中看不到 CALC 这个名字
// 单步调试时无法"进入"宏
// 报错时显示的是展开后的长表达式
```

#### 2.6 作用域无视

```c
class Config {
    // 宏不遵守类作用域和命名空间
    #define MAX_SIZE 100   // 这个宏污染了全局！
    int data[MAX_SIZE];
};

// 即使在类外也能用 MAX_SIZE
int buffer[MAX_SIZE];  // OK，但这个宏本应只在 Config 内部

// 内联函数/static const 正确遵守作用域
class Config {
    static constexpr int MAX_SIZE = 100;  // 属于 Config 的成员
    int data[MAX_SIZE];
};
```

### 3. 内联函数的优势

| 方面 | 宏 | 内联函数 |
|------|----|---------|
| 处理阶段 | 预处理（文本替换） | 编译（真正的函数） |
| 类型检查 | ❌ 无 | ✅ 有 |
| 参数求值 | 可能多次 | **恰好一次** |
| 作用域 | 全局 | 遵守作用域规则 |
| 调试 | 困难 | **可以单步调试** |
| 递归 | ❌ 不支持 | ✅ 支持 |
| 取地址 | ❌ 不可以 | ✅ 可以取函数指针 |
| 条件编译内部 | ✅ 可以用 `#ifdef` | ❌ 不行 |

```c
// 宏能做到的，内联函数大部分都能做得更好
static inline int max(int a, int b) {
    return a > b ? a : b;
}

static inline int square(int x) {
    return x * x;
}

static inline void safe_free(void** pp) {
    if (pp && *pp) {
        free(*pp);
        *pp = NULL;
    }
}
```

### 4. 有些情况必须用内联函数

#### 4.1 参数有副作用时

```c
// 必须用内联函数
static inline int clamp(int val, int lo, int hi) {
    return val < lo ? lo : (val > hi ? hi : val);
}

int v = clamp(read_register++, 0, 100);
// 宏版本会让 read_register 自增多次
// 内联函数只自增一次
```

#### 4.2 需要类型安全时

```c
// 嵌入式中交换两个变量
static inline void swap_u16(uint16_t* a, uint16_t* b) {
    uint16_t tmp = *a;
    *a = *b;
    *b = tmp;
}

// 如果用宏，可能传入不同类型的指针，编译器不会警告
#define SWAP(a, b, type)  { type tmp = a; a = b; b = tmp; }
```

#### 4.3 C++ 中需要重载时

```cpp
// C++ 可以重载内联函数
inline int max(int a, int b) { return a > b ? a : b; }
inline float max(float a, float b) { return a > b ? a : b; }

// 宏无法重载，一个 MAX 通吃所有类型，没有类型检查
```

### 5. 有些情况必须用宏（内联函数替代不了）

#### 5.1 条件编译

```c
// 宏可以和 #ifdef 配合，内联函数不行
#ifdef DEBUG
#define LOG(fmt, ...)  printf("[DBG] " fmt "\n", ##__VA_ARGS__)
#else
#define LOG(fmt, ...)  ((void)0)
#endif

LOG("value = %d", x);   // DEBUG 模式输出，Release 模式完全消除
```

#### 5.2 字符串化与拼接

```c
// # 运算符：把参数变成字符串
#define TO_STR(x)   #x
const char* name = TO_STR(hello);  // → "hello"

// ## 运算符：拼接标识符
#define MAKE_FUNC(name)  handle_##name
void MAKE_FUNC(uart)(void);   // → void handle_uart(void);
void MAKE_FUNC(spi)(void);    // → void handle_spi(void);
```

#### 5.3 可变参数

```c
// C99 之前，__VA_ARGS__ 只能用宏实现
#define DEBUG_PRINT(fmt, ...) \
    do { \
        if (debug_enabled) \
            printf("[DBG %s:%d] " fmt "\n", __FILE__, __LINE__, ##__VA_ARGS__); \
    } while(0)

// C99 虽然有了 inline，但可变参数宏仍然更灵活
```

#### 5.4 编译时生成代码

```c
// 用宏批量生成枚举和字符串的对应关系
#define SENSOR_LIST  \
    X(TEMPERATURE)   \
    X(HUMIDITY)      \
    X(PRESSURE)      \
    X(LIGHT)

// 生成枚举
typedef enum {
    #define X(name) SENSOR_##name,
    SENSOR_LIST
    #undef X
    SENSOR_COUNT
} sensor_type_t;

// 生成字符串数组
const char* sensor_names[] = {
    #define X(name) #name,
    SENSOR_LIST
    #undef X
};

// 结果：
// sensor_type_t: { SENSOR_TEMPERATURE, SENSOR_HUMIDITY, SENSOR_PRESSURE, SENSOR_LIGHT, SENSOR_COUNT }
// sensor_names:  { "TEMPERATURE", "HUMIDITY", "PRESSURE", "LIGHT" }
```

#### 5.5 文件和行号信息

```c
// __FILE__、__LINE__、__func__ 只能在宏中捕获调用位置
#define ASSERT(cond) \
    do { \
        if (!(cond)) { \
            printf("ASSERT FAILED: %s, file %s, line %d\n", \
                   #cond, __FILE__, __LINE__); \
            while(1); \
        } \
    } while(0)

ASSERT(x > 0);  // 输出: ASSERT FAILED: x > 0, file main.c, line 42
```

### 6. 现代替代方案总结

| 宏的用途 | C 语言替代 | C++ 替代 |
|---------|-----------|---------|
| 常量 | `static const` / `enum` | `constexpr` |
| 函数式宏 | `static inline` 函数 | `inline` / `constexpr` 函数 |
| 类型泛型 | `_Generic`（C11） | 模板 |
| 条件编译 | 无替代（必须用宏） | `#if` / `#ifdef`（仍需宏） |
| 字符串化/拼接 | 无替代（必须用宏） | 无替代 |
| 断言/日志 | 无替代（需要 `__LINE__`） | 无替代 |
| X-Macro 代码生成 | 无替代 | 模板元编程（部分替代） |

### 7. 面试速记

- **宏的本质**：预处理阶段纯文本替换，编译器看不到宏
- **六大陷阱**：多次求值、优先级、悬空 else、类型不安全、调试困难、无视作用域
- **必须用内联函数**：参数有副作用、需要类型安全、C++ 需要重载
- **必须用宏**：条件编译、`#`/`##` 运算符、可变参数日志、`__FILE__`/`__LINE__`、X-Macro 代码生成
- **原则**：能用 `inline`/`constexpr` 就不用宏，宏只用于编译器本身不支持的功能
