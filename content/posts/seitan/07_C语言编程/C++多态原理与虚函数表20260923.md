### 1. 多态的概念

#### 编译时多态（静态绑定）
```cpp
// 函数重载
class Calculator {
public:
    int add(int a, int b) {
        return a + b;
    }

    double add(double a, double b) {
        return a + b;
    }

    std::string add(std::string a, std::string b) {
        return a + b;
    }
};

// 使用
Calculator calc;
calc.add(1, 2);          // 调用 int add(int, int)
calc.add(1.5, 2.5);      // 调用 double add(double, double)
```

#### 运行时多态（动态绑定）
```cpp
// 虚函数实现多态
class Animal {
public:
    virtual void makeSound() {
        std::cout << "Animal makes sound" << std::endl;
    }
};

class Dog : public Animal {
public:
    void makeSound() override {
        std::cout << "Dog barks" << std::endl;
    }
};

class Cat : public Animal {
public:
    void makeSound() override {
        std::cout << "Cat meows" << std::endl;
    }
};

// 使用
Animal* animal1 = new Dog();
Animal* animal2 = new Cat();

animal1->makeSound();  // 输出: Dog barks
animal2->makeSound();  // 输出: Cat meows
```

### 2. 虚函数表（vtable）

#### vtable 结构
```cpp
// 每个包含虚函数的类都有一个虚函数表
class Base {
public:
    virtual void func1() { std::cout << "Base::func1" << std::endl; }
    virtual void func2() { std::cout << "Base::func2" << std::endl; }
    void func3() { std::cout << "Base::func3" << std::endl; }
};

// Base 类的 vtable 结构
// vptr ──► [ &Base::func1 ]
//          [ &Base::func2 ]
//          [ NULL ]  // func3 不是虚函数，不在 vtable 中

class Derived : public Base {
public:
    void func1() override { std::cout << "Derived::func1" << std::endl; }
    virtual void func4() { std::cout << "Derived::func4" << std::endl; }
};

// Derived 类的 vtable 结构
// vptr ──► [ &Derived::func1 ]  // 覆盖基类
//          [ &Base::func2 ]       // 继承基类
//          [ &Derived::func4 ]    // 新增虚函数
```

#### 内存布局
```cpp
class Base {
public:
    virtual void func1() {}
    virtual void func2() {}
    int base_data;
};

class Derived : public Base {
public:
    void func1() override {}
    virtual void func3() {}
    int derived_data;
};

// Base 对象内存布局
// ┌─────────────┐
// │   vptr      │  指向 Base 的 vtable
// ├─────────────┤
// │  base_data  │
// └─────────────┘

// Derived 对象内存布局
// ┌─────────────┐
// │   vptr      │  指向 Derived 的 vtable
// ├─────────────┤
// │  base_data  │
// ├─────────────┤
// │derived_data │
// └─────────────┘
```

### 3. 动态绑定过程

#### 调用过程
```cpp
// 编译器如何处理虚函数调用
Base* ptr = new Derived();
ptr->func1();  // 虚函数调用

// 编译器生成的代码（伪汇编）
mov eax, [ptr]          // 获取对象地址
mov ecx, [eax]          // 获取 vptr
mov edx, [ecx + 0]      // 获取 func1 在 vtable 中的地址
call edx                // 调用函数

// 对比普通函数调用
ptr->func3();  // 非虚函数调用
// 直接调用 Base::func3，不需要查表
call Base::func3
```

#### 多重继承
```cpp
class Base1 {

class Base2 {

class Derived : public Base1, public Base2 {

// │ vptr2       │  指向 Derived 的 vtable（Base2 部分）

// │ base2_data  │

```

### 4. 虚析构函数

#### 为什么需要虚析构函数
```cpp
// 不使用虚析构函数的问题
class Base {
public:
    ~Base() {
        std::cout << "Base destructor" << std::endl;
    }
};

class Derived : public Base {
public:
    ~Derived() {
        std::cout << "Derived destructor" << std::endl;
    }
};

// 问题：派生类析构函数不会被调用
Base* ptr = new Derived();
delete ptr;  // 只调用 Base::~Base()，内存泄漏！

// 使用虚析构函数
class Base {
public:
    virtual ~Base() {
        std::cout << "Base destructor" << std::endl;
    }
};

class Derived : public Base {
public:
    ~Derived() {
        std::cout << "Derived destructor" << std::endl;
    }
};

// 正确：派生类析构函数会被调用
Base* ptr = new Derived();
delete ptr;  // 先调用 Derived::~Derived()，再调用 Base::~Base()
```

### 5. 纯虚函数与抽象类

#### 纯虚函数
```cpp
// 定义纯虚函数
class Shape {
public:
    // 纯虚函数，没有实现
    virtual double area() = 0;
    virtual double perimeter() = 0;
};

// 抽象类不能实例化
// Shape shape;  // 编译错误！

// 派生类必须实现纯虚函数
class Rectangle : public Shape {
private:
    double width;
    double height;

public:
    Rectangle(double w, double h) : width(w), height(h) {}

    double area() override {
        return width * height;
    }

    double perimeter() override {
        return 2 * (width + height);
    }
};

class Circle : public Shape {
private:
    double radius;

public:
    Circle(double r) : radius(r) {}

    double area() override {
        return 3.14159 * radius * radius;
    }

    double perimeter() override {
        return 2 * 3.14159 * radius;
    }
};

// 使用多态
Shape* shapes[] = {
    new Rectangle(3, 4),
    new Circle(5)
};

for (auto shape : shapes) {
    std::cout << "Area: " << shape->area() << std::endl;
}
```

### 6. 虚继承

#### 菱形继承问题
```cpp
// 菱形继承
class A {
public:
    int data;
};

class B : public A {};
class C : public A {};

class D : public B, public C {
    // 问题：A::data 存在两份！
};

// D 对象内存布局
// ┌─────────────┐
// │   A::data   │  来自 B
// ├─────────────┤
// │   A::data   │  来自 C（重复！）
// └─────────────┘

// 使用虚继承解决
class A {
public:
    int data;
};

class B : virtual public A {};
class C : virtual public A {};

class D : public B, public C {
    // A::data 只有一份
};

// D 对象内存布局
// ┌─────────────┐
// │   vptr      │
// ├─────────────┤
// │   A::data   │  只有一份
// └─────────────┘
```

### 7. RTTI（运行时类型识别）

#### dynamic_cast
```cpp
class Base {
public:
    virtual ~Base() {}
};

class Derived : public Base {
public:
    void derivedMethod() {
        std::cout << "Derived method" << std::endl;
    }
};

void processObject(Base* ptr) {
    // 使用 dynamic_cast 进行安全转换
    Derived* derivedPtr = dynamic_cast<Derived*>(ptr);

    if (derivedPtr != nullptr) {
        // 转换成功
        derivedPtr->derivedMethod();
    } else {
        // 转换失败
        std::cout << "Not a Derived object" << std::endl;
    }
}

// 使用
Derived derived;
processObject(&derived);  // 转换成功

Base base;
processObject(&base);     // 转换失败
```

#### typeid
```cpp
#include <typeinfo>

void checkType(Base* ptr) {
    // 获取类型信息
    const std::type_info& info = typeid(*ptr);
    std::cout << "Type: " << info.name() << std::endl;

    // 比较类型
    if (info == typeid(Derived)) {
        std::cout << "Is Derived" << std::endl;
    } else if (info == typeid(Base)) {
        std::cout << "Is Base" << std::endl;
    }
}

// 使用
Derived derived;
checkType(&derived);  // 输出: Is Derived

Base base;
checkType(&base);     // 输出: Is Base
```

### 8. 多态的性能影响

#### 虚函数调用开销
```cpp
// 普通函数调用
class Normal {
public:
    void func() {}
};

Normal normal;
normal.func();
// 开销：1 次函数调用

// 虚函数调用
class Virtual {
public:
    virtual void func() {}
};

Virtual* vptr = new Virtual();
vptr->func();
// 开销：
// 1. 读取 vptr
// 2. 查找 vtable
// 3. 读取函数地址
// 4. 函数调用
// 约 2-3 倍于普通函数调用

// 性能测试
void performanceTest() {
    const int N = 10000000;

    // 测试普通函数
    Normal normal;
    auto start1 = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        normal.func();
    }
    auto end1 = std::chrono::high_resolution_clock::now();

    // 测试虚函数
    Virtual* vptr = new Virtual();
    auto start2 = std::chrono::high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        vptr->func();
    }
    auto end2 = std::chrono::high_resolution_clock::now();

    // 虚函数约慢 2-3 倍
}
```

### 9. 多态的应用场景

#### 1. 插件系统
```cpp
// 插件接口
class Plugin {
public:
    virtual void initialize() = 0;
    virtual void execute() = 0;
    virtual void cleanup() = 0;
    virtual ~Plugin() {}
};

// 具体插件
class AudioPlugin : public Plugin {
public:
    void initialize() override {
        std::cout << "Audio plugin initialized" << std::endl;
    }

    void execute() override {
        std::cout << "Audio plugin executing" << std::endl;
    }

    void cleanup() override {
        std::cout << "Audio plugin cleaned up" << std::endl;
    }
};

class VideoPlugin : public Plugin {
public:
    void initialize() override {
        std::cout << "Video plugin initialized" << std::endl;
    }

    void execute() override {
        std::cout << "Video plugin executing" << std::endl;
    }

    void cleanup() override {
        std::cout << "Video plugin cleaned up" << std::endl;
    }
};

// 插件管理器
class PluginManager {
private:
    std::vector<Plugin*> plugins;

public:
    void addPlugin(Plugin* plugin) {
        plugins.push_back(plugin);
        plugin->initialize();
    }

    void executeAll() {
        for (auto plugin : plugins) {
            plugin->execute();
        }
    }

    void cleanupAll() {
        for (auto plugin : plugins) {
            plugin->cleanup();
            delete plugin;
        }
        plugins.clear();
    }
};

// 使用
PluginManager manager;
manager.addPlugin(new AudioPlugin());
manager.addPlugin(new VideoPlugin());
manager.executeAll();
manager.cleanupAll();
```

#### 2. 策略模式
```cpp
// 策略接口
class SortStrategy {
public:
    virtual void sort(std::vector<int>& arr) = 0;
    virtual ~SortStrategy() {}
};

// 具体策略
class BubbleSort : public SortStrategy {
public:
    void sort(std::vector<int>& arr) override {
        // 冒泡排序实现
        for (size_t i = 0; i < arr.size(); i++) {
            for (size_t j = 0; j < arr.size() - i - 1; j++) {
                if (arr[j] > arr[j + 1]) {
                    std::swap(arr[j], arr[j + 1]);
                }
            }
        }
    }
};

class QuickSort : public SortStrategy {
public:
    void sort(std::vector<int>& arr) override {
        quickSort(arr, 0, arr.size() - 1);
    }

private:
    void quickSort(std::vector<int>& arr, int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }

    int partition(std::vector<int>& arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;

        for (int j = low; j < high; j++) {
            if (arr[j] < pivot) {
                i++;
                std::swap(arr[i], arr[j]);
            }
        }
        std::swap(arr[i + 1], arr[high]);
        return i + 1;
    }
};

// 上下文
class Sorter {
private:
    SortStrategy* strategy;

public:
    void setStrategy(SortStrategy* s) {
        strategy = s;
    }

    void sort(std::vector<int>& arr) {
        if (strategy) {
            strategy->sort(arr);
        }
    }
};

// 使用
Sorter sorter;
std::vector<int> arr = {64, 34, 25, 12, 22, 11, 90};

// 使用冒泡排序
sorter.setStrategy(new BubbleSort());
sorter.sort(arr);

// 使用快速排序
sorter.setStrategy(new QuickSort());
sorter.sort(arr);
```

### 10. 返回值类型不同可以作为多态的实现方式吗

**不能。** 返回值类型不能用于区分重载函数，也不能作为多态的条件。

#### 重载规则

```cpp
class Example {
public:
    int func(int x);
    double func(int x);
};
```

上面的代码编译错误：两个函数具有相同的参数列表，仅返回值类型不同，编译器无法在调用点根据返回值类型来选择正确的函数。

```cpp
Example e;
e.func(10);
```

编译器在解析 `e.func(10)` 时，只看函数名和参数列表，此时 `int func(int)` 和 `double func(int)` 完全无法区分。

#### 协变返回类型

虽然返回值类型不能用于重载，但 C++ 允许虚函数重写时返回类型**协变**（covariant）：

```cpp
class Base {
public:
    virtual Base* clone() const {
        return new Base(*this);
    }
};

class Derived : public Base {
public:
    Derived* clone() const override {
        return new Derived(*this);
    }
};
```

这里 `Derived::clone()` 返回 `Derived*` 而非 `Base*`，这是合法的协变返回类型，因为 `Derived*` 可以隐式转换为 `Base*`。但注意，这**不是**靠返回值类型来实现多态，多态仍然是通过虚函数表机制实现的。

### 11. 编译阶段怎么检查函数重载

#### 重载决议（Overload Resolution）的三个步骤

**第一步：候选函数查找（Candidate Finding）**

编译器找到所有同名函数：

```cpp
void print(int x);
void print(double x);
void print(const std::string& s);
void print(int x, int y);

print(42);
```

候选函数：上述4个 `print` 函数。

**第二步：可行函数筛选（Viability Filtering）**

编译器检查参数个数和类型是否可以匹配：

```cpp
print(42);
```

- `print(int x)` ✅ 精确匹配
- `print(double x)` ✅ int 可以隐式转为 double
- `print(const std::string& s)` ❌ int 无法转为 string
- `print(int x, int y)` ❌ 参数个数不匹配

可行函数：前两个。

**第三步：最佳匹配选择（Best Match Ranking）**

编译器对可行函数进行排名，选择最优匹配：

| 优先级 | 转换等级 | 示例 |
|--------|----------|------|
| 1 | 精确匹配 | `int → int`、数组→指针、左值→右值 |
| 2 | 类型提升 | `char → int`、`short → int`、`float → double` |
| 3 | 标准转换 | `int → double`、`double → int`（截断） |
| 4 | 用户定义转换 | 构造函数、转换运算符 |
| 5 | 省略号匹配 | `...` |

```cpp
void f(int x);
void f(double x);

f(42);
```

`f(int)` 是精确匹配，`f(double)` 需要标准转换，所以选择 `f(int)`。

#### 二义性

```cpp
void f(long x);
void f(unsigned int x);

f(42);
```

`int → long` 和 `int → unsigned int` 都是标准转换，优先级相同，编译器报二义性错误。

### 12. 虚函数的什么特性使得其支持重载

这里的"重载"实际指的是**重写（Override）**，即派生类重新定义基类的虚函数。

#### 关键特性：动态绑定

虚函数的核心特性是**通过虚函数表实现动态绑定**，使得函数调用在运行时根据对象的实际类型（而非指针/引用的静态类型）来决定调用哪个版本。

```cpp
class Base {
public:
    virtual void func() { std::cout << "Base" << std::endl; }
};

class Derived : public Base {
public:
    void func() override { std::cout << "Derived" << std::endl; }
};

Base* ptr = new Derived();
ptr->func();
```

编译器处理 `ptr->func()` 时：
1. 通过 `ptr` 访问对象的 `vptr`
2. 在 vtable 中查找 `func` 对应的槽位
3. 找到 `Derived::func` 的地址
4. 调用 `Derived::func`

#### 虚函数支持重写的条件

```cpp
class Base {
public:
    virtual void func(int x);
    virtual void func(double x) const;
};

class Derived : public Base {
public:
    void func(int x) override;
    void func(double x) const override;
};
```

两个虚函数签名不同，各自在 vtable 中有独立的槽位，派生类可以分别重写。

#### override 关键字的作用

```cpp
class Derived : public Base {
public:
    void func(int x, int y) override;
};
```

编译器会检查基类中是否存在签名完全匹配的虚函数，如果不存在则编译错误，防止因签名不匹配导致意外隐藏而非重写。

### 13. 一个类对象的大小由什么决定

#### 影响类对象大小的因素

**1. 非静态成员变量**

```cpp
class Empty {};

class Person {
    int age;
    double salary;
    std::string name;
};

sizeof(Empty);
sizeof(Person);
```

**2. 虚函数表指针（vptr）**

```cpp
class WithoutVirtual {
    int x;
};

class WithVirtual {
    int x;
    virtual void func();
};

sizeof(WithoutVirtual);
sizeof(WithVirtual);
```

**3. 内存对齐（Padding）**

```cpp
class Bad {
    char a;
    double b;
    char c;
};

class Good {
    double b;
    char a;
    char c;
};

sizeof(Bad);
sizeof(Good);
```

`Bad` 的布局：
```
┌───────┬─────────────────────┬───────┬──────────┐
│ char  │   7 bytes padding   │ double│   char   │ 7 bytes  │
│  (1)  │                     │  (8)  │   (1)    │ padding  │
└───────┴─────────────────────┴───────┴──────────┘
总计：1 + 7 + 8 + 1 + 7 = 24 字节
```

`Good` 的布局：
```
┌────────────────┬───────┬───────┬──────────┐
│    double      │ char  │ char  │ 6 bytes  │
│     (8)        │  (1)  │  (1)  │ padding  │
└────────────────┴───────┴───────┴──────────┘
总计：8 + 1 + 1 + 6 = 16 字节
```

**4. 继承**

```cpp
class Base {
    int x;
};

class Derived : public Base {
    int y;
};

sizeof(Derived);
```

**5. 虚继承**

```cpp
class A {
    int a;
};

class B : virtual public A {
    int b;
};

sizeof(B);
```

虚继承会引入虚基类表指针（vbptr），增加额外开销。

**6. 静态成员不影响对象大小**

```cpp
class WithStatic {
    int x;
    static int count;
};

sizeof(WithStatic);
```

静态成员存储在全局数据区，不计入对象大小。

#### 空类的大小

```cpp
class Empty {};

sizeof(Empty);
```

C++ 规定每个对象必须有唯一的地址，所以空类大小为 1 字节。

#### 总结

| 因素 | 影响 | 示例 |
|------|------|------|
| 非静态成员变量 | 累加各成员大小 | `int x;` → 4 字节 |
| 虚函数表指针 | 增加 1 个指针大小 | 64 位系统 → 8 字节 |
| 内存对齐 | 可能增加 padding | `char + double` → 可能 16 字节 |
| 继承 | 包含基类子对象 | 基类大小 + 派生类成员 |
| 虚继承 | 增加虚基类表指针 | 额外 8 字节（64 位） |
| 静态成员 | 不影响 | 存储在全局区 |

### 14. 虚函数里面父类和子类的交互

#### 场景1：子类调用父类方法

派生类可以通过 `Base::func()` 显式调用基类的方法：

```cpp
class Base {
public:
    virtual void init() {
        std::cout << "Base init" << std::endl;
        commonInit();
    }

    void commonInit() {
        std::cout << "Common init" << std::endl;
    }
};

class Derived : public Base {
public:
    void init() override {
        std::cout << "Derived init start" << std::endl;
        Base::init();
        std::cout << "Derived init end" << std::endl;
    }
};

Derived d;
d.init();
// 输出：
// Derived init start
// Base init
// Common init
// Derived init end
```

#### 场景2：父类指针调用子类重写的方法

通过基类指针/引用调用虚函数时，会动态绑定到派生类的实现：

```cpp
class Shape {
public:
    virtual double area() {
        return 0.0;
    }

    void printArea() {
        std::cout << "Area: " << area() << std::endl;
    }
};

class Rectangle : public Shape {
    double width, height;
public:
    Rectangle(double w, double h) : width(w), height(h) {}

    double area() override {
        return width * height;
    }
};

Shape* s = new Rectangle(10, 20);
s->printArea();
```

`printArea()` 内部调用 `area()` 时，虽然 `this` 是 `Shape*`，但 `area()` 是虚函数，会通过 vtable 绑定到 `Rectangle::area()`，输出 `Area: 200`。

#### 场景3：纯虚函数接口的交互

```cpp
class Interface {
public:
    virtual void processData() = 0;
    virtual ~Interface() = default;
};

class ConcreteA : public Interface {
public:
    void processData() override {
        std::cout << "ConcreteA processing" << std::endl;
    }
};

class ConcreteB : public Interface {
public:
    void processData() override {
        std::cout << "ConcreteB processing" << std::endl;
    }
};

Interface* objs[] = { new ConcreteA(), new ConcreteB() };
for (auto obj : objs) {
    obj->processData();
}
```

### 15. 构造函数能不能为虚函数

**不能。** 原因如下：

1. **vptr 尚未初始化**：构造函数执行时，对象的 vptr 正在被设置。在基类构造函数中，vptr 指向基类的 vtable；在派生类构造函数中，vptr 才指向派生类的 vtable。如果构造函数本身是虚函数，就无法通过 vtable 调用它。

2. **对象类型不完整**：虚函数依赖对象的完整类型信息，而构造函数执行时对象尚未完全构造。

3. **语义矛盾**：构造函数的作用是初始化对象（包括设置 vptr），如果构造函数本身需要通过 vptr 调用，就形成了鸡生蛋的问题。

```cpp
class Base {
public:
    // virtual Base() {}  // 编译错误！
    Base() {
        std::cout << "Base ctor" << std::endl;
    }
};
```

#### 构造函数中调用虚函数的陷阱

```cpp
class Base {
public:
    Base() {
        init();
    }
    virtual void init() {
        std::cout << "Base init" << std::endl;
    }
};

class Derived : public Base {
public:
    void init() override {
        std::cout << "Derived init" << std::endl;
    }
};

Derived d;
```

输出 `Base init` 而非 `Derived init`。在 Base 构造函数中调用 `init()` 时，vptr 还指向 Base 的 vtable，所以调用的是 `Base::init()`。

### 16. 静态成员函数能不能为虚函数

**不能。** 原因如下：

1. **没有 this 指针**：静态成员函数属于类，不属于对象实例，没有 `this` 指针。

2. **无法通过 vtable 调用**：虚函数的动态绑定依赖于通过对象访问 vptr，静态函数不与任何对象关联。

3. **不依赖对象实例**：静态函数可以通过 `ClassName::func()` 直接调用，此时根本没有对象，也就没有 vptr 可查。

```cpp
class Example {
public:
    // virtual static void func() {}  // 编译错误！

    static void staticFunc() {}
    virtual void virtualFunc() {}
};
```

#### 对比

| 特性 | 虚函数 | 静态成员函数 |
|------|--------|-------------|
| 关联对象 | 是（通过 this） | 否（属于类） |
| this 指针 | 有 | 无 |
| 动态绑定 | 是 | 否 |
| 可被重写 | 是 | 否 |
| vtable 条目 | 是 | 否 |

### 17. 纯虚函数和普通虚函数的区别

#### 核心区别

| 维度 | 纯虚函数 | 普通虚函数 |
|------|----------|-----------|
| 声明语法 | `virtual void func() = 0;` | `virtual void func();` |
| 是否有实现 | 可以没有（C++ 也允许有） | 必须有 |
| 所在类 | 抽象类，不能实例化 | 普通类，可以实例化 |
| 派生类要求 | **必须**重写才能实例化 | **可以**不重写，使用基类版本 |
| 设计意图 | 定义接口规范，强制派生类实现 | 提供默认行为，允许派生类选择性重写 |

#### 示例对比

```cpp
class AbstractShape {
public:
    virtual double area() = 0;
    virtual void describe() {
        std::cout << "I am a shape" << std::endl;
    }
};

class Circle : public AbstractShape {
    double radius;
public:
    Circle(double r) : radius(r) {}

    double area() override {
        return 3.14159 * radius * radius;
    }
};
```

`area()` 是纯虚函数，`Circle` **必须**实现它；`describe()` 是普通虚函数，`Circle` **可以**不重写，直接使用基类版本。

### 18. 嵌入式开发中虚函数的注意事项

在嵌入式开发中，动态多态的使用需谨慎。

#### 适用场景

- 设备驱动抽象层（统一的驱动接口）
- 通信协议栈（不同协议的统一处理）
- 用户界面框架（控件的多态渲染）

#### 避免场景

- 性能关键路径（中断处理函数）
- 资源受限系统（RAM 有限的 MCU）
- 实时性要求高的代码（vtable 查表时间不确定）

#### 替代方案

```cpp
// 1. 函数指针替代虚函数
struct DriverOps {
    void (*init)(void);
    int (*read)(void* buf, int len);
    void (*write)(const void* buf, int len);
};

// 2. 回调机制
typedef void (*Callback)(int event, void* data);
void registerCallback(Callback cb);

// 3. 静态多态（模板，编译时确定）
template <typename Device>
void process(Device& dev) {
    dev.init();
    dev.read();
}
```
