# FreeRTOS信号量使用

## 题目

FreeRTOS中信号量的使用场景有哪些？如何创建和使用二值信号量、计数信号量？

## 答案

### 信号量类型

**FreeRTOS提供三种信号量：**
- 二值信号量（Binary Semaphore）
- 计数信号量（Counting Semaphore）
- 互斥信号量（Mutex Semaphore）

### 二值信号量

**创建二值信号量：**

```c
#include "FreeRTOS.h"
#include "semphr.h"

SemaphoreHandle_t binary_semaphore = NULL;

void create_binary_semaphore() {
    binary_semaphore = xSemaphoreCreateBinary();
    
    if (binary_semaphore != NULL) {
        printf("Binary semaphore created successfully\n");
    } else {
        printf("Failed to create binary semaphore\n");
    }
}
```

**二值信号量使用场景：**

```c
void interrupt_handler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    xSemaphoreGiveFromISR(binary_semaphore, &xHigherPriorityTaskWoken);
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void waiting_task(void *pvParameters) {
    while(1) {
        if (xSemaphoreTake(binary_semaphore, portMAX_DELAY) == pdTRUE) {
            printf("Semaphore taken, processing event\n");
            
        }
    }
}

void setup_interrupt_sync() {
    create_binary_semaphore();
    
    xTaskCreate(waiting_task, "WaitTask", 
                128, NULL, 1, NULL);
}
```

### 计数信号量

**创建计数信号量：**

```c
SemaphoreHandle_t counting_semaphore = NULL;

void create_counting_semaphore() {
    UBaseType_t max_count = 10;
    UBaseType_t initial_count = 0;
    
    counting_semaphore = xSemaphoreCreateCounting(max_count, initial_count);
    
    if (counting_semaphore != NULL) {
        printf("Counting semaphore created: max=%u, initial=%u\n",
               max_count, initial_count);
    }
}
```

**资源池管理：**

```c
typedef struct {
    int buffer[10];
    SemaphoreHandle_t semaphore;
} BufferPool;

void init_buffer_pool(BufferPool *pool) {
    pool->semaphore = xSemaphoreCreateCounting(10, 10);
    
    for (int i = 0; i < 10; i++) {
        pool->buffer[i] = i;
    }
}

int acquire_buffer(BufferPool *pool, TickType_t timeout) {
    if (xSemaphoreTake(pool->semaphore, timeout) == pdTRUE) {
        for (int i = 0; i < 10; i++) {
            if (pool->buffer[i] != -1) {
                int buffer = pool->buffer[i];
                pool->buffer[i] = -1;
                return buffer;
            }
        }
    }
    return -1;
}

void release_buffer(BufferPool *pool, int buffer) {
    for (int i = 0; i < 10; i++) {
        if (pool->buffer[i] == -1) {
            pool->buffer[i] = buffer;
            xSemaphoreGive(pool->semaphore);
            break;
        }
    }
}
```

### 互斥信号量

**创建互斥信号量：**

```c
SemaphoreHandle_t mutex_semaphore = NULL;

void create_mutex() {
    mutex_semaphore = xSemaphoreCreateMutex();
    
    if (mutex_semaphore != NULL) {
        printf("Mutex created successfully\n");
    }
}
```

**共享资源保护：**

```c
int shared_variable = 0;

void task1(void *pvParameters) {
    while(1) {
        if (xSemaphoreTake(mutex_semaphore, portMAX_DELAY) == pdTRUE) {
            shared_variable++;
            printf("Task1: shared_variable = %d\n", shared_variable);
            xSemaphoreGive(mutex_semaphore);
        }
        vTaskDelay(1000 / portTICK_PERIOD_MS);
    }
}

void task2(void *pvParameters) {
    while(1) {
        if (xSemaphoreTake(mutex_semaphore, portMAX_DELAY) == pdTRUE) {
            shared_variable += 10;
            printf("Task2: shared_variable = %d\n", shared_variable);
            xSemaphoreGive(mutex_semaphore);
        }
        vTaskDelay(1500 / portTICK_PERIOD_MS);
    }
}

void setup_mutex_example() {
    create_mutex();
    
    xTaskCreate(task1, "Task1", 128, NULL, 1, NULL);
    xTaskCreate(task2, "Task2", 128, NULL, 1, NULL);
}
```

### 多任务同步

```c
SemaphoreHandle_t sync_semaphore;
#define NUM_TASKS 3

void synchronized_task(void *pvParameters) {
    int task_id = (int)pvParameters;
    
    printf("Task %d waiting for synchronization\n", task_id);
    
    if (xSemaphoreTake(sync_semaphore, portMAX_DELAY) == pdTRUE) {
        printf("Task %d synchronized\n", task_id);
        
        vTaskDelay(1000 / portTICK_PERIOD_MS);
        
        xSemaphoreGive(sync_semaphore);
    }
    
    vTaskDelete(NULL);
}

void setup_synchronization() {
    sync_semaphore = xSemaphoreCreateCounting(NUM_TASKS, 0);
    
    for (int i = 0; i < NUM_TASKS; i++) {
        xTaskCreate(synchronized_task, "SyncTask", 
                    128, (void*)i, 1, NULL);
    }
    
    vTaskDelay(1000 / portTICK_PERIOD_MS);
    
    for (int i = 0; i < NUM_TASKS; i++) {
        xSemaphoreGive(sync_semaphore);
        vTaskDelay(500 / portTICK_PERIOD_MS);
    }
}
```

```c
void check_semaphore_status() {
    UBaseType_t count = uxSemaphoreGetCount(counting_semaphore);
```


### 实际应用：按键消抖

```c
SemaphoreHandle_t button_semaphore;

void button_isr(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xSemaphoreGiveFromISR(button_semaphore, &xHigherPriorityTaskWoken);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void button_task(void *pvParameters) {
    const TickType_t debounce_delay = pdMS_TO_TICKS(50);
    
    while(1) {
        if (xSemaphoreTake(button_semaphore, portMAX_DELAY) == pdTRUE) {
            vTaskDelay(debounce_delay);
            
            if (button_pressed()) {
                printf("Button pressed and debounced\n");
                handle_button_press();
            }
        }
    }
}
```

### 总结

**信号量使用场景：**
- 二值信号量：中断与任务同步、事件通知
- 计数信号量：资源池管理、多事件计数
- 互斥信号量：互斥访问共享资源

**关键要点：**
- 中断中使用GiveFromISR
- 注意优先级继承问题
- 合理设置超时时间
- 避免死锁情况

**最佳实践：**
- 使用明确的超时时间
- 检查返回值
- 及时释放信号量
- 避免嵌套获取
