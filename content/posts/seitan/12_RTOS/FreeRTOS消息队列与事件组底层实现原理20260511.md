# FreeRTOS消息队列与事件组底层实现原理

## 题目

知道消息队列或者事件组的底层实现原理吗，有没有看过源码？

## 考察点

FreeRTOS 队列和事件组的源码级实现、数据结构、阻塞/唤醒机制。

## 回答要点

### 1. 消息队列（Queue）底层实现

#### 1.1 数据结构

```c
// queue.c 中的核心结构（简化）
typedef struct QueueDefinition {
    int8_t *pcHead;              // 队列存储区起始地址
    int8_t *pcTail;              // 队列存储区结束地址
    int8_t *pcWriteTo;           // 下一个写入位置
    int8_t *pcReadFrom;          // 下一个读取位置（指向最后读出项）

    List_t xTasksWaitingToSend;  // 等待发送的任务链表（队列满时）
    List_t xTasksWaitingToReceive; // 等待接收的任务链表（队列空时）

    volatile UBaseType_t uxMessagesWaiting; // 当前队列中的消息数
    UBaseType_t uxLength;        // 队列容量（最大消息数）
    UBaseType_t uxItemSize;      // 每条消息的大小（字节）

    int8_t *pcQueueStorage;      // 实际存储区域
} Queue_t;
```

#### 1.2 存储模型：环形缓冲区

```
队列存储区（连续内存）：

pcHead                              pcTail
  │                                    │
  ▼                                    ▼
┌──────┬──────┬──────┬──────┬──────┬──────┐
│ msg0 │ msg1 │ msg2 │ msg3 │ msg4 │      │
│  ✓   │  ✓   │  ✓   │      │      │      │
└──────┴──────┴──────┴──────┴──────┴──────┘
  ▲                          ▲
  │                          │
pcReadFrom                pcWriteTo
（下一个读取位置）        （下一个写入位置）

FIFO 模式：
  pcWriteTo 到达末尾 → 回绕到 pcHead
  pcReadFrom 到达末尾 → 回绕到 pcHead

uxMessagesWaiting = 3（当前有 3 条消息）
uxLength = 5（最多 5 条）
```

#### 1.3 发送流程（xQueueSend）

```c
BaseType_t xQueueGenericSend(QueueHandle_t xQueue,
                              const void *pvItemToQueue,
                              TickType_t xTicksToWait,
                              BaseType_t xCopyPosition) {
    Queue_t *pxQueue = (Queue_t *)xQueue;

    for (;;) {
        taskENTER_CRITICAL();
        {
            if (pxQueue->uxMessagesWaiting < pxQueue->uxLength) {
                // 队列未满，可以写入
                // 1. 拷贝数据到环形缓冲区
                prvCopyDataToQueue(pxQueue, pvItemToQueue, xCopyPosition);

                // 2. 检查是否有任务在等待接收
                if (listLIST_IS_NOT_EMPTY(&(pxQueue->xTasksWaitingToReceive))) {
                    // 唤醒等待接收的任务
                    if (xTaskRemoveFromEventList(&(pxQueue->xTasksWaitingToReceive)) != pdFALSE) {
                        // 被唤醒的任务优先级更高，触发调度
                        queueYIELD_IF_USING_PREEMPTION();
                    }
                }
                taskEXIT_CRITICAL();
                return pdPASS;
            }
        }
        taskEXIT_CRITICAL();

        // 队列满了，看是否需要阻塞等待
        if (xTicksToWait == 0) return errQUEUE_FULL;

        // 将当前任务挂到等待发送链表
        vTaskPlaceOnEventList(&(pxQueue->xTasksWaitingToSend), xTicksToWait);
        // 任务被阻塞，等待队列有空间或超时
    }
}
```

**关键点**：数据是**拷贝**进队列的（值传递），不是传指针。这意味着队列中存的是数据的副本，发送者发送后可以立即修改原始数据。

#### 1.4 接收流程（xQueueReceive）

```c
BaseType_t xQueueGenericReceive(QueueHandle_t xQueue,
                                 void *pvBuffer,
                                 TickType_t xTicksToWait,
                                 BaseType_t xJustPeeking) {
    Queue_t *pxQueue = (Queue_t *)xQueue;

    for (;;) {
        taskENTER_CRITICAL();
        {
            if (pxQueue->uxMessagesWaiting > 0) {
                // 队列非空，可以读取
                // 1. 从环形缓冲区拷贝数据到 pvBuffer
                prvCopyDataFromQueue(pxQueue, pvBuffer);

                if (!xJustPeeking) {
                    pxQueue->uxMessagesWaiting--;

                    // 2. 检查是否有任务在等待发送
                    if (listLIST_IS_NOT_EMPTY(&(pxQueue->xTasksWaitingToSend))) {
                        // 唤醒等待发送的任务
                        if (xTaskRemoveFromEventList(&(pxQueue->xTasksWaitingToSend)) != pdFALSE) {
                            queueYIELD_IF_USING_PREEMPTION();
                        }
                    }
                }
                taskEXIT_CRITICAL();
                return pdPASS;
            }
        }
        taskEXIT_CRITICAL();

        // 队列空，阻塞等待
        if (xTicksToWait == 0) return errQUEUE_EMPTY;

        vTaskPlaceOnEventList(&(pxQueue->xTasksWaitingToReceive), xTicksToWait);
    }
}
```

### 2. 事件组（Event Group）底层实现

#### 2.1 数据结构

```c
// event_groups.c 中的核心结构
typedef struct EventGroupDef_t {
    EventBits_t uxEventBits;     // 24 位事件标志位（实际用 24 bit）
    List_t xTasksWaitingForBits; // 等待事件的任务链表
} EventGroup_t;

// 等待任务节点中存储的额外信息
typedef struct {
    ListItem_t xListItem;        // 链表节点
    EventBits_t uxBitsToWaitFor; // 等待哪些位
    BaseType_t xWaitForAllBits;  // 是等待全部位还是任一位
    BaseType_t xClearOnExit;     // 退出时是否清除位
} EventWaitItem_t;
```

#### 2.2 设置事件（xEventGroupSetBits）

```c
EventBits_t xEventGroupSetBits(EventGroupHandle_t xEventGroup,
                                const EventBits_t uxBitsToSet) {
    EventGroup_t *pxEventBits = (EventGroup_t *)xEventGroup;
    ListItem_t *pxListItem, *pxNext;
    BaseType_t xMatchFound = pdFALSE;

    taskENTER_CRITICAL();
    {
        // 1. 设置指定位
        pxEventBits->uxEventBits |= uxBitsToSet;

        // 2. 遍历等待链表，检查是否有任务满足条件
        pxListItem = listGET_HEAD_ENTRY(&(pxEventBits->xTasksWaitingForBits));

        while (pxListItem != listGET_END_MARKER(&(pxEventBits->xTasksWaitingForBits))) {
            pxNext = listGET_NEXT(pxListItem);

            // 检查等待条件是否满足
            if (prvTestWaitCondition(pxListItem)) {
                xMatchFound = pdTRUE;

                // 如果设置了 xClearOnExit，清除对应位
                if (等待任务的 xClearOnExit) {
                    pxEventBits->uxEventBits &= ~uxBitsToWaitFor;
                }

                // 将任务从等待链表移除，加入就绪链表
                xTaskRemoveFromEventList(pxListItem);
            }

            pxListItem = pxNext;
        }
    }
    taskEXIT_CRITICAL();

    if (xMatchFound) {
        // 触发调度（可能有更高优先级任务就绪）
        portYIELD_WITHIN_API();
    }

    return pxEventBits->uxEventBits;
}
```

#### 2.3 等待事件（xEventGroupWaitBits）

```c
EventBits_t xEventGroupWaitBits(EventGroupHandle_t xEventGroup,
                                 const EventBits_t uxBitsToWaitFor,
                                 const BaseType_t xClearOnExit,
                                 const BaseType_t xWaitForAllBits,
                                 TickType_t xTicksToWait) {
    EventGroup_t *pxEventBits = (EventGroup_t *)xEventGroup;

    taskENTER_CRITICAL();
    {
        // 1. 检查当前位是否已经满足条件
        if (prvTestWaitCondition(当前位 & uxBitsToWaitFor, xWaitForAllBits)) {
            // 满足，直接返回
            if (xClearOnExit) {
                pxEventBits->uxEventBits &= ~uxBitsToWaitFor;
            }
            taskEXIT_CRITICAL();
            return pxEventBits->uxEventBits;
        }

        // 2. 不满足，阻塞等待
        if (xTicksToWait > 0) {
            // 将当前任务加入等待链表
            // 记录：等待哪些位、是否等待全部、退出时是否清除
            vTaskPlaceOnEventList(&(pxEventBits->xTasksWaitingForBits),
                                  xTicksToWait);
        }
    }
    taskEXIT_CRITICAL();

    // 任务被阻塞...
    // 被唤醒后从这里继续执行
    return pxEventBits->uxEventBits;
}
```

#### 2.4 事件位只有 24 位的原因

```c
// FreeRTOS 事件组用 EventBits_t（通常 32 位）
// 但只有 24 位可用，高 8 位被内核使用：

//  bit 31-24: 内核内部使用（标记等待条件类型）
//  bit 23-0:  用户可用的事件位

#define eventCLEAR_EVENTS_ON_EXIT_BIT   0x01000000UL
#define eventUNBLOCKED_DUE_TO_BIT_SET   0x02000000UL
#define eventWAIT_FOR_ALL_BITS          0x04000000UL
#define eventEVENT_BITS_CONTROL_BYTES   0xFF000000UL  // 高 8 位
```

### 3. 队列 vs 事件组对比

| 方面 | 消息队列 | 事件组 |
|------|---------|--------|
| 数据传递 | ✅ 拷贝数据（值传递） | ❌ 只传递"发生了"的信号 |
| 缓冲区 | 环形缓冲区（多条消息排队） | 无缓冲（只有 24 个标志位） |
| 阻塞方式 | 满/空时阻塞 | 条件不满足时阻塞 |
| 等待条件 | 只等"有消息/有空间" | 可等任意位组合、全匹配/任一匹配 |
| 事件丢失 | 不会（缓冲区未满时） | 可能（连续 SetBit 但没人在 Wait） |
| 典型场景 | 传递数据包、命令 | 多事件同步、状态通知 |

### 4. 面试速记

- **队列**：环形缓冲区 + 两个等待链表（发送/接收），数据是拷贝的，信号量就是元素大小为 0 的队列
- **事件组**：24 位标志 + 一个等待链表，不传数据只传信号，支持"全部匹配/任一匹配"
- **核心机制**：所有 IPC 都基于 `List_t`（双向链表）和 `vTaskPlaceOnEventList`（将任务挂到等待链表并阻塞）
