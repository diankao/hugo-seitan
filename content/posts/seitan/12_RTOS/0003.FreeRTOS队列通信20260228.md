# FreeRTOS队列通信

## 题目

FreeRTOS中队列的创建和使用方法是什么？如何在任务间传递数据？

## 答案

### 队列创建

**基本队列创建：**

```c
#include "FreeRTOS.h"
#include "queue.h"

QueueHandle_t data_queue = NULL;

void create_queue() {
    UBaseType_t queue_length = 10;
    UBaseType_t item_size = sizeof(int);
    
    data_queue = xQueueCreate(queue_length, item_size);
    
    if (data_queue != NULL) {
        printf("Queue created successfully\n");
        printf("Queue length: %u, Item size: %u\n", 
               queue_length, item_size);
    } else {
        printf("Failed to create queue\n");
    }
}
```

**创建复杂类型队列：**

```c
typedef struct {
    int id;
    char message[50];
    float value;
    uint32_t timestamp;
} DataPacket;

QueueHandle_t packet_queue = NULL;

void create_packet_queue() {
    data_queue = xQueueCreate(5, sizeof(DataPacket));
    
    if (packet_queue != NULL) {
        printf("Packet queue created\n");
    }
}
```

### 队列发送

**阻塞发送：**

```c
void sender_task(void *pvParameters) {
    int data = 0;
    
    while(1) {
        if (xQueueSend(data_queue, &data, portMAX_DELAY) == pdPASS) {
            printf("Sent data: %d\n", data);
            data++;
        }
        
        vTaskDelay(1000 / portTICK_PERIOD_MS);
    }
}
```

**非阻塞发送：**

```c
void non_blocking_sender(void *pvParameters) {
    int data = 0;
    
    while(1) {
        if (xQueueSend(data_queue, &data, 0) == pdPASS) {
            printf("Sent data: %d\n", data);
            data++;
        } else {
            printf("Queue full, data %d not sent\n", data);
        }
        
        vTaskDelay(100 / portTICK_PERIOD_MS);
    }
}
```

**发送到队列前端：**

```c
void urgent_sender(void *pvParameters) {
    int urgent_data = 999;
    
    if (xQueueSendToFront(data_queue, &urgent_data, 
                         portMAX_DELAY) == pdPASS) {
        printf("Urgent data sent to front: %d\n", urgent_data);
    }
}
```

### 队列接收

**阻塞接收：**

```c
void receiver_task(void *pvParameters) {
    int received_data;
    
    while(1) {
        if (xQueueReceive(data_queue, &received_data, 
                         portMAX_DELAY) == pdPASS) {
            printf("Received data: %d\n", received_data);
        }
    }
}
```

**带超时的接收：**

```c
void timeout_receiver(void *pvParameters) {
    int received_data;
    TickType_t timeout = pdMS_TO_TICKS(2000);
    
    while(1) {
        if (xQueueReceive(data_queue, &received_data, 
                         timeout) == pdPASS) {
            printf("Received data: %d\n", received_data);
        } else {
            printf("Receive timeout\n");
        }
    }
}
```

**查看队列数据但不删除：**

```c
void peek_task(void *pvParameters) {
    int data;
    
    if (xQueuePeek(data_queue, &data, 
                   pdMS_TO_TICKS(100)) == pdPASS) {
        printf("Peeked data: %d\n", data);
```

### 队列操作示例

```c
QueueHandle_t cmd_queue = NULL;

typedef enum {
    CMD_LED_ON,
    CMD_LED_OFF,
    CMD_READ_SENSOR,
    CMD_CALIBRATE
} CommandType;

typedef struct {
    CommandType cmd;
    int parameter;
} Command;

void command_sender(void *pvParameters) {
    Command cmd;
    
    while(1) {
        cmd.cmd = CMD_LED_ON;
        cmd.parameter = 1;
        
        if (xQueueSend(cmd_queue, &cmd, 
                      pdMS_TO_TICKS(100)) == pdPASS) {
            printf("Sent LED_ON command\n");
        }
        
        vTaskDelay(1000 / portTICK_PERIOD_MS);
        
        cmd.cmd = CMD_LED_OFF;
        cmd.parameter = 0;
        
        if (xQueueSend(cmd_queue, &cmd, 
                      pdMS_TO_TICKS(100)) == pdPASS) {
            printf("Sent LED_OFF command\n");
        }
        
        vTaskDelay(1000 / portTICK_PERIOD_MS);
    }
}

void command_processor(void *pvParameters) {
    Command cmd;
    
    while(1) {
        if (xQueueReceive(cmd_queue, &cmd, 
                         portMAX_DELAY) == pdPASS) {
            switch(cmd.cmd) {
                case CMD_LED_ON:
                    printf("Processing: LED ON, param=%d\n", 
                           cmd.parameter);
                    turn_on_led(cmd.parameter);
                    break;
                    
                case CMD_LED_OFF:
                    printf("Processing: LED OFF\n");
                    turn_off_led();
                    break;
                    
                case CMD_READ_SENSOR:
                    printf("Processing: Read sensor\n");
                    read_sensor();
                    break;
                    
                case CMD_CALIBRATE:
                    printf("Processing: Calibrate\n");
                    calibrate_system();
                    break;
                    
                default:
                    printf("Unknown command\n");
            }
        }
    }
}

void setup_command_system() {
    cmd_queue = xQueueCreate(10, sizeof(Command));
    
    if (cmd_queue != NULL) {
        xTaskCreate(command_sender, "Sender", 
                    128, NULL, 1, NULL);
        xTaskCreate(command_processor, "Processor", 
                    256, NULL, 2, NULL);
    }
}
```

### 队列状态查询

```c
void check_queue_status() {
    UBaseType_t messages_waiting = uxQueueMessagesWaiting(data_queue);
    UBaseType_t spaces_available = uxQueueSpacesAvailable(data_queue);
    
    printf("Messages waiting: %u\n", messages_waiting);
    printf("Spaces available: %u\n", spaces_available);
    
    if (uxQueueMessagesWaiting(data_queue) == 0) {
        printf("Queue is empty\n");
    }
    
    if (uxQueueSpacesAvailable(data_queue) == 0) {
        printf("Queue is full\n");
    }
}
```

### 多发送者单接收者

```c
QueueHandle_t sensor_queue = NULL;

typedef struct {
    int sensor_id;
    float value;
    uint32_t timestamp;
} SensorData;

void sensor_task(void *pvParameters) {
    int sensor_id = (int)pvParameters;
    SensorData data;
    
    data.sensor_id = sensor_id;
    
    while(1) {
        data.value = read_sensor(sensor_id);
        data.timestamp = xTaskGetTickCount();
        
        if (xQueueSend(sensor_queue, &data, 
                      pdMS_TO_TICKS(100)) == pdPASS) {
            printf("Sensor %d sent: %.2f\n", 
                   sensor_id, data.value);
        } else {
            printf("Sensor %d: Queue full\n", sensor_id);
        }
        
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

void data_collector(void *pvParameters) {
    SensorData data;
    
    while(1) {
        if (xQueueReceive(sensor_queue, &data, 
                         portMAX_DELAY) == pdPASS) {
            printf("Collected from sensor %d: %.2f at %lu\n",
                   data.sensor_id, data.value, data.timestamp);
            
            process_sensor_data(data);
        }
    }
}

void setup_sensor_system() {
    sensor_queue = xQueueCreate(20, sizeof(SensorData));
    
    if (sensor_queue != NULL) {
        xTaskCreate(sensor_task, "Sensor1", 
                    128, (void*)1, 1, NULL);
        xTaskCreate(sensor_task, "Sensor2", 
                    128, (void*)2, 1, NULL);
        xTaskCreate(sensor_task, "Sensor3", 
                    128, (void*)3, 1, NULL);
        xTaskCreate(data_collector, "Collector", 
                    256, NULL, 2, NULL);
    }
}
```

### 队列在ISR中使用

```c
QueueHandle_t event_queue = NULL;

typedef struct {
    int event_type;
    int event_data;
} Event;

void gpio_isr(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    Event event = {EVENT_GPIO, gpio_read()};
    
    xQueueSendFromISR(event_queue, &event, 
                     &xHigherPriorityTaskWoken);
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void event_processor(void *pvParameters) {
    Event event;
    
    while(1) {
        if (xQueueReceive(event_queue, &event, 
                         portMAX_DELAY) == pdPASS) {
            printf("Processing event type %d\n", event.event_type);
            handle_event(event);
        }
    }
}
```

### 队列删除

```c
void cleanup_queue(QueueHandle_t queue) {
    if (queue != NULL) {
        vQueueDelete(queue);
        printf("Queue deleted\n");
    }
}

void reset_queue() {
    xQueueReset(data_queue);
    printf("Queue reset\n");
}
```

### 总结

**队列使用场景：**
- 任务间数据传递
- 命令模式实现
- 事件分发系统
- 传感器数据采集
- 生产者-消费者模式

**关键要点：**
- 合理设置队列长度和项目大小
- 使用超时避免死锁
- ISR中使用FromISR版本
- 检查返回值处理错误

**最佳实践：**
- 使用结构体传递复杂数据
- 设置合理的超时时间
- 监控队列使用情况
- 避免队列溢出
- 及时处理队列消息
