# 中断与任务间数据通信

## 标签
#操作系统 #面试题 #FreeRTOS #中断 #任务通信 #RTOS

## 问题描述
如何处理中断与任务之间的数据通信？

## 参考答案

### 基本概念

#### 中断与任务的关系

**中断**：
- 硬件或软件触发的异步事件
- 优先级高于所有任务
- 执行时间要尽可能短
- 不能调用阻塞函数

**任务**：
- 由调度器管理的执行单元
- 有不同的优先级
- 可以调用阻塞函数
- 执行时间可以较长

**通信挑战**：
- 中断上下文受限，不能使用阻塞API
- 数据一致性需要保证
- 性能和实时性要求

### 通信机制

#### 1. 信号量通知

**适用场景**：中断通知任务处理事件

```c
SemaphoreHandle_t xBinarySemaphore;

void UART_Init(void) {
    xBinarySemaphore = xSemaphoreCreateBinary();
    UART_Receive_IT();
}

void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    if (__HAL_UART_GET_FLAG(&huart1, UART_FLAG_RXNE)) {
        xSemaphoreGiveFromISR(xBinarySemaphore, &xHigherPriorityTaskWoken);
        __HAL_UART_CLEAR_FLAG(&huart1, UART_FLAG_RXNE);
    }
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    while(1) {
        xSemaphoreTake(xBinarySemaphore, portMAX_DELAY);
        ProcessUARTData();
    }
}
```

#### 2. 消息队列传输

**适用场景**：中断向任务传递数据

```c
QueueHandle_t xQueue;

void UART_Init(void) {
    xQueue = xQueueCreate(32, sizeof(uint8_t));
    UART_Receive_IT();
}

void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    uint8_t data = (uint8_t)(huart1.Instance->DR & 0xFF);
    
    xQueueSendFromISR(xQueue, &data, &xHigherPriorityTaskWoken);
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    uint8_t data;
    while(1) {
        if (xQueueReceive(xQueue, &data, portMAX_DELAY) == pdTRUE) {
            ProcessByte(data);
        }
    }
}
```

#### 3. 双缓冲区

**适用场景**：批量数据传输，减少中断开销

```c
#define BUFFER_SIZE 256

typedef struct {
    uint8_t buffer1[BUFFER_SIZE];
    uint8_t buffer2[BUFFER_SIZE];
    uint8_t *active_buffer;
    uint8_t *inactive_buffer;
    uint16_t index;
    SemaphoreHandle_t xSemaphore;
} DoubleBuffer_t;

DoubleBuffer_t rx_buffer;

void DoubleBuffer_Init(void) {
    rx_buffer.active_buffer = rx_buffer.buffer1;
    rx_buffer.inactive_buffer = rx_buffer.buffer2;
    rx_buffer.index = 0;
    rx_buffer.xSemaphore = xSemaphoreCreateBinary();
}

void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    uint8_t data = (uint8_t)(huart1.Instance->DR & 0xFF);
    
    rx_buffer.active_buffer[rx_buffer.index++] = data;
    
    if (rx_buffer.index >= BUFFER_SIZE) {
        uint8_t *temp = rx_buffer.active_buffer;
        rx_buffer.active_buffer = rx_buffer.inactive_buffer;
        rx_buffer.inactive_buffer = temp;
        rx_buffer.index = 0;
        
        xSemaphoreGiveFromISR(rx_buffer.xSemaphore, &xHigherPriorityTaskWoken);
    }
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    while(1) {
        xSemaphoreTake(rx_buffer.xSemaphore, portMAX_DELAY);
        ProcessBuffer(rx_buffer.inactive_buffer, BUFFER_SIZE);
    }
}
```

#### 4. 环形缓冲区

**适用场景**：连续数据流处理

```c
#define RING_BUFFER_SIZE 512

typedef struct {
    volatile uint16_t head;
    volatile uint16_t tail;
    uint8_t buffer[RING_BUFFER_SIZE];
} RingBuffer_t;

RingBuffer_t ring_buffer;

void RingBuffer_Init(RingBuffer_t *rb) {
    rb->head = 0;
    rb->tail = 0;
}

bool RingBuffer_Put(RingBuffer_t *rb, uint8_t data) {
    uint16_t next_head = (rb->head + 1) % RING_BUFFER_SIZE;
    if (next_head == rb->tail) {
        return false;  // 缓冲区满
    }
    rb->buffer[rb->head] = data;
    rb->head = next_head;
    return true;
}

bool RingBuffer_Get(RingBuffer_t *rb, uint8_t *data) {
    if (rb->head == rb->tail) {
        return false;  // 缓冲区空
    }
    *data = rb->buffer[rb->tail];
    rb->tail = (rb->tail + 1) % RING_BUFFER_SIZE;
    return true;
}

void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    uint8_t data = (uint8_t)(huart1.Instance->DR & 0xFF);
    
    RingBuffer_Put(&ring_buffer, data);
    xSemaphoreGiveFromISR(xTaskSemaphore, &xHigherPriorityTaskWoken);
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    uint8_t data;
    while(1) {
        if (RingBuffer_Get(&ring_buffer, &data)) {
            ProcessByte(data);
        } else {
            xSemaphoreTake(xTaskSemaphore, 10);
        }
    }
}
```

### 中断处理最佳实践

#### 1. 快速响应

**原则**：中断服务程序（ISR）要尽可能短

```c
// 不好的做法：ISR中做复杂处理
void UART_IRQHandler(void) {
    uint8_t data = UART_ReceiveByte();
    
    ParseProtocol(data);        // 复杂解析
    UpdateStateMachine(data);   // 状态机更新
    CheckChecksum(data);         // 校验计算
    LogData(data);              // 日志记录
    
    UART_ClearFlag();
}

// 好的做法：ISR只保存数据
void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    uint8_t data = UART_ReceiveByte();
    
    xQueueSendFromISR(xQueue, &data, &xHigherPriorityTaskWoken);
    UART_ClearFlag();
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}
```

#### 2. 数据一致性

**原则**：确保数据在中断和任务间传输时的一致性

```c
// 使用临界区保护共享数据
volatile uint32_t shared_counter;

void EXTI_IRQHandler(void) {
    UBaseType_t uxSavedInterruptStatus = taskENTER_CRITICAL_FROM_ISR();
    
    shared_counter++;  // 临界区保护
    
    taskEXIT_CRITICAL_FROM_ISR(uxSavedInterruptStatus);
}

void Task_ReadCounter(void *param) {
    uint32_t local_counter;
    
    taskENTER_CRITICAL();
    local_counter = shared_counter;  // 临界区保护
    taskEXIT_CRITICAL();
    
    ProcessCounter(local_counter);
}
```

#### 3. DMA传输

**原则**：使用DMA减少中断次数

```c
#define DMA_BUFFER_SIZE 1024
uint8_t dma_buffer[DMA_BUFFER_SIZE];
SemaphoreHandle_t xDMASemaphore;

void UART_DMA_Init(void) {
    xDMASemaphore = xSemaphoreCreateBinary();
    
    HAL_UART_Receive_DMA(&huart1, dma_buffer, DMA_BUFFER_SIZE);
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    xSemaphoreGiveFromISR(xDMASemaphore, &xHigherPriorityTaskWoken);
    
    HAL_UART_Receive_DMA(&huart1, dma_buffer, DMA_BUFFER_SIZE);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void UART_Task(void *param) {
    while(1) {
        xSemaphoreTake(xDMASemaphore, portMAX_DELAY);
        ProcessDMABuffer(dma_buffer, DMA_BUFFER_SIZE);
    }
}
```

### 实际应用示例

#### 传感器数据采集

```c
typedef struct {
    uint16_t adc_value;
    uint32_t timestamp;
} SensorData_t;

QueueHandle_t xSensorQueue;
SemaphoreHandle_t xADCSemaphore;

void ADC_Init(void) {
    xSensorQueue = xQueueCreate(32, sizeof(SensorData_t));
    xADCSemaphore = xSemaphoreCreateBinary();
    
    HAL_ADC_Start_DMA(&hadc1, (uint32_t*)adc_buffer, ADC_BUFFER_SIZE);
}

void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef *hadc) {
    static uint16_t index = 0;
    SensorData_t sensor_data;
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    
    sensor_data.adc_value = adc_buffer[index];
    sensor_data.timestamp = GetTimestamp();
    
    xQueueSendFromISR(xSensorQueue, &sensor_data, &xHigherPriorityTaskWoken);
    xSemaphoreGiveFromISR(xADCSemaphore, &xHigherPriorityTaskWoken);
    
    index = (index + 1) % ADC_BUFFER_SIZE;
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void Sensor_Task(void *param) {
    SensorData_t data;
    while(1) {
        if (xQueueReceive(xSensorQueue, &data, 100) == pdTRUE) {
            ProcessSensorData(data);
        }
    }
}
```

#### 网络数据接收

```c
typedef struct {
    uint8_t data[256];
    uint16_t length;
    uint32_t timestamp;
} NetworkPacket_t;

QueueHandle_t xNetworkQueue;

void WiFi_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    NetworkPacket_t packet;
    
    packet.length = WiFi_ReceivePacket(packet.data, sizeof(packet.data));
    packet.timestamp = GetTimestamp();
    
    if (packet.length > 0) {
        xQueueSendFromISR(xNetworkQueue, &packet, &xHigherPriorityTaskWoken);
    }
    
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void Network_Task(void *param) {
    NetworkPacket_t packet;
    while(1) {
        if (xQueueReceive(xNetworkQueue, &packet, portMAX_DELAY) == pdTRUE) {
            ProcessNetworkPacket(packet.data, packet.length);
        }
    }
}
```

### 性能优化

#### 1. 减少上下文切换

```c
// 使用事件组批量处理多个中断
EventGroupHandle_t xEventGroup;

#define UART_EVENT  (1 << 0)
#define TIMER_EVENT (1 << 1)
#define GPIO_EVENT  (1 << 2)

void UART_IRQHandler(void) {
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xEventGroupSetBitsFromISR(xEventGroup, UART_EVENT, &xHigherPriorityTaskWoken);
    portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void EventHandler_Task(void *param) {
    EventBits_t uxEvents;
    
    while(1) {
        uxEvents = xEventGroupWaitBits(
            xEventGroup,
            UART_EVENT | TIMER_EVENT | GPIO_EVENT,
            pdTRUE,  // 清除事件位
            pdFALSE, // 等待任意一个事件
            portMAX_DELAY
        );
        
        if (uxEvents & UART_EVENT) {
            ProcessUART();
        }
        
        if (uxEvents & TIMER_EVENT) {
            ProcessTimer();
        }
        
        if (uxEvents & GPIO_EVENT) {
            ProcessGPIO();
        }
    }
}
```

### 常见问题

#### 1. 数据丢失

**问题**：中断频率高，任务处理慢导致数据丢失

**解决方案**：
- 增大队列长度
- 使用双缓冲区
- 使用DMA传输
- 提高任务优先级

#### 2. 优先级反转

**问题**：低优先级任务持有高优先级任务需要的资源

**解决方案**：
- 使用互斥锁（优先级继承）
- 合理设计任务优先级
- 减少临界区执行时间

#### 3. 栈溢出

**问题**：中断中使用过多的栈空间

**解决方案**：
- ISR中减少局部变量
- 使用静态缓冲区
- 检查栈使用情况
- 合理设置中断栈大小
