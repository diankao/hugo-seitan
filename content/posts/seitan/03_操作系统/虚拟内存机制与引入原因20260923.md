# 虚拟内存机制与引入原因

## 题目

了解虚拟内存机制吗，为什么要引入？

## 考察点

内存管理、操作系统原理

## 回答要点

### 1. 虚拟内存的概念

#### 定义
虚拟内存是一种内存管理技术，它将物理内存（RAM）和磁盘空间结合起来，为每个进程提供一个独立的、连续的地址空间。

#### 地址空间
```
用户进程
├── 虚拟地址空间（0 - 4GB）
│   ├── 用户空间（0 - 3GB）
│   └── 内核空间（3GB - 4GB）
└── 物理内存
    ├── 页框 0
    ├── 页框 1
    └── ...

虚拟地址 → 页表 → 物理地址
```

### 2. 为什么要引入虚拟内存

#### 解决物理内存的局限性

**1. 内存不足问题**
```c
// 没有虚拟内存时的问题
void process_large_file() {
    // 文件大小 1GB，物理内存只有 512MB
    // 无法一次性加载整个文件
    char* buffer = malloc(1 * 1024 * 1024 * 1024);  // 失败！

    // 有虚拟内存后
    char* buffer = mmap(NULL, 1 * 1024 * 1024 * 1024,
                       PROT_READ, MAP_PRIVATE, fd, 0);
    // 成功！操作系统按需加载
}
```

**2. 内存碎片问题**
```
物理内存碎片：
[已用][空闲][已用][空闲][已用][空闲]
  ↑ 256KB  ↑ 128KB  ↑ 512KB
  需要连续 400KB？无法分配

虚拟内存：
连续的虚拟地址空间 → 映射到不连续的物理页
```

**3. 多进程隔离问题**
```c
// 没有虚拟内存时
void process_a() {
    int* ptr = 0x10000000;
    *ptr = 10;  // 可能覆盖进程 B 的数据！
}

void process_b() {
    int* ptr = 0x10000000;
    *ptr = 20;  // 覆盖了进程 A 的数据！
}

// 有虚拟内存后
// 进程 A 和 B 的 0x10000000 映射到不同的物理地址
```

### 3. 虚拟内存的核心机制

#### 分页机制
```c
// 页表项结构（32位系统）
typedef struct {
    unsigned int present    : 1;   // 页是否存在
    unsigned int rw         : 1;   // 读写权限
    unsigned int user       : 1;   // 用户/内核模式
    unsigned int pwt        : 1;   // 写穿透
    unsigned int pcd        : 1;   // 缓存禁用
    unsigned int accessed   : 1;   // 是否被访问
    unsigned int dirty      : 1;   // 是否被修改
    unsigned int pat        : 1;   // 页属性表
    unsigned int global     : 1;   // 全局页
    unsigned int available  : 3;   // 可用位
    unsigned int frame      : 20;  // 物理页框号
} page_table_entry_t;

// 虚拟地址到物理地址的转换
unsigned int virt_to_phys(unsigned int virt_addr) {
    // 1. 提取页目录索引（高10位）
    int dir_index = (virt_addr >> 22) & 0x3FF;

    // 2. 提取页表索引（中间10位）
    int table_index = (virt_addr >> 12) & 0x3FF;

    // 3. 提取页内偏移（低12位）
    int offset = virt_addr & 0xFFF;

    // 4. 查页表
    page_table_entry_t* pte = &page_tables[dir_index][table_index];

    // 5. 计算物理地址
    unsigned int phys_addr = (pte->frame << 12) | offset;

    return phys_addr;
}
```

#### 多级页表
```c
// 64位系统的多级页表
typedef enum {
    LEVEL_PML4,     // PML4 表（第4级）
    LEVEL_PDPT,     // 页目录指针表（第3级）
    LEVEL_PD,       // 页目录表（第2级）
    LEVEL_PT        // 页表（第1级）
} page_table_level_t;

// 遍历多级页表
unsigned int virt_to_phys_64(uint64_t virt_addr) {
    // 每级使用9位索引
    int indices[4];
    for (int i = 0; i < 4; i++) {
        indices[i] = (virt_addr >> (12 + i * 9)) & 0x1FF;
    }

    // 遍历4级页表
    uint64_t* table = read_cr3();  // 根页表
    for (int i = 0; i < 4; i++) {
        if (!table[indices[i]].present) {
            handle_page_fault(virt_addr);
            return 0;
        }
        table = (uint64_t*)(table[indices[i]].addr << 12);
    }

    // 计算物理地址
    uint64_t phys_addr = (table[indices[3]].addr << 12) |
                        (virt_addr & 0xFFF);

    return phys_addr;
}
```

### 4. 页面置换算法

#### LRU（最近最少使用）
```c
// LRU 页面替换
typedef struct {
    unsigned long timestamp;
    int page_num;
} lru_entry_t;

void lru_replace_page(void) {
    lru_entry_t lru_pages[MAX_PAGES];
    int oldest_index = 0;

    // 找到最久未使用的页面
    for (int i = 1; i < MAX_PAGES; i++) {
        if (lru_pages[i].timestamp < lru_pages[oldest_index].timestamp) {
            oldest_index = i;
        }
    }

    // 替换该页面
    swap_out(lru_pages[oldest_index].page_num);
    lru_pages[oldest_index].timestamp = get_current_time();
}
```

#### Clock 算法
```c
// Clock 算法（改进的 LRU）
typedef struct {
    int reference_bit;  // 访问位
    int modify_bit;     // 修改位
} clock_page_t;

void clock_replace_page(void) {
    clock_page_t pages[MAX_PAGES];
    int clock_hand = 0;

    while (1) {
        // 如果页面未被访问，替换它
        if (pages[clock_hand].reference_bit == 0) {
            if (pages[clock_hand].modify_bit == 1) {
                // 页面被修改过，写回磁盘
                swap_out(clock_hand);
            }
            // 替换页面
            swap_in(clock_hand);
            return;
        }

        // 清除访问位，移动指针
        pages[clock_hand].reference_bit = 0;
        clock_hand = (clock_hand + 1) % MAX_PAGES;
    }
}
```

### 5. TLB（Translation Lookaside Buffer）

#### TLB 结构
```c
// TLB 表项
typedef struct {
    unsigned int virtual_page_num;
    unsigned int physical_frame_num;
    unsigned int asid;          // 地址空间 ID
    int valid;
    int global;
} tlb_entry_t;

// TLB 查找
tlb_entry_t* tlb_lookup(unsigned int vpn, unsigned int asid) {
    for (int i = 0; i < TLB_SIZE; i++) {
        tlb_entry_t* entry = &tlb[i];
        if (entry->valid &&
            entry->virtual_page_num == vpn &&
            (entry->global || entry->asid == asid)) {
            return entry;  // TLB 命中
        }
    }
    return NULL;  // TLB 未命中
}
```

#### TLB 管理
```c
// TLB 刷新（切换进程时）
void tlb_flush(unsigned int asid) {
    for (int i = 0; i < TLB_SIZE; i++) {
        if (tlb[i].asid == asid && !tlb[i].global) {
            tlb[i].valid = 0;  // 标记为无效
        }
    }
}

// TLB 失效处理
void handle_tlb_miss(unsigned int vpn, unsigned int asid) {
    // 1. 查页表
    unsigned int pfn = walk_page_table(vpn, asid);

    // 2. 分配 TLB 表项
    tlb_entry_t* entry = allocate_tlb_entry();

    // 3. 更新 TLB
    entry->virtual_page_num = vpn;
    entry->physical_frame_num = pfn;
    entry->asid = asid;
    entry->valid = 1;
}
```

### 6. 页面错误处理

#### 缺页异常处理
```c
// 缺页异常处理程序
void handle_page_fault(unsigned long fault_addr, unsigned long error_code) {
    // 1. 判断错误类型
    if (!(error_code & 0x1)) {
        // 页不存在
        handle_page_not_present(fault_addr);
    } else if (error_code & 0x2) {
        // 写权限错误
        handle_write_protection(fault_addr);
    } else {
        // 读权限错误
        handle_read_protection(fault_addr);
    }
}

// 页面不在内存中的处理
void handle_page_not_present(unsigned long fault_addr) {
    // 1. 查找 VMA（虚拟内存区域）
    vma_t* vma = find_vma(fault_addr);
    if (!vma) {
        // 非法访问
        kill_current_process(SIGSEGV);
        return;
    }

    // 2. 分配物理页面
    unsigned long phys_page = alloc_page();

    // 3. 从磁盘加载页面（如果需要）
    if (vma->file_backed) {
        read_from_disk(vma->file, fault_addr, phys_page);
    }

    // 4. 更新页表
    update_page_table(fault_addr, phys_page);

    // 5. 刷新 TLB
    flush_tlb_entry(fault_addr);
}
```

### 7. 内存映射

#### mmap 实现
```c
// 创建内存映射
void* mmap(void* addr, size_t length, int prot, int flags,
           int fd, off_t offset) {
    // 1. 验证参数
    if (length == 0) {
        return NULL;
    }

    // 2. 查找可用的虚拟地址
    if (addr == NULL) {
        addr = find_free_vma(length);
    }

    // 3. 创建 VMA
    vma_t* vma = create_vma(addr, length, prot, flags);

    // 4. 如果是文件映射，设置文件信息
    if (flags & MAP_SHARED && fd != -1) {
        vma->file = get_file(fd);
        vma->offset = offset;
    }

    // 5. 插入进程的 VMA 列表
    insert_vma(vma);

    return addr;
}

// 访问映射区域
void access_mapped_region(void* addr) {
    // 触发缺页异常
    // 操作系统处理缺页，加载页面
    // 然后正常访问
    int* ptr = (int*)addr;
    *ptr = 10;
}
```

### 8. 虚拟内存的优势

#### 1. 内存利用率
```c
// 按需加载
void demand_paging_example() {
    // 分配 1GB 虚拟内存
    char* buffer = malloc(1 * 1024 * 1024 * 1024);

    // 只访问前 4KB
    buffer[0] = 'A';

    // 只分配和映射了 1 个物理页（4KB）
    // 而不是整个 1GB
}
```

#### 2. 进程隔离
```c
// 进程间独立地址空间
void process_isolation_example() {
    pid_t pid = fork();

    if (pid == 0) {
        // 子进程
        int* ptr = malloc(sizeof(int));
        *ptr = 10;
        printf("Child: *ptr = %d\n", *ptr);
    } else {
        // 父进程
        int* ptr = malloc(sizeof(int));
        *ptr = 20;
        printf("Parent: *ptr = %d\n", *ptr);
    }
    // 两个进程的 ptr 指向不同的物理地址
}
```
