# 共享内存原理与 IPC 机制

## 题目

共享内存的原理？

## 考察点

进程间通信、内存管理

## 回答要点

### 1. 共享内存的概念

#### 定义
共享内存是一种高效的进程间通信（IPC）机制，允许多个进程访问同一块物理内存区域。

#### 工作原理
```
进程 A                    物理内存                   进程 B
┌─────────────┐            ┌─────────────┐            ┌─────────────┐
│ 虚拟地址空间 │            │  共享页面   │            │ 虚拟地址空间 │
│             │◄──────────►│             │◄──────────►│             │
│ 0x7FFF...   │            │  物理页框   │            │ 0x7FFF...   │
│             │            │             │            │             │
└─────────────┘            └─────────────┘            └─────────────┘
     ↑                                                     ↑
   mmap                                                  mmap
```

### 2. 共享内存的实现

#### 创建共享内存
```c
// 创建共享内存对象
int shm_create(const char* name, size_t size) {
    // 1. 创建共享内存对象
    int fd = shm_open(name, O_CREAT | O_RDWR, 0666);
    if (fd == -1) {
        perror("shm_open failed");
        return -1;
    }

    // 2. 设置共享内存大小
    if (ftruncate(fd, size) == -1) {
        perror("ftruncate failed");
        close(fd);
        return -1;
    }

    return fd;
}

// 映射共享内存
void* shm_map(int fd, size_t size) {
    // 1. 映射到进程地址空间
    void* addr = mmap(NULL, size, PROT_READ | PROT_WRITE,
                      MAP_SHARED, fd, 0);
    if (addr == MAP_FAILED) {
        perror("mmap failed");
        return NULL;
    }

    return addr;
}
```

#### 访问共享内存
```c
// 写入数据
void shm_write(void* shm_addr, const char* data) {
    // 1. 使用同步机制保护共享区域
    pthread_mutex_lock(&shm_mutex);

    // 2. 写入数据
    strncpy((char*)shm_addr, data, SHM_SIZE - 1);

    // 3. 释放锁
    pthread_mutex_unlock(&shm_mutex);
}

// 读取数据
void shm_read(void* shm_addr, char* buffer) {
    // 1. 使用同步机制保护共享区域
    pthread_mutex_lock(&shm_mutex);

    // 2. 读取数据
    strncpy(buffer, (const char*)shm_addr, SHM_SIZE - 1);

    // 3. 释放锁
    pthread_mutex_unlock(&shm_mutex);
}
```

### 3. 同步机制

#### 互斥锁
```c
// 共享内存中的互斥锁
typedef struct {
    pthread_mutex_t mutex;
    pthread_cond_t cond;
    int data_ready;
    char buffer[SHM_SIZE];
} shared_data_t;

// 初始化共享内存
void shm_init(shared_data_t* shm) {
    // 1. 初始化互斥锁（进程共享）
    pthread_mutexattr_t attr;
    pthread_mutexattr_init(&attr);
    pthread_mutexattr_setpshared(&attr, PTHREAD_PROCESS_SHARED);
    pthread_mutex_init(&shm->mutex, &attr);

    // 2. 初始化条件变量
    pthread_condattr_t cond_attr;
    pthread_condattr_init(&cond_attr);
    pthread_condattr_setpshared(&cond_attr, PTHREAD_PROCESS_SHARED);
    pthread_cond_init(&shm->cond, &cond_attr);

    shm->data_ready = 0;
}

// 生产者
void producer(shared_data_t* shm, const char* data) {
    pthread_mutex_lock(&shm->mutex);

    // 等待消费者消费完
    while (shm->data_ready) {
        pthread_cond_wait(&shm->cond, &shm->mutex);
    }

    // 写入数据
    strncpy(shm->buffer, data, SHM_SIZE - 1);
    shm->data_ready = 1;

    // 通知消费者
    pthread_cond_signal(&shm->cond);
    pthread_mutex_unlock(&shm->mutex);
}

// 消费者
void consumer(shared_data_t* shm, char* buffer) {
    pthread_mutex_lock(&shm->mutex);

    // 等待数据
    while (!shm->data_ready) {
        pthread_cond_wait(&shm->cond, &shm->mutex);
    }

    // 读取数据
    strncpy(buffer, shm->buffer, SHM_SIZE - 1);
    shm->data_ready = 0;

    // 通知生产者
    pthread_cond_signal(&shm->cond);
    pthread_mutex_unlock(&shm->mutex);
}
```

#### 信号量
```c
// 使用信号量同步
#include <semaphore.h>

typedef struct {
    sem_t mutex;      // 互斥信号量
    sem_t empty;      // 空槽位信号量
    sem_t full;       // 满槽位信号量
    int count;
    char buffer[BUFFER_SIZE];
} shm_buffer_t;

// 初始化信号量
void shm_buffer_init(shm_buffer_t* buf) {
    sem_init(&buf->mutex, 1, 1);     // 初始值为1
    sem_init(&buf->empty, 1, BUFFER_SIZE);  // 初始值为缓冲区大小
    sem_init(&buf->full, 1, 0);      // 初始值为0
    buf->count = 0;
}

// 生产者
void produce(shm_buffer_t* buf, char item) {
    sem_wait(&buf->empty);   // 等待空槽位
    sem_wait(&buf->mutex);    // 获取互斥锁

    buf->buffer[buf->count++] = item;

    sem_post(&buf->mutex);    // 释放互斥锁
    sem_post(&buf->full);     // 增加满槽位
}

// 消费者
char consume(shm_buffer_t* buf) {
    sem_wait(&buf->full);     // 等待满槽位
    sem_wait(&buf->mutex);    // 获取互斥锁

    char item = buf->buffer[--buf->count];

    sem_post(&buf->mutex);    // 释放互斥锁
    sem_post(&buf->empty);    // 增加空槽位

    return item;
}
```

### 4. System V 共享内存

#### 创建和操作
```c
#include <sys/shm.h>

// 创建共享内存
int create_shm(key_t key, size_t size) {
    // 创建共享内存段
    int shmid = shmget(key, size, IPC_CREAT | 0666);
    if (shmid == -1) {
        perror("shmget failed");
        return -1;
    }
    return shmid;
}

// 附加共享内存
void* attach_shm(int shmid) {
    // 附加到进程地址空间
    void* shmaddr = shmat(shmid, NULL, 0);
    if (shmaddr == (void*)-1) {
        perror("shmat failed");
        return NULL;
    }
    return shmaddr;
}

// 分离共享内存
int detach_shm(void* shmaddr) {
    return shmdt(shmaddr);
}

// 删除共享内存
int remove_shm(int shmid) {
    return shmctl(shmid, IPC_RMID, NULL);
}
```

#### 使用示例
```c
// 生产者进程
void producer_process() {
    key_t key = ftok("/tmp/shmfile", 65);
    int shmid = create_shm(key, SHM_SIZE);

    char* shm_data = (char*)attach_shm(shmid);

    while (1) {
        // 读取输入
        char input[256];
        fgets(input, sizeof(input), stdin);

        // 写入共享内存
        strcpy(shm_data, input);

        if (strcmp(input, "exit\n") == 0) {
            break;
        }
    }

    detach_shm(shm_data);
}

// 消费者进程
void consumer_process() {
    key_t key = ftok("/tmp/shmfile", 65);
    int shmid = create_shm(key, SHM_SIZE);

    char* shm_data = (char*)attach_shm(shmid);

    while (1) {
        // 从共享内存读取
        if (strlen(shm_data) > 0) {
            printf("Received: %s", shm_data);

            if (strcmp(shm_data, "exit\n") == 0) {
                break;
            }

            shm_data[0] = '\0';  // 清空
        }
        sleep(1);
    }

    detach_shm(shm_data);
    remove_shm(shmid);
}
```

### 5. 共享内存的特点

#### 优点
```c
// 1. 最快的 IPC 机制
void performance_comparison() {
    // 管道：需要内核中转
    write(pipe_fd, data, size);  // 用户态→内核态
    read(pipe_fd, buffer, size); // 内核态→用户态

    // 消息队列：需要消息拷贝
    msgsnd(msgqid, &msg, sizeof(msg), 0);  // 拷贝到内核
    msgrcv(msgqid, &msg, sizeof(msg), 0, 0);  // 从内核拷贝

    // 共享内存：直接访问，无需拷贝
    strcpy(shm_addr, data);  // 直接写入共享内存
    strcpy(buffer, shm_addr); // 直接从共享内存读取
}

// 2. 支持大数据传输
void large_data_transfer() {
    // 可以传输大块数据
    size_t large_size = 100 * 1024 * 1024;  // 100MB
    void* large_shm = create_shm(large_size);

    // 直接访问，无需多次拷贝
    memcpy(large_shm, source, large_size);
}
```

```c
void image_processing_example() {

    void* shm_image = create_shm(IMAGE_WIDTH * IMAGE_HEIGHT * 3);

    while (1) {
        // 采集图像
```

#### 2. 进程崩溃处理
```c
// 使用心跳机制检测进程崩溃
typedef struct {
    pid_t owner_pid;
    time_t last_heartbeat;
    int is_valid;
} shm_control_t;

// 更新心跳
void update_heartbeat(shm_control_t* ctrl) {
    ctrl->owner_pid = getpid();
    ctrl->last_heartbeat = time(NULL);
    ctrl->is_valid = 1;
}

// 检查进程是否存活
int is_process_alive(shm_control_t* ctrl) {
    if (!ctrl->is_valid) {
        return 0;
    }

    // 检查心跳超时
    if (time(NULL) - ctrl->last_heartbeat > HEARTBEAT_TIMEOUT) {
        ctrl->is_valid = 0;
        return 0;
    }

    // 发送信号检查进程
    if (kill(ctrl->owner_pid, 0) == -1) {
        ctrl->is_valid = 0;
        return 0;
    }

    return 1;
}
```
